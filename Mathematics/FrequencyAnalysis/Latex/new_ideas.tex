\section{Functional Equation}
We have
\begin{align}
F_j^M(\epsilon) &= \sum_{n=0}^{N-1} c_1w_n \e^{2\pi\i\epsilon_j n}
\end{align}
with $\epsilon_j = \nu - \frac{j}{N}$.
Given $R = \frac{F_{j+1}}{F_{j}}$ and $R^M \approx R$ solve
\begin{align}
0 &= F_{j+1}^M - RF_j^M
\end{align}
for $\epsilon$ and define $\nu^M := \epsilon + \frac{j}{N}$.
For a given signal $z_n$ both $j$ and $\nu $ are constants independent of the window function $w(x)$.
In integral approximation we have
\begin{align}
f(\epsilon) &:= \int_{0}^{1} w(x) \e^{2\pi\i \kla\left( N\epsilon - 1 \right) x} \, \mathrm{d}x - R\intx\int_{0}^{1} w(x) \e^{2\pi\i N\epsilon x}\, \mathrm{d}x
\end{align}
with $f(\epsilon) = 0$.
We want to minimize $D[w(x)] := |\nu^M[w(x)] - \nu|$, i.e. $\delta D[w(x)] = 0$.
Here we defined
\begin{align}
\nu^M[w(x)] &:= \frac{j}{N} + \epsilon^M[w(x)]
\end{align}
where $\epsilon^M$ is given by $f^{-1}(0)$.
Functional derivative of $D$ gives
\begin{align}
\frac{\delta D[w(x)]}{\delta w(x)} &= \frac{\nu^M[w(x)]}{D} \frac{\delta \nu^M [w(x)]}{\delta w(x)} = \frac{\nu^M[w(x)]}{D} \frac{\delta f^{-1}(0) [w(x)]}{\delta w(x)} = \\
&= \frac{\nu^M[w(x)]}{D} \klam\left[ \frac{\delta f(\epsilon) [w(x)]}{\delta w(x)} \right]^{-1}_{\epsilon = f^{-1}(0)} = 0
\end{align}
Here
\begin{align}
\frac{\delta f(\epsilon) [w(x)]}{\delta w(x)} &= \e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x},\\
0 &= 
\end{align}
Note that
\begin{align}
\frac{\delta f(\epsilon)}{\delta w} &= \e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x} + w(x) \klam\left[ \e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x} \right] 2\pi\i N x \frac{\delta \epsilon}{\delta w},\\
\frac{\delta f(\epsilon)}{\delta w} &= f'(\epsilon) \frac{\delta \epsilon}{\delta w} = \int_0^1 w(x) \klam\left[ \e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x} \right] 2\pi\i N x\,\mathrm{d}x  \frac{\delta \epsilon}{\delta w}.
\end{align}
This gives
\begin{align}
\frac{\delta \epsilon}{\delta w(x)} &= \frac{\e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x}}{f'(\epsilon) - w(x) \klam\left[ \e^{2\pi\i (N\epsilon - 1) x} - R\e^{2\pi\i N\epsilon x} \right] 2\pi\i N x }.
\end{align}
