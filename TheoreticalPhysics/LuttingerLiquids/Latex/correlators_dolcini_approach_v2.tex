\section{Bosonization identities}
We will make use of the identities
\begin{align}
H &= \vf \sum_k |k| b_k^\dagger b_k,\\
\klb*{b_k, b_{k'}^\dagger} &= \delta_{kk'},\\
b_k(t) &= b_k\powe{-\i |k|\vf t},\\
\bra*{b_k^\dagger b_{k'}} &= \delta_{kk'} \frac{1}{\powe{\beta \vf |k|} - 1},\\
\bra*{b_k b_{k'}} &= 0,\\
\phi(x,t) &= -\frac{\i}{2} \sqrt{\frac{2\pi}{L}} \sum_k \frac{\sqrt{|k|}}{k} \powe{-\alpha \frac{|k|}{2}} \kla*{b_k^\dagger \powe{\i (|k|\vf t - kx)} - b_k \powe{-\i (|k|\vf t - kx)}},\\
\theta(x,t) &= \frac{\i}{2} \sqrt{\frac{2\pi}{L}} \sum_k \frac{\sqrt{|k|}}{|k|} \powe{-\alpha \frac{|k|}{2}} \kla*{b_k^\dagger \powe{\i (|k|\vf t - kx)} - b_k \powe{-\i (|k|\vf t - kx)}}
\end{align}
valid in the limit $L\rightarrow \infty$.
To lighten the notation we set $\vf = 1$ for now.
Since there is no other constant with the dimension of a velocity reinserting it at the end will be simple.
There are several ways to compute the Green's functions, but following the path integral approach we construct all complicated correlators from the fundamental building blocks
\begin{align}
\bra*{\klb*{\phi(x,t) - \phi(0,0)}^2} &= F_1(x,t),\\
2\bra*{\phi(x,t)\theta(0,0) - \phi(0,0)\theta(0,0)} &= F_2(x,t),\\
\bra*{\klb*{\theta(x,t) - \theta(0,0)}^2} &= F_3(x,t),\\
2\bra*{\theta(x,t)\phi(0,0) - \theta(0,0)\phi(0,0)} &= F_4(x,t).
\end{align}

\section{Computing the fundamental correlators}
Exploiting translational invariance in space and time allows us to write
\begin{align}
\bra*{F(x,t)F(x',t')} &= \bra*{F(x-x',t-t')F(0,0)}
\end{align}
for any operator $F$.
This simplifies some of the following calculations and is also the reason we can restrict ourselves to functions of only one variable in both space and time.
\subsection{$F_1(x,t)$}
The calculations splits into two parts
\begin{align}
F_1(x,t) &= 2\bra*{\phi(0,0)^2 - \phi(x,t)\phi(0,0)} \\
&= 2 \frac{-1}{4} \frac{2\pi}{L} \sum_{k,k'} \frac{\sqrt{|kk'|}}{kk'} \powe{-\frac{\alpha}{2} \kla*{|k| + |k'|}} \bra*{-b_k^\dagger b_{k'} \kla*{1 - \powe{\i (|k|t-kx)}} - b_k b_{k'}^\dagger \kla*{1 - \powe{-\i (|k|t-kx)}}},\\
&= \frac{1}{2} \intr{ \frac{|k|}{k^2} \powe{-\alpha |k|} \klb*{\frac{1 - \powe{\i (|k|t-kx)}}{\powe{\beta |k|} - 1} + \frac{1 - \powe{-\i (|k|t-kx)}}{\powe{\beta |k|} - 1} + \kla*{1 - \powe{-\i (|k|t-kx)}}} }{k}\\
&= \intr{\frac{1}{|k|}\powe{-\alpha |k|} \frac{1 - \cos\kla*{|k|t - kx}}{\powe{\beta |k|} - 1}}{k} + \frac{1}{2} \intr{\frac{1}{|k|} \powe{-\alpha |k|} \klb*{1 - \powe{-\i (|k|t-kx)}}}{k}\\
&= \intx{0}{\infty}{\frac{1}{k} \powe{-\alpha k} \frac{1 - \cos\kla*{kt-kx}}{\powe{\beta k} - 1}}{k} + (x\rightarrow -x) + \intx{0}{\infty}{\frac{1}{k} \powe{-\alpha k} \klb*{1 - \powe{-\i kt} \cos(kx)}}{k}.
\end{align}
The second integral is somewhat easier and gives\footnote{$\frac{1}{k} = \intx{0}{\infty}{\powe{-bk}}{b}$}
\begin{align}
\text{II} &= \intx{0}{\infty}{\intx{0}{\infty}{\powe{-bk} \powe{-\alpha k} \klb*{1 - \frac{1}{2} \powe{\i k(x-t)} - \frac{1}{2} \powe{-\i k(x+t)}}}{k}}{b}\\
&= \intx{0}{\infty}{\klb*{\frac{1}{\alpha + b} - \frac{1}{2} \frac{1}{\alpha + b + \i (t-x)} - \frac{1}{2} \frac{1}{\alpha + b + \i (t+x)}}}{b}\\
&= \lim_{B\rightarrow \infty} \klb*{\log \kla*{\frac{\alpha + B}{\alpha}} - \frac{1}{2} \log\kla*{\frac{\alpha + \i (t-x) + B}{\alpha + \i(t-x)}} - \frac{1}{2} \log\kla*{\frac{\alpha + \i(t+x) + B}{\alpha + \i (t+x)}}}\\
&= \frac{1}{2} \log\kla*{\frac{(\alpha + \i t)^2 + x^2}{\alpha^2}}.
\end{align}
The first integral can be done using a similar approach.
The part depending on $z=t-x$ gives
\begin{align}
\text{I}_z &= \sum_{n\ge 1} \intx{0}{\infty}{ \intx{0}{\infty}{\powe{-bk} \powe{-\alpha k} \klb*{1 - \frac{1}{2} \powe{\i k z} - \frac{1}{2} \powe{-\i kz}} \powe{-\beta kn}}{k} }{b}\\
&= \sum_{n\ge 1} \intx{0}{\infty}{\klb*{\frac{1}{\alpha + b + \beta n} - \frac{1}{2} \frac{1}{\alpha + b + \beta n - \i z} - \frac{1}{2} \frac{1}{\alpha + b + \beta n + \i z}}}{b}\\
&= -\sum_{n\ge 1} \klb*{\log\kla*{\alpha + \beta n} - \frac{1}{2} \log\kla*{\alpha + \beta n - \i z} - \frac{1}{2} \log \kla*{\alpha + \beta n + \i z}}\\
&= -\frac{1}{2} \sum_{n\ge 1} \log\klb[\bigg]{\frac{(\alpha + \beta n)^2}{(\alpha + \beta n)^2 + z^2}}\\
&= -\frac{1}{2} \log \klb[\bigg]{\Gamma\kla*{1 + \frac{\alpha}{\beta} + \i \frac{z}{\beta}} \Gamma\kla*{1 + \frac{\alpha}{\beta} - \i \frac{z}{\beta}}} + \log \klb*{\Gamma\kla*{1+ \frac{\alpha}{\beta}}}.
\end{align}
Combining all terms yields\footnote{$\Gamma(1+\i z)\Gamma(1-\i z) = \i z \pi \csc(\pi\i z) = \pi z \csch(\pi z)$ and $\sinh(a-b)\sinh(a+b) = \sinh^2(a) - \sinh^2(b)$}
\begin{align}
F_1(x,t) &= \bsplit{&\frac{1}{2} \log \klb[\bigg]{ \Gamma\kla*{1 + \frac{\alpha}{\beta}}^4 \frac{(\alpha + \i t)^2 + x^2}{\alpha^2}} \\
&- \frac{1}{2} \log \klb[\bigg]{\Gamma\kla*{1 + \frac{\alpha}{\beta} + \i \frac{x-t}{\beta}} \Gamma\kla*{1 + \frac{\alpha}{\beta} - \i \frac{x-t}{\beta}}}\\
&- \frac{1}{2} \log \klb[\bigg]{\Gamma\kla*{1 + \frac{\alpha}{\beta} + \i \frac{x+t}{\beta}} \Gamma\kla*{1 + \frac{\alpha}{\beta} - \i \frac{x+t}{\beta}}}}\\
&\simeq \frac{1}{2} \log\klb*{\frac{(\alpha + \i t)^2 + x^2}{\alpha^2}} - \frac{1}{2} \log\klb*{\frac{\pi^2}{\beta^2} \kla*{x^2 - t^2}} + \frac{1}{2} \log\klb*{\sinh^2\kla*{\frac{\pi x}{\beta}} - \sinh^2\kla*{\frac{\pi t}{\beta}}}\\
&= \log\klb*{\frac{\beta}{\pi\alpha}} + \frac{1}{2} \log\klb*{\frac{x^2 - t^2 + \alpha^2 + 2\i\alpha t}{x^2 - t^2}} + \frac{1}{2} \log\klb*{\sinh^2\kla*{\frac{\pi x}{\beta}} - \sinh^2\kla*{\frac{\pi t}{\beta}}}.
\end{align}
\subsection{$F_2(x,t)$}
The calculation again splits into two parts
\begin{align}
F_2(x,t) &= 2\bra*{\phi(x,t)\theta(0,0) - \phi(0,0)\theta(0,0)}\\
&= 2 \frac{1}{4} \frac{2\pi}{L} \sum_{k,k'} \frac{\sqrt{|kk'|}}{k|k'|} \powe{-\frac{\alpha}{2} (|k|+|k'|)} \bra*{-b_k^\dagger b_{k'} \kla*{\powe{\i (|k|t - kx)} - 1} - b_k b_{k'}^\dagger \kla*{\powe{-\i (|k|t - kx)} - 1}}\\
&= \frac{1}{2} \intr{\frac{|k|}{k|k|} \powe{-\alpha |k|} \klb*{\frac{1-\powe{\i (|k|t - kx)}}{\powe{\beta |k|} - 1} + \frac{1-\powe{-\i (|k|t - kx)}}{\powe{\beta |k|} - 1} + \kla*{1 - \powe{-\i (|k|t - kx)}}}}{k}\\
&= \intr{\frac{1}{k} \powe{-\alpha |k|} \frac{1 - \cos\kla*{|k|t - kx}}{\powe{\beta |k|} - 1}}{k} - \frac{1}{2} \intr{\frac{\powe{\i kx}}{k} \powe{-\alpha |k|} \powe{-\i |k| t}}{k}\\
&= \intx{0}{\infty}{\frac{1}{k} \powe{-\alpha k} \frac{1 - \cos(kt-kx)}{\powe{\beta k} - 1}}{k} - (x\rightarrow -x) - \i \intx{0}{\infty}{\frac{\sin(kx)}{k} \powe{-\alpha k} \powe{-\i kt}}{k}.
\end{align}
We already calculated the first integral previously.
The second integral gives
\begin{align}
\text{II} &= -\frac{1}{2} \intx{0}{\infty}{\intx{0}{\infty}{\powe{-bk} \powe{-\alpha k} \powe{\i k(x-t)}}{k}}{b} - (x\rightarrow -x)\\
&= -\frac{1}{2} \intx{0}{\infty}{\frac{1}{\alpha + b + \i (t-x)}}{b} - (x\rightarrow - x)\\
&= -\frac{1}{2} \intx{0}{\infty}{\frac{2\i x}{\kla*{\alpha + b + \i t}^2 + x^2}}{b}\\
&= -\i \arctan\kla*{\frac{\alpha + \i t + b}{x}}\bigg\vert_{b=0}^{b=\infty}\\
&= -\i \frac{\pi}{2} + \i \arctan\kla*{\frac{\alpha + \i t}{x}} = -\i\frac{\pi}{2} + \frac{1}{2} \log\kla*{\frac{x - t + \i\alpha}{x + t - \i\alpha}}.
\end{align}
%c_quad(lambda k: np.sin(k*x)/k * np.exp(-alpha*k) * np.exp(-1j*k*t), 0, np.inf, limit=10000), np.pi/2 - np.arctan((alpha + 1j*t)/x)
Combining all terms yields
\begin{align}
F_2(x,t) &= \bsplit{&\frac{1}{2} \log\klb*{\frac{t-x - \i\alpha}{t+x - \i\alpha}} \\
&- \frac{1}{2} \log \klb[\bigg]{\Gamma\kla*{1 + \frac{\alpha}{\beta} + \i \frac{x-t}{\beta}} \Gamma\kla*{1 + \frac{\alpha}{\beta} - \i \frac{x-t}{\beta}}}\\
&+ \frac{1}{2} \log \klb[\bigg]{\Gamma\kla*{1 + \frac{\alpha}{\beta} + \i \frac{x+t}{\beta}} \Gamma\kla*{1 + \frac{\alpha}{\beta} - \i \frac{x+t}{\beta}}}}\\
&\simeq \frac{1}{2} \log\klb*{\frac{x-t + \i\alpha}{x-t} \frac{t+x}{t+x - \i\alpha}} - \frac{1}{2} \log \klb*{\sinh\kla*{\frac{\pi}{\beta} [x-t]}} + \frac{1}{2} \log \klb*{\sinh\kla*{\frac{\pi}{\beta} [x+t]}}.
\end{align}

\end{align}

\section{Single-particle Green's function of the sine-Gordon model}
In this section we want to extend our Hamiltonian by the sine-Gordon perturbation
\begin{align}
H_\Text{int} &= g\intr{\cos\kla*{2\sqrt{K} \phi}}{x}.
\end{align}
The model is said to be integrable, but thus far the problem of obtaining the exact two-point correlation functions has resisted a great many attempts.
Here we want to take a more modest approach and merely treat the term in the first few orders of a perturbation series.
Recall that given some Hamiltonian $H=H_0+H_\Text{int}$ we calculate expectation values via
\begin{align}
\bra*{\mathcal{O}} &= \frac{1}{\mathcal{Z}} \int \mathcal{O} \powe{\i S[\Phi]}\,\mathcal{D}\Phi
\end{align} 
where $\mathcal{Z} = \int \powe{\i S[\Phi]}\,\mathcal{D}\Phi$ is the full partition function, $\mathcal{O}$ is some operator and $\Phi$ represents the fields appearing in the action.
Since the action decomposes into  a free and an interacting part via $S=S_0 + S_\Text{int}$ we find
\begin{align}
\mathcal{Z} &= \int \powe{\i S_0[\Phi]} \powe{\i S_\Text{int}[\Phi]}\,\mathcal{D}\Phi = \mathcal{Z}_0 \bra*{\powe{\i S_\Text{int}[\Phi]}}_0
\end{align}
where $\bra*{\mathcal{O}}_0$ denotes the average with respect to $S_0$.
Similarly
\begin{align}
\bra*{\mathcal{O}} &= \frac{1}{\mathcal{Z}}\int \mathcal{O} \powe{\i S_\Text{int}[\Phi]} \powe{\i S_0[\Phi]} \,\mathcal{D}\Phi = \frac{\mathcal{Z}_0 \bra*{\mathcal{O}\powe{\i S_\Text{int}}}_0}{\mathcal{Z}_0 \bra*{\powe{\i S_\Text{int}}}_0} = \frac{1}{\bra*{\powe{\i S_\Text{int}}}_0} \bra*{\mathcal{O}\powe{\i S_\Text{int}}}_0.
\end{align}
Expanding into a power series then yields
\begin{align}
\bra*{\mathcal{O}} &= \frac{1}{1 + \i \bra*{S_\Text{int}}_0 + \frac{\i^2}{2} \bra*{S_\Text{int}^2}_0 + \dots} \klc*{\bra*{\mathcal{O}}_0 + \i\bra*{\mathcal{O}S_\Text{int}}_0 + \frac{\i^2}{2} \bra*{\mathcal{O} S_\Text{int}^2} + \dots}\\
&= \bra*{\mathcal{O}}_0 + \i\bra*{\mathcal{O}S_\Text{int}}_0^c + \frac{\i^2}{2} \bra*{\mathcal{O}S_\Text{int}^2}_0^c - \i^2 \bra*{\mathcal{O}S_\Text{int}}_0^c \bra*{S_\Text{int}}_0 + \dots
\end{align}
where $\bra*{AB}^c = \bra*{AB} - \bra*{A}\bra*{B}$ denotes the connected average.
While we have converted the calculation of a single expectation value into a somewhat complicated series expansion, we only ever evaluated averages with respect to the free action $S_0$.
In our context of bosonization and the single-particle Green's function we can even go one step further and write down an explicit formula for the most general kind of expectation value we will encounter.
Assuming that $\phi(x,t)$ and $\theta(x,t)$ have $\mathbb{C}$-valued commutators for any $x$ and $t$ and that both $\bra*{\phi(0,0)^2}_0$ and $\bra*{\theta(0,0)^2}_0$ diverge one has
\begin{align}
\bra[\Big]{\prod_j \powe{\i A_j\phi(x_j,t_j) + \i B_j\theta(x_j,t_j)}} &= \powe{\frac{1}{2} \sum_{j<k} \klb*{\kla*{A_jA_k + B_jB_k} F_1(x_j-x_k,t_j-t_k) - \kla*{A_jB_k + A_kB_j} F_2(x_j-x_k,t_j-t_k)}}
\end{align}
under the neutrality condition $\sum_jA_j=0=\sum_jB_j$.
If this condition is not met, the expectation value gives zero.
Note that we can express all quantities and integrals in terms of $\xi_\pm$ via
\begin{align}
S_\Text{int} &= \frac{g}{u}\intrnd{2}{\cos\kla*{2\sqrt{K}\phi}}{\xi},\\
\psi_\ell &= \frac{1}{\sqrt{2\pi\alpha}} \powe{-\i\varphi_\ell} = \frac{1}{\sqrt{2\pi\alpha}} \powe{-\i \ell \sqrt{K} \phi + \i\frac{1}{\sqrt{K}} \theta},\\
F(\xi) &= \log\klb*{\frac{\i\beta u}{\pi\alpha}} - \log\klb*{\csch\kla*{\frac{\pi \xi}{\beta u}}},\\
F_1(\xi) &= \log\klb*{
\frac{\i\beta u}{\pi\alpha}} - \frac{1}{2}\log\klb*{ \csch\kla*{\frac{\pi \xi_-}{\beta u}} \csch\kla*{\frac{\pi \xi_+}{\beta u}}}.
\end{align}
The general formula for the expectation value admits a recursion relation.
Writing $V_j = \powe{\i A_j\phi(\xi_j) + \i B_j \theta(\xi_j)}$ and $\Sigma_{jk} = \kla*{A_jA_k + B_jB_k}F_1(\xi_j-\xi_k) - \kla*{A_jB_k + A_kB_j}F_2(\xi_j-\xi_k)$ we have
\begin{align}
\bra[\Big]{\prod_{j=1}^{N+1} V_j } &= \powe{\frac{1}{2} \sum_{k=2}^{N+1} \sum_{j=1}^{k-1} \Sigma_{jk}} = \powe{\frac{1}{2} \sum_{k=2}^N \sum_{j=1}^{k-1} \Sigma_{jk} + \frac{1}{2}\sum_{j=1}^N \Sigma_{j,N+1} } = \bra[\Big]{\prod_{j=1}^NV_j} \powe{\frac{1}{2} \sum_{j=1}^N \Sigma_{j,N+1}}.
\end{align}
Another interesting observation is that swapping $V_1$ and $V_2$ only leads to the sign change $\xi_1 - \xi_2 \rightarrow \xi_2 - \xi_1$, leaving everything else unchanged.
Since $F^\ast(-\xi)=F(\xi)$ and $F_1^\ast(-\xi) = F_1(\xi)$ it follows that
\begin{align}
\bra[\Big]{\klc*{V_1, V_2} \prod_{j=3}^NV_j} &= 2\Re \klc*{\powe{\frac{1}{2} \Sigma_{12}}} \powe{\frac{1}{2} \sum_{j<k} \kla*{1 - \delta_{j1} \delta_{k2}} \Sigma_{jk}}.
\end{align}
Using these tools we will calculate the single-particle Green's function, i.e. we set $\mathcal{O} = \klc*{\psi_\ell(\xi), \psi_{\ell'}(0)^\dagger}$.
Since the perturbation only contains $\phi$ we can simply our general formula a bit\footnote{$A_{1} = -\ell \sqrt{K}$, $B_{1} = \frac{1}{\sqrt{K}}$, $\xi_{1} = \xi$ and $A_{2} = \ell' \sqrt{K}$, $B_{2} = -\frac{1}{\sqrt{K}}$, $\xi_{2} = 0$}
\begin{align}
\mathcal{E} &= \bra[\Big]{\klc*{\psi_\ell(\xi), \psi_{\ell'}(0)^\dagger} \prod_{j=3}^N \powe{\i A_j\phi(\xi_j)} }_0 \\
&= \frac{1}{2\pi\alpha} 2\Re \klc*{\powe{-\frac{1}{2} \klb*{\ell\ell' K + \frac{1}{K}}F_1(\xi) - \ell F_2(\xi)}} \powe{\frac{1}{2} \sum_{k=2}^{N} \sum_{j=1}^{k-1} \kla*{1 - \delta_{j1} \delta_{k2}} \Sigma_{jk}}.
\end{align}
We can also write down the expectation value more explicitly as
\begin{align}
\mathcal{E} &= \mathcal{C}_0 \mathcal{C} \powe{\frac{1}{2} \sum_{k=4}^N \sum_{j=3}^{k-1} A_jA_kF_1(\xi_j-\xi_k)},\\
\mathcal{C}_0 &= \frac{1}{\pi\alpha} \Re \klc*{\powe{-\frac{1}{2} \klb*{\ell\ell' K + \frac{1}{K}}F_1(\xi) - \frac{1}{2} \klb*{\ell + \ell'} F_2(\xi)}},\\
\mathcal{C} &= \prod_{k=3}^N \powe{-\frac{1}{2} \ell \sqrt{K} A_k F_1(\xi - \xi_k) - \frac{1}{2\sqrt{K}} A_k F_2(\xi - \xi_k)} \powe{\frac{1}{2} \ell' \sqrt{K} A_k F_1(-\xi_k) + \frac{1}{2\sqrt{K}} A_k F_2(-\xi_k)}
\end{align}
where the only relevant neutrality condition is $0=\frac{1}{\sqrt{K}}\sum_jA_j = -\ell + \ell' + \frac{1}{\sqrt{K}} \sum_{j=3}^N A_j$ since the one for the $B_j$ is met by construction.
Finally, we can replace $F_2$ by $F$ via
\begin{align}
\mathcal{C}_0 &= \frac{\delta_{\ell\ell'}}{\pi\alpha} \Re\klc*{\powe{- F(\xi_{-\ell})} \powe{-\frac{1}{2} \klb*{K + \frac{1}{K} - 2}F_1(\xi)}} + \frac{\delta_{\ell,-\ell'}}{\pi\alpha} \Re\klc*{\powe{\frac{1}{2} \klb*{K - \frac{1}{K}} F_1(\xi)}},\\
\mathcal{C} &= \prod_{k=3}^N \powe{-\frac{1}{2\sqrt{K}} \ell A_k F\kla{\xi_{-\ell} - \xi_{k,-\ell}}} \powe{\frac{1}{2\sqrt{K}} \ell'A_k F\kla{-\xi_{k,-\ell'}}} \powe{-\frac{1}{2} \ell \frac{K-1}{\sqrt{K}} A_k F_1(\xi - \xi_k)} \powe{\frac{1}{2} \ell' \frac{K-1}{\sqrt{K}} F_1(-\xi_k)}.
\end{align}
Note that we made use of the abbreviations $\xi = (\xi_+,\xi_-)$ and that $\xi_{k,\pm} \equiv ut_k \pm x_k \equiv \kla*{\xi_k}_\pm$.
\subsection{Zeroth order and first order}
We expand the Green's function into a power series of the form
\begin{align}
G_{\ell\ell'}^\r(\xi) &= \sum_{p\ge 0} \frac{\kla*{\i g}^p}{p!} G_{p,\ell\ell'}^\r(\xi).
\end{align}
The free contribution to the Green's function is
\begin{align}
G_{0,\ell\ell'}^\r(\xi) &= -\i\Theta(t) \mathcal{C}_0\\
&= -\i\Theta(t) \frac{\delta_{\ell\ell'}}{\pi\alpha} \Re \klc*{\powe{-F(\xi_{-\ell}) - MF_1(\xi)}}\\
&= -\i\Theta(t) \frac{\delta_{\ell\ell'}}{\pi\alpha} \Re \klc*{\csch\kla*{\frac{\pi\xi_{-\ell}}{\beta u}}^\frac{2+M}{2} \csch\kla*{\frac{\pi\xi_\ell}{\beta u}}^\frac{M}{2} \kla*{\frac{\pi\alpha}{\i\beta u}}^{1+M}}\\
&= \i\Theta(\xi_+)\Theta(\xi_-) \frac{\delta_{\ell\ell'} \sin\kla*{\frac{\pi M}{2}}}{\beta u} \kla*{\frac{\pi\alpha}{\beta u}}^{M} \csch\kla*{\frac{\pi\xi_{-\ell}}{\beta u}}^\frac{2+M}{2} \csch\kla*{\frac{\pi\xi_\ell}{\beta u}}^\frac{M}{2}
\end{align}
where we defined $M = \frac{1}{2} \klb*{K + \frac{1}{K} - 2}\ge 0$.
The Fourier transform gives
\begin{align}
G_{0,\ell\ell'}^\r(k,\omega) &= \i \delta_{\ell\ell'} \frac{\beta \sin\kla*{\frac{\pi M}{2}}}{2\pi^2} \kla*{\frac{\pi\alpha}{\beta u}}^{M} J_0\kla*{\frac{\beta u}{\pi} k_\ell, \frac{2 + M}{2}} J_0\kla*{\frac{\beta u}{\pi} k_{-\ell}, \frac{M}{2}}.
\end{align}
%prefactor = 1j*np.sin(np.pi*M/2) * (np.pi * alpha /(beta*u))**M * beta / (2*np.pi**2)
%k_p = (omega + u*k_vals) / (2*u)
%k_m = (omega - u*k_vals) / (2*u)
%j_0 = lambda z, K: 2**(K-1) * special.gamma(1-K) * special.gamma(K/2 - 1j * z/2) / special.gamma(1-K/2 - 1j * z/2)
%g_rr = prefactor * j_0(beta*u * k_p / np.pi, 1 + M/2) * j_0(beta*u * k_m / np.pi, M/2)
%g_ll = prefactor * j_0(beta*u * k_m / np.pi, 1 + M/2) * j_0(beta*u * k_p / np.pi, M/2)
where we defined $k_\pm = \frac{\omega \pm uk}{2u}$ and the integral
\begin{align}
J_0(z,K) &= \intx{0}{\infty}{\powe{\i zx} \csch^K(x)}{x} = 2^{K-1} B\kla*{1 - K, \frac{K - \i z}{2}}.
\end{align}
As shown above the first order correction only consists of the connected average $\bra*{\klc*{\psi_\ell, \psi_{\ell'}^\dagger} S_\Text{int}}_0^c$.
The neutrality condition immediately gives $\bra*{S_\Text{int}}_0 = 0$ so we are left with
\begin{align}
G_{1,\ell\ell'}^\r(\xi) &= -\i\Theta(t) \frac{1}{u} \sum_{s=\pm 1} \intrnd{2}{ \mathcal{E} }{\xi'},\\
\mathcal{E} &= \bra*{\klc*{\psi_\ell(\xi), \psi_{\ell'}(0)^\dagger} \powe{2s\i\sqrt{K} \phi(\xi')}}_0 = \mathcal{C}_0 \mathcal{C} \delta_{0, -\ell + \ell' + 2s}.
\end{align}
The Kronecker delta decomposes into $\delta_{\ell,-\ell'} \delta_{s,\ell}$ which simplifies the components to
\begin{align}
\mathcal{C}_0 &= \frac{1}{\pi\alpha} \Re\klc*{\powe{\frac{1}{2} \klb*{K - \frac{1}{K}} F_1(\xi)}} = \frac{1}{\pi\alpha} \Re \klc*{ \kla*{\frac{\pi\alpha}{\i\beta u}}^N \csch\kla*{\frac{\pi\xi_-}{\beta u}}^\frac{N}{2} \csch\kla*{\frac{\pi\xi_+}{\beta u}}^\frac{N}{2} }\\
&= \frac{\cos\kla*{\frac{\pi N}{2}}}{\pi\alpha} \kla*{\frac{\pi\alpha}{\beta u}}^N \Theta(\xi_- \xi_+) \csch\kla*{\frac{\pi\xi_-}{\beta u}}^\frac{N}{2} \csch\kla*{\frac{\pi\xi_+}{\beta u}}^\frac{N}{2},\\
\mathcal{C} &= \powe{-F(\xi_{-\ell} - \xi'_{-\ell})} \powe{-F(-\xi'_{\ell})} \powe{-(K-1) F_1(\xi - \xi')} \powe{-(K-1) F_1(-\xi')}\\
&= \kla*{\frac{\pi\alpha}{\i\beta u}}^{2K} \csch\kla*{\frac{\pi [\xi_{-\ell} + \xi_{-\ell}']}{\beta u}}^{K_+} \csch\kla*{\frac{\pi [\xi_{\ell} + \xi_{\ell}']}{\beta u}}^{K_-} \csch\kla*{\frac{\pi \xi_{\ell}'}{\beta u}}^{K_+} \csch\kla*{\frac{\pi \xi_{-\ell}'}{\beta u}}^{K_-} 
\end{align}
where we defined $N=\frac{1}{2} \klb*{\frac{1}{K} - K}\ge 0$ and $K_\pm = \frac{K \pm 1}{2}$.
Note that we replaced $\xi'\rightarrow -\xi'$ which is valid under the integral.
At this point we realize that the two-dimensional integral once again decomposes into the product of one-dimensional integrals, this time of the type
\begin{align}
I_1(x,K) &= \intr{\csch^\frac{K+1}{2}\kla*{x+x' - \i\delta} \csch^\frac{K-1}{2}\kla*{x' - \i\delta}}{x'}
\end{align}

%x=0.3; K=1.0; delta=1e-1

%#OUT: ((-0.8554485+2.6327998j), 3.85e-09)
where we have reinserted a regularization to ensure convergence.
One can numerically check that
\begin{align}
I_1(x,K) &\approx \kla*{1 - \powe{\pi\i (K-1)}} \frac{8^\frac{K-1}{2} B\kla*{\frac{K}{2}, 1-K} \sech^\frac{K-1}{2}(x)}{1 + \kla*{2^\frac{K-1}{2} - 1} \sech^\frac{K+7}{10}(x)}
\end{align}

is a very good approximation.
Thus the correlation function becomes
\begin{align}
G_{1,\ell\ell'}^\r(\xi) &= \bsplit{&-\i\Theta(\xi_+)\Theta(\xi_-) \frac{\delta_{\ell,-\ell'}\cos\kla*{\frac{\pi N}{2}}}{\pi\alpha u} \kla*{\frac{\pi\alpha}{\beta u}}^{N+2K} \powe{-\i\pi K} \csch\kla*{\frac{\pi \xi_-}{\beta u}}^\frac{N}{2} \csch\kla*{\frac{\pi \xi_+}{\beta u}}^\frac{N}{2}\\
&\times \frac{\beta u}{\pi} I_1\kla*{\frac{\pi\xi_{-\ell}}{\beta u}, K} \frac{\beta u}{\pi} I_1\kla*{\frac{\pi\xi_\ell}{\beta u}, K} }\\
&\approx \bsplit{&-\i\Theta(\xi_+)\Theta(\xi_-) \frac{\delta_{\ell,-\ell'}\beta \cos\kla*{\frac{\pi N}{2}}}{\pi^2} \kla*{\frac{\pi\alpha}{\beta u}}^{N+2K-1} \powe{-\i\pi K} \kla*{1 - \powe{\pi\i (K-1)}}^2 8^{K-1} \\
&\times \frac{\csch\kla*{\frac{\pi \xi_-}{\beta u}}^\frac{N}{2} \sech^\frac{K-1}{2}\kla*{\frac{\pi\xi_{-}}{\beta u}}}{1 + \kla*{2^\frac{K-1}{2} - 1} \sech^\frac{K+7}{10}\kla*{\frac{\pi\xi_{-}}{\beta u}}} \frac{\csch\kla*{\frac{\pi \xi_+}{\beta u}}^\frac{N}{2} \sech^\frac{K-1}{2}\kla*{\frac{\pi\xi_{+}}{\beta u}}}{1 + \kla*{2^\frac{K-1}{2} - 1} \sech^\frac{K+7}{10}\kla*{\frac{\pi\xi_{+}}{\beta u}}} B\kla*{\frac{K}{2}, 1-K}^2. }
\end{align}
The Fourier transform is given by
\begin{align}
G_{1,\ell\ell'}^\r(k,\omega) &= -\i \delta_{\ell,-\ell'} \frac{2^{3K-1} \beta^3u \cos\kla*{\frac{\pi N}{2}}}{2\pi^4} \kla*{\frac{\pi\alpha}{\beta u}}^{N+2K-1} J_1\kla*{\frac{\beta u}{\pi}k_+, K} J_1\kla*{\frac{\beta u}{\pi}k_-, K}
\end{align}
where we defined the integral
\begin{align}
J_1(z,K) &= \cos\kla*{\frac{\pi K}{2}} B\kla*{\frac{K}{2}, 1-K} \intx{0}{\infty}{\powe{\i zx} \frac{\csch(x)^{\frac{1}{2} \kla*{\frac{1}{K} - K}} \sech(x)^\frac{K-1}{2}}{1 + \kla*{2^\frac{K-1}{2} - 1} \sech(x)^\frac{K+7}{10}} }{x}.
\end{align}
