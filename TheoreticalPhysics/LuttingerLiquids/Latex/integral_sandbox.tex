\section{First model}
\subsection{Symmetry analysis of the self-energy}
Define $f_1(r,r') := F_1(r-r') + F_1(r')$ and $f_2(r,r') := F_2(r-r') - F_2(r')$.
Then to first order in $g$ the self-energy can be written as
\begin{align}
\Sigma(r) &= g\powe{-\frac{1}{2} \klb*{K + \frac{1}{K}}F_1(r)} \begin{pmatrix}
0 & \i I_+(r) \\ -\i I_-(r) & 0
\end{pmatrix}
\end{align}
where the special functions are slightly changed to
\begin{align}
I_\ell(r) &= \frac{1}{4} \intx{0}{\beta}{\intr{\powe{-Kf_1(r,r') - \ell f_2(r,r')} \nabla'^3 \kla*{Kf_1(r,r') + \ell f_2(r,r')}}{x'}}{\tau'}.
\end{align}
We have the following reflection identities
\begin{align}
f_1(x,-\tau, x',\tau') &= F_1(x-x',-\tau-\tau') + F_1(x',\tau') = f_1(x,\tau,x',-\tau'),\\
f_1(-x,\tau,x',\tau') &= F_1(-x-x',\tau-\tau') + F_1(x',\tau') = f_1(x,\tau,-x',\tau'),\\
f_2(x,-\tau,x',\tau') &= F_2(x-x',-\tau-\tau') - F_2(x',\tau') = \\
&= \kla*{F_2(-x+x',\tau + \tau') -\i\pi} - \kla*{F_2(-x',-\tau') - \i\pi} = f_2(-x,\tau,-x',-\tau'),\\
f_2(-x,\tau,x',\tau') &= F_2(-x-x',\tau-\tau') - F_2(x',\tau') = \\
&= \kla*{F_2(x+x',-\tau + \tau') -\i\pi} - \kla*{F_2(-x',-\tau') - \i\pi} = f_2(x,-\tau,-x',-\tau').
\end{align}
Defining $s=0^+ \sgn(t)$ we now find
\begin{align}
\Sigma^\dagger(k,\omega) &= \intrnd{2}{\powe{\i (kx-\omega t)} \klb*{\Sigma \kla*{-x, \i t - s}}^\dagger}{(x,t)}.
\end{align}
Since the self-energy in real time is
\begin{align}
\Sigma(x,t) &= \Sigma(x, \i t + s) = g\powe{-\frac{1}{2} \klb*{K + \frac{1}{K}} F_1(x,\i t + s)} \begin{pmatrix}
0 & \i I_+(x,\i t + s) \\ -\i I_-(x,\i t + s) & 0
\end{pmatrix}
\end{align}
we find the hermitian conjugate to be
\begin{align}
\Sigma^\dagger(x,t) &= g\powe{-\frac{1}{2} \klb*{K + \frac{1}{K}} F_1(x,-\i t + s)} \begin{pmatrix}
0 & \i \klb*{I_-(x,\i t + s)}^\ast \\ -\i \klb*{I_+(x,\i t + s)}^\ast & 0
\end{pmatrix}.
\end{align}
At this point we need to consider
\begin{align}
\klb*{I_\ell(x,\i t + s)}^\ast &= \frac{1}{4} \intx{0}{\beta}{\intr{\powe{-Kf_1(x,-\i t + s, x',\tau') + \ell f_2(x, -\i t + s,x',\tau')} \nabla'^3 \kla*{Kf_1 - \ell f_2} }{x'} }{\tau'} = \\
&= I_{-\ell}(x, -\i t + s)
\end{align}
which yields $\Sigma^\dagger(x,t) = \Sigma(x,-t)$ and thus
\begin{align}
\Sigma^\dagger(k,\omega) &= \intrnd{2}{\powe{\i (kx-\omega t)} \Sigma(-x, t) }{(x,t)}.
\end{align}
Since the prefactor of the self-energy is symmetric under $x\rightarrow -x$, the question of whether the self-energy in momentum space is hermitian boils down to the transformation of the special functions.

\end{align}

\subsection{Analytical solution of the special integrals}
In this section we want to analytically compute the integrals $I_\ell(r)$ in order to simplify the subsequent analytical continuation and Fourier transformation to momentum space.
Of course an exact solution is too much to hope for, but we will try to find a very good approximation.
This is reasonable as the integrand drops of very quickly around the two isolated peaks at $r'=0$ and $r'=r$.
First note that writing $f(r,r') = Kf_1(r,r') + \ell f_2(r,r')$ we have
\begin{align}
I_\ell(r) &= \frac{u}{4} \intx{0}{\beta}{\intr{\powe{-f(r,r')} \nabla'^3 f(r,r')}{x'}}{\tau'} = \frac{u}{8} \intx{0}{\beta}{\intr{\powe{-f(r,r')} \klb*{\nabla' f(r,r')}^3}{x'}}{\tau'}.
\end{align}
This is due to partial integration of the form
\begin{align}
\intr{\powe{-f(x)} f'''(x)}{x} &= -\intr{\powe{-f}(-f')f''}{x} = \intr{\powe{-f} (-f')^2f'}{x} + \intr{\powe{-f}(-f'') f'}{x}
\end{align}
and solving for the desired integral.
Due to this trick we only need the first derivatives\footnote{Note that $\arg(z) = \arg(x+\i y)$ has the derivatives $\partial_x \arg(z) = \frac{-y}{x^2 + y^2}$ and $\partial_y \arg(z) = \frac{x}{x^2 + y^2}$.}
\begin{align}
\nabla F_1(r) &= \frac{\sinh\kla*{\frac{\pi x}{\beta u}} \cosh\kla*{\frac{\pi x}{\beta u}} \frac{\pi}{\beta u}}{\sin^2\kla*{\frac{\pi\tau}{\beta}} + \sinh^2\kla*{\frac{\pi x}{\beta u}}},\\
\nabla F_2(r) &= -\i \frac{\tanh\kla*{\frac{\pi x}{\beta u}} \sech^2 \kla*{\frac{\pi x}{\beta u}} \frac{\pi}{\beta u}}{\tan^2\kla*{\frac{\pi\tau}{\beta} + \frac{\pi\alpha \sgn(\tau)}{\beta u}} + \tanh^2\kla*{\frac{\pi x}{\beta u}}}.
\end{align}
Introducing the variables $y = \frac{\pi\tau}{\beta} + \frac{\pi\alpha \sgn(\tau)}{\beta u}$ and $\xi = \frac{\pi x}{\beta u}$ we find
\begin{align}
\bsplit{I_\ell(y,\xi) = \frac{\pi}{8\beta u} \intx{0}{\pi}{\intr{&\kla*{\frac{\pi\alpha}{\beta u}}^{2K} \frac{\powe{-\i\ell\arg \klb*{\tan(y') + \i \tanh(\xi')}}}{\klb*{\sin^2(y') + \sinh^2(\xi')}^{K/2}} \frac{\powe{\i\ell\arg \klb*{\tan(y-y') + \i \tanh(\xi-\xi')}}}{\klb*{\sin^2(y-y') + \sinh^2(\xi-\xi')}^{K/2}} \times \\
& \times\klb[\bigg]{K\frac{\sinh(\xi')\cosh(\xi')}{\sin^2(y') + \sinh^2(\xi')} - K\frac{\sinh(\xi-\xi')\cosh(\xi-\xi')}{\sin^2(y-y') + \sinh^2(\xi-\xi')} + \\
&\qquad + \i\ell \frac{\tanh(\xi') \sech^2(\xi')}{\tan^2(y') + \tanh^2(\xi')} - \i\ell \frac{\tanh(\xi-\xi') \sech^2(\xi-\xi')}{\tan^2(y-y') + \tanh^2(\xi-\xi')}}^3}{\xi'} }{y'}.}
\end{align}
Using the definition of the complex argument we can write $\powe{\i \arg(z)} = \frac{z}{|z|}$.
Substituting $w = \cot(y)$ gives\footnote{$\sin^2(y') = \frac{1}{1 + w'^2}$, $\sin^2(y'-y) = \frac{(w'-w)^2}{(1+w^2)(1+w'^2)}$, $\tan^2(y') = \frac{1}{w'^2}$ and $\tan^2(y'-y) = \frac{(w'-w)^2}{(1 + ww')^2}$}
\begin{align}
\bsplit{I_\ell(w,v) = \frac{\pi}{8\beta u} \intr{\intr{&\kla*{\frac{\pi\alpha}{\beta u}}^{2K} \frac{1}{1 + w'^2} \klb*{\frac{1}{1 + w'^2} + \sinh^2(\xi')}^{-K/2} \times \\
&\times \klb*{\frac{(w'-w)^2}{(1+w^2)(1+w'^2)} + \sinh^2(\xi'-\xi)}^{-K/2} \times \\
&\times \frac{\frac{1}{w'} - \i\ell \tanh(\xi')}{\frac{1}{w'^2} + \tanh^2(\xi')} \frac{\frac{w - w'}{1 + ww'} - \i\ell \tanh(\xi'-\xi)}{\frac{(w'-w)^2}{(1+ww')^2} + \tanh^2(\xi'-\xi)} \times \\
&\times \klb[\Bigg]{K \frac{\sinh(\xi')\cosh(\xi')}{\frac{1}{1 + w'^2} + \sinh^2(\xi')} + K\frac{\sinh(\xi'-\xi) \cosh(\xi'-\xi)}{\frac{(w'-w)^2}{(1+w^2)(1+w'^2)} + \sinh^2(\xi'-\xi)} + \\
&\qquad + \i\ell \frac{\tanh(\xi') \sech^2(\xi')}{\frac{1}{w'^2} + \tanh^2(\xi')} + \i\ell \frac{\tanh(\xi'-\xi) \sech^2(\xi'-\xi)}{\frac{(w'-w)^2}{(1+ww')^2} + \tanh^2(\xi'-\xi)}}^3}{\xi'} }{w'}.}
\end{align}

\subsection{Cosecant 2d Fourier transform}
In order to compute the self-energy for the general case of $K\neq 1$ we need to solve integrals of the form
\begin{align}
I_{ab} &= \intx{0}{\pi}{\intr{ \frac{\powe{\i (kx - \omega y)}}{\sin^a(y + \i x) \sin^b(y - \i x)}}{x} }{y} \equiv J_{ab}(k,\omega) + J_{ba}(-k,\omega).
\end{align}
Note that
\begin{align}
\sin(y+\i x) &= \frac{\powe{\i y - x} - \powe{-\i y + x}}{2\i} = \powe{x - \i y} \frac{1 - \powe{ - 2x + 2\i y}}{-2\i}.
\end{align}
Then $v= 2y$ and $u = \powe{-2x}$ give
\begin{align}
J_{ab} &= (2\i)^{a+b} \intx{0}{\pi}{\intx{0}{\infty}{ \powe{-(x-\i y)a} \powe{-(x+\i y)b} \frac{\powe{\i (kx - \omega y)}}{ \kla*{1 - \powe{-2x+2\i y}}^a \kla*{1 - \powe{-2x-2\i y}}^b}}{x} }{y}\\
&= (2\i)^{a+b} \frac{1}{4} \intx{0}{2\pi}{ \powe{\i \frac{v}{2} (a -b - \omega)} \intx{0}{1}{u^{\frac{a+b - \i k}{2} - 1} \klb*{1 - u \powe{\i v}}^{-a} \klb*{1 - u \powe{-\i v}}^{-b} }{u} }{v}\\
&= (2\i)^{a+b} \frac{1}{4p} \intx{0}{2\pi}{\powe{\i vq } F_1\kla*{p; -b,-a;p+1; \powe{-\i v}, \powe{\i v}}}{v}
\end{align}
where $q = \frac{a-b-\omega}{2}$, $p = \frac{a+b-\i k}{2}$ and $F_1$ is the Appell hypergeometric function defined by
\begin{align}
F_1\kla*{a;b_1,b_2;c;x,y} &= \sum_{n,m\ge 0} \frac{x^m y^n}{m!n!} \frac{\Gamma(a+m+n)}{\Gamma(a)} \frac{\Gamma(c)}{\Gamma(c+m+n)} \frac{\Gamma(b_1 + m)}{\Gamma(b_1)} \frac{\Gamma(b_2 + n)}{\Gamma(b_2)}.
\end{align}

\subsubsection{Contour integration}
We already found
\begin{align}
J_{ab} &= (2\i)^{a+b} \frac{1}{4} \intx{0}{2\pi}{ \powe{\i \frac{v}{2} (a -b - \omega)} \intx{0}{1}{u^{\frac{a+b - \i k}{2} - 1} \klb*{1 - u \powe{\i v}}^{-a} \klb*{1 - u \powe{-\i v}}^{-b} }{u} }{v}.
\end{align}
Note that the $v$-integration corresponds to parametrizing the integral
\begin{align}
(2\i)^{a+b} \frac{1}{4\i} \oint_{|z|=1} z^{\frac{a+b-\omega}{2}-1} \intx{0}{1}{u^{\frac{a+b - \i k}{2} - 1} \klb*{1 - u z}^{-a} \klb*{z - u}^{-b} }{u} \,\d z
\end{align}
with $z=\powe{\i v}$.
This $z$-integration has poles at $z=u$, $z=\frac{1}{u}$ and $z=0$.
Now let $z=\alpha\powe{\i v}$ and consider the limit $\alpha\ll 1$
\begin{align}
J_{ab} &\simeq (2\i)^{a+b} \frac{1}{4} \alpha^\frac{a+b-\omega}{2} \intx{0}{2\pi}{ \powe{\i v \frac{a+b-\omega}{2}} \intx{0}{1}{u^{\frac{a+b - \i k}{2} - 1} \klb*{\alpha\powe{\i v} - u}^{-b} }{u} }{v} \\
