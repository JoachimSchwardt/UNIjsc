\section{Master formula}
The starting point for our analysis is the exact solution
\begin{align}
J_0(k,\i\omega_n^\F;b) &= \intx{0}{\beta}{\intr{\powe{\i\omega_n^\F\tau - \i kx} \csc(z_+)^{1+b} \csc (z_-)^b}{x} }{\tau} = -\i\frac{u\beta^2}{\pi^2} \sin(\pi b) I_0\kla*{w_-^\F; 1+b} I_0\kla*{w_+^\F; b},\\
I_0(z; b) &= \frac{2^{b-1}}{\pi} \Gamma(1-b) \Gamma\kla*{\frac{b+\i z}{2}} \Gamma\kla*{\frac{b - \i z}{2}} \sin\kla*{\pi\frac{b+\i z}{2}} = 2^{b-1} B\kla*{1-b, \frac{b-\i z}{2}}
\end{align}
where $z_{\pm} = \frac{\pi}{\beta u} \kla*{u\tau \pm \i x}$, $w_\pm^\F = \beta \frac{\i\omega_n^\F \pm uk}{2\pi}$ and $\omega_n^\F = \frac{(2n+1)\pi}{\beta}$ for $n\in\mathbb{Z}$.
We will similarly define the frequencies $\omega_n^\B = \frac{2n\pi}{\beta}$ and $w_\pm^\B = \beta \frac{\i\omega_n^\B \pm uk}{2\pi}$.

To make progress we now rewrite the second term in the bracket using our relations for $I_0$ as
\begin{align}
I_0\kla*{w_-^\B + \i, 1+b} &= -I_0\kla*{w_-^\B -\i, 1+b} \frac{b - \i w_-^\B}{b + \i w_-^\B}
\end{align}

\begin{align}

&= \sin(\pi b) \frac{u\beta^2}{\pi^2} I_0\kla*{w_-^\B; b} I_0\kla*{w_+^\B; b}.
\end{align}

\subsubsection{Residue integral first}
The integral has the structure
\begin{align}
\intr{\frac{L_0(k',\i\omega_m^\B; b)}{k'-z} \sin(k'a)}{k'}.
\end{align}
Depending on the sign of the exponent the counter must be closed in the upper or lower half of the complex plane.
For $\powe{+\i k' a}$ we find a residue at $k'=z$ with value
\begin{align}
\powe{\i za} L_0(z,\i\omega; b) \Theta(\Im z)
\end{align}
and for $\powe{-\i k'a}$ the value changes to
\begin{align}
-\powe{-\i za} L_0(z,\i\omega; b)\Theta(-\Im z)
\end{align}
where the sign change comes from the change of the orientation of the contour.
This part of the integral contributes
\begin{align}
\frac{2\pi\i}{2\i} \klc*{\powe{\i za} L_0(z,\omega; b) \Theta(\Im z) - (-1) \powe{-\i za} L_0(z,\omega; b)\Theta(-\Im z)} &= \pi L_0(z,\omega; b) \powe{\i a\Re z} \powe{-a\abs*{\Im z}}.
\end{align}
The other poles are contained in the function $L_0$ that we will rewrite as
\begin{align}
L_0 &= C_b \prod_\ell \frac{\Gamma\kla*{\frac{b}{2} - \i k_\ell}}{\Gamma\kla*{1-\frac{b}{2} - \i k_\ell}}
\end{align}
where $C_b = \sin(\pi b)2^{2b-2} \Gamma(1-b)^2$ and $k_\ell = \frac{\omega + \ell k}{4}$.\\
The poles are at
\begin{align}
-l &= \frac{b}{2} - \i k_\ell \qquad \Rightarrow\qquad k = -2\i\ell (2l+b) - \ell\omega.
\end{align}
This gives $k'=-\ell(4\i l + 2\i b + \omega)$ for $l\ge 0$.
For $\powe{+\i k'a}$ the residues are at $\Im k' > 0$ and have value
\begin{align}
\frac{\powe{\i k'a}}{k'-z} &C_b \lim \eta\frac{\Gamma\kla*{\frac{b}{2} - \i \frac{\omega + (-\ell 2\i b -\ell 4\i l - \ell \omega + \eta)}{4}}}{\Gamma\kla*{1-\frac{b}{2} - \i \frac{\omega + (-\ell 2\i b -\ell 4\i l - \ell \omega)}{4}}} \frac{\Gamma\kla*{\frac{b}{2} - \i \frac{\omega - (-\ell 2\i b -\ell 4\i l - \ell \omega)}{4}}}{\Gamma\kla*{1-\frac{b}{2} - \i \frac{\omega - (-\ell 2\i b -\ell 4\i l - \ell \omega)}{4}}}\\
&= -4\i C_b \frac{\powe{\i \Re k'a - |\Im k'|a}}{z + \ell (2\i b + \omega + 4\i l)} \frac{(-1)^l}{l!} \frac{1}{\Gamma(1-b-l)} \frac{\Gamma\kla*{b + l-\i\frac{\omega}{2}}}{\Gamma\kla*{1+l-\i\frac{\omega}{2}}}
\end{align}
