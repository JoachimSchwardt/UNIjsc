\section{Hamiltonian}
The sine-Gordon Hamiltonian is described by $H=H_0+H_\Text{int}$ where
\begin{align}
H_0 &= \frac{u}{2\pi} \intr{\normord{\klb*{K\kla*{\partial_x\theta}^2 + \frac{1}{K} \kla*{\partial_x\phi}^2}}}{x},\\
H_\Text{int} &= \frac{g}{2\pi\alpha} \sum_\ell \eta_{\ell}\eta_{-\ell} \intr{\powe{2\i\ell \phi}}{x}
\end{align}
$u$ is the renormalized Fermi velocity, $K$ the Luttinger parameter and $g$ a coupling constant and we set $\hbar=1$.
Assuming that $\phi(x,t)$ and $\theta(x,t)$ have $\mathbb{C}$-valued commutators for any $x$ and $t$ and that both $\bra*{\phi(0,0)^2}_0$ and $\bra*{\theta(0,0)^2}_0$ diverge one has
\begin{align}
\bra[\Big]{\prod_j \powe{\i A_j\phi(x_j,t_j) + \i B_j\theta(x_j,t_j)}} &= \powe{\frac{1}{2} \sum_{j<k} \klb*{\kla*{K A_jA_k + \frac{1}{K} B_jB_k} F_1(x_j-x_k,t_j-t_k) - \kla*{A_jB_k + A_kB_j} F_2(x_j-x_k,t_j-t_k)}}
\end{align}
under the neutrality condition $\sum_jA_j=0=\sum_jB_j$.
If this condition is not met, the expectation value gives zero.
Note that we can express all quantities in terms of the coordinates $\xi_\pm = ut\pm x$ via
\begin{align}
S_\Text{int} &= \frac{g}{4\pi\alpha u} \sum_\ell \eta_\ell \eta_{-\ell} \intrnd{2}{\powe{2\i\ell\phi}}{\xi},\\
\psi_\ell &= \frac{\eta_\ell}{\sqrt{2\pi\alpha}} \powe{-\i\varphi_\ell} = \frac{\eta_\ell}{\sqrt{2\pi\alpha}} \powe{-\i \ell \phi + \i \theta},\\
F(\xi_\ell) &= \log\klb*{\frac{\i\beta u}{\pi\alpha}} - \log\klb*{\csch\kla*{\frac{\pi \xi_\ell}{\beta u}}} = F_1(\xi) - \ell F_2(\xi),\\
F_1(\xi) &= \log\klb*{
\frac{\i\beta u}{\pi\alpha}} - \frac{1}{2}\log\klb*{ \csch\kla*{\frac{\pi \xi_-}{\beta u}} \csch\kla*{\frac{\pi \xi_+}{\beta u}}}
\end{align}
where we made use of the abbreviations $\xi = (\xi_+,\xi_-)$ and $\xi_{k,\pm} \equiv ut_k \pm x_k \equiv \kla*{\xi_k}_\pm$.
One can also check that $F^\ast(-\xi)=F(\xi)$ and $F_1^\ast(-\xi) = F_1(\xi)$.
\section{Single-particle Green's function in perturbation theory}
We will compute the time-ordered or causal Green's function defined as
\begin{align}
G_{\ell\ell'}^\c(x,t) &= -\i\mathcal{T} \bra*{\psi_\ell(x,t) \psi_{\ell'}^\dagger(0,0)}.
\end{align}
We expand the Green's function into a power series of the form
\begin{align}
G_{\ell\ell'}^\c(\xi) &= \sum_{m\ge 0} \frac{g^m}{m!} G_{m,\ell\ell'}^\c(\xi)
\end{align}
and neglect initial correlations, effectively meaning that we approximate $\powe{-\beta H} \approx \powe{-\beta H_0}$ in the thermal averages.
Including the full Hamiltonian would adjust the $F_1$ and $F_2$ functions.
Let us also introduce the dimensionless coordinate $y = \frac{\pi \xi}{\beta u}$ and write
\begin{align}
G_{\ell\ell'}^\c(\xi) &= -\i\Theta(t) \bra*{\psi_\ell(\xi) \psi_{\ell'}^\dagger(0)} + \i\Theta(-t) \bra*{\psi_{\ell'}^\dagger(0) \psi_\ell(\xi)} = \Theta(t) G_{\ell\ell'}^>(\xi) + \Theta(-t) G_{\ell'\ell}^<(\xi).
\end{align}
Note that the retarded Green's function is given by $G_{\ell\ell'}^\r(\xi) = \Theta(t) G_{\ell\ell'}^>(\xi) + \Theta(t) G_{\ell\ell'}^<(-\xi)^\ast$.
\subsection{Zeroth order}
The free contribution to the upper Green's function is
\begin{align}
G_{0,\ell\ell'}^>(\xi) &= -\i\bra*{\psi_\ell(\xi) \psi_{\ell'}^\dagger(0)}_0 = -\i \frac{1}{2\pi\alpha} \bra*{\cabx{-\ell&\ell'}{1&-1}{\xi&0}}_0 = -\i \frac{\delta_{\ell\ell'}}{2\pi\alpha} \powe{\frac{1}{2} \klb*{-K - \frac{1}{K}}F_1(\xi) - \ell F_2(\xi)}\\
&= -\i \frac{\delta_{\ell\ell'}}{2\pi\alpha} \powe{-MF_1(\xi)} \powe{-F(\xi_{-\ell})} = -\frac{\delta_{\ell\ell'}}{2\beta u} \kla*{\frac{\pi\alpha}{\beta u\i}}^M \csch(y_{-\ell})^\frac{2+M}{2} \csch(y_\ell)^\frac{M}{2}
\end{align}
where we defined $M = \frac{1}{2} \klb*{K + \frac{1}{K} - 2}\ge 0$.
The contribution to the lower Green's function is
\begin{align}
G_{0,\ell\ell'}^<(\xi) &= \i\bra*{\psi_{\ell}^\dagger(0) \psi_{\ell'}(\xi)}_0 = \i \frac{1}{2\pi\alpha} \bra*{\cabx{\ell&-\ell'}{-1&1}{0&\xi}}_0 = \i \frac{\delta_{\ell\ell'}}{2\pi\alpha} \powe{\frac{1}{2} \klb*{-K - \frac{1}{K}}F_1(-\xi) - \ell F_2(-\xi)}\\
&= \i \frac{\delta_{\ell\ell'}}{2\pi\alpha} \powe{-MF_1(-\xi)} \powe{-F(-\xi_{-\ell})} = \frac{\delta_{\ell\ell'}}{2\beta u} \kla*{\frac{\pi\alpha}{\beta u\i}}^M \csch(-y_{-\ell})^\frac{2+M}{2} \csch(-y_\ell)^\frac{M}{2}.
\end{align}
Note that all of the hyperbolic cosecant functions so far implicitly had an imaginary regularization of the form $\csch(x) = \csch(x-\i\delta)$, which will become relevant when taking complex conjugates or extracting minus signs.
For clarity we now define $\csch_\pm(x) \equiv \csch(x\pm \i\delta)$.
Let us evaluate the retarded Green's function to verify our calculation.
We have
\begin{align}
G_{0,\ell\ell'}^\r(\xi) &= -\i\Theta(t) \frac{\delta_{\ell\ell'}}{2\pi\alpha}\klc*{\powe{-MF_1(\xi) - F(\xi_{-\ell})} + \powe{-MF_1(\xi)^\ast - F(\xi_{-\ell})^\ast}}\\
&= -\i\Theta(t) \frac{\delta_{\ell\ell'}}{2\beta u} \kla*{\frac{\pi\alpha}{\beta u}}^M \Re \klc*{\powe{-\i\frac{\pi}{2} (M+1)} \csch_-(\xi_\ell)^\frac{M}{2} \csch_-(\xi_{-\ell})^\frac{2+M}{2}}\\
&= \i\Theta(t) \Theta(\xi_+\xi_-) \frac{\delta_{\ell\ell'} \sin\kla*{\frac{\pi M}{2}}}{2\beta u} \kla*{\frac{\pi\alpha}{\beta u}}^M \csch(\xi_\ell)^\frac{M}{2} \csch(\xi_{-\ell})^\frac{2+M}{2}.
\end{align}
Using $\Theta(t) = \Theta(\xi_++\xi_-)$ we may further simplify $\Theta(t) \Theta(\xi_+\xi_-) = \Theta(\xi_+)\Theta(\xi_-)$.
\subsection{First order}
The first order contribution to the upper Green's function is
\begin{align}
G_{1,\ell\ell'}^>(\xi) &= \i \cdot -\i \mathcal{T} \bra*{\psi_{\ell}(\xi) \psi_{\ell'}^\dagger(0) S_\Text{int}}_0\\
&= \frac{1}{(2\pi\alpha)^2 2u} \sum_{s=\pm 1} \eta_{\ell}\eta_{\ell'} \eta_s\eta_{-s} \intrnd{2}{ \mathcal{T} \bra*{ \cabx{-\ell& \ell'& 2s}{1& -1& 0}{\xi& 0& \xi'} }_0 }{\xi'}\\
&= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2} \intr{ \sum_{j=1}^3 \intx{t_{j-1}^>}{t_j^>}{g_{j,\ell}^>(\xi,\xi')}{t'} }{x'}
\end{align}
where we defined the times $t_0^> = -\infty$, $t_1^> = 0$, $t_2^> = t$ and $t_3^>=\infty$ as well as the three components
\begin{align}
g_{1,\ell}^> &= \bra*{\cabx{-\ell&-\ell&2\ell}{1&-1&0}{\xi&0&\xi'}}_0 = \powe{\frac{1}{2} \klb*{K - \frac{1}{K}}F_1(\xi)} \powe{-KF_1(\xi-\xi') - \ell F_2(\xi-\xi')} \powe{-KF_1(-\xi') + \ell F_2(-\xi')} \\
&= \powe{-NF_1(\xi)} \powe{-(K-1)F_1(\xi-\xi')} \powe{-F(\xi_{-\ell} - \xi_{-\ell}')} \powe{-(K-1)F_1(-\xi')} \powe{-F(-\xi_\ell')},\\
g_{2,\ell}^> &= \bra*{\cabx{-\ell&2\ell&-\ell}{1&0&-1}{\xi&\xi'&0}}_0 = \powe{-NF_1(\xi)} \powe{-(K-1)F_1(\xi-\xi')} \powe{-F(\xi_{-\ell} - \xi_{-\ell}')} \powe{-(K-1)F_1(\xi')} \powe{-F(\xi_\ell')},\\
g_{3,\ell}^> &= \bra*{\cabx{2\ell&-\ell&-\ell}{0&1&-1}{\xi'&\xi&0}}_0 = \powe{-NF_1(\xi)} \powe{-(K-1)F_1(\xi'-\xi)} \powe{-F(\xi_{-\ell}' - \xi_{-\ell})} \powe{-(K-1)F_1(\xi')} \powe{-F(\xi_\ell')},
\end{align}
with $N = \frac{1}{2}\klb*{\frac{1}{K} - K}$.
Similarly the lower Green's function is
\begin{align}
G_{1,\ell\ell'}^<(\xi) &= \i \cdot \i \mathcal{T} \bra*{\psi_{\ell}^\dagger(0) \psi_{\ell'}(\xi) S_\Text{int}}_0\\
&= -\frac{1}{(2\pi\alpha)^2 2u} \sum_{s=\pm 1} \eta_{\ell} \eta_{\ell'} \eta_s\eta_{-s} \intrnd{2}{ \mathcal{T} \bra*{ \cabx{\ell& -\ell'& 2s}{-1& 1& 0}{0& \xi& \xi'} }_0 }{\xi'}\\
&= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2} \intr{ \sum_{j=1}^3 \intx{t_{j-1}^<}{t_j^<}{g_{j,\ell}^<(\xi,\xi')}{t'} }{x'}
\end{align}
where we defined the times $t_0^< = -\infty$, $t_1^< = t$, $t_2^< = 0$ and $t_3^< = \infty$ as well as the three components
\begin{align}
g_{1,\ell}^< &= \bra*{\cabx{\ell&\ell&-2\ell}{-1&1&0}{0&\xi&\xi'}}_0 = \powe{\frac{1}{2} \klb*{K - \frac{1}{K}}F_1(-\xi)} \powe{-KF_1(\xi-\xi') + \ell F_2(\xi-\xi')} \powe{-KF_1(-\xi') - \ell F_2(-\xi')} \\
&= \powe{-NF_1(-\xi)} \powe{-(K-1)F_1(\xi-\xi')} \powe{-F(\xi_{\ell} - \xi_{\ell}')} \powe{-(K-1)F_1(-\xi')} \powe{-F(-\xi_{-\ell}')},\\
g_{2,\ell}^< &= \bra*{\cabx{\ell&-2\ell&\ell}{-1&0&1}{0&\xi'&\xi}}_0 = \powe{-NF_1(-\xi)} \powe{-(K-1)F_1(\xi'-\xi)} \powe{-F(\xi_{\ell}' - \xi_{\ell})} \powe{-(K-1)F_1(-\xi')} \powe{-F(-\xi_{-\ell}')},\\
g_{3,\ell}^< &= \bra*{\cabx{-2\ell&\ell&\ell}{0&-1&1}{\xi'&0&\xi}}_0 = \powe{-NF_1(-\xi)} \powe{-(K-1)F_1(\xi'-\xi)} \powe{-F(\xi_{\ell}' - \xi_{\ell})} \powe{-(K-1)F_1(\xi')} \powe{-F(\xi_{-\ell}')}.
\end{align}
Making the substitutions $x',t'\rightarrow -x',-t'$ we find
\begin{align}
G_{1,\ell\ell'}^<(-\xi) &= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2} \intr{ \sum_{j=1}^3 \intx{t_{j-1}^>}{t_j^>}{g_{4-j,\ell}^<(-\xi,-\xi')}{t'} }{x'}\\
&= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2} \intr{ \sum_{j=1}^3 \intx{t_{j-1}^>}{t_j^>}{g_{j,-\ell}^>(\xi,\xi')}{t'} }{x'}
\end{align}
and consequently the retarded Green's function becomes
\begin{align}
G_{1,\ell\ell'}^\r(\xi) &= -\Theta(t) \frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2} \intr{\sum_{j=1}^3 \intx{t_{j-1}^>}{t_j^>}{ g_{j,\ell}^\r }{t'} }{x'}
\end{align}
where we defined the components $g_{j,\ell}^\r = g_{j,\ell}^> + \kla*{g_{j,-\ell}^>}^\ast$.
Let us be more explicit and evaluate
\begin{align}
g_{1,\ell}^> &= \bsplit{&\kla*{\frac{\pi\alpha}{\i\beta u}}^{N+2K} \csch_-(y_{-\ell})^\frac{N}{2} \csch_-(y_\ell)^\frac{N}{2} \csch_-(-y_{-\ell}')^\frac{K-1}{2} \csch_-(-y_\ell')^\frac{K+1}{2} \\
&\times \csch_-(y_{-\ell}-y_{-\ell}')^\frac{K+1}{2} \csch_-(y_\ell-y_\ell')^\frac{K-1}{2}},\\
g_{2,\ell}^> &= \bsplit{&\kla*{\frac{\pi\alpha}{\i\beta u}}^{N+2K} \csch_-(y_{-\ell})^\frac{N}{2} \csch_-(y_\ell)^\frac{N}{2} \csch_-(y_{-\ell}')^\frac{K-1}{2} \csch_-(y_\ell')^\frac{K+1}{2} \\
&\times \csch_-(y_{-\ell}-y_{-\ell}')^\frac{K+1}{2} \csch_-(y_\ell-y_\ell')^\frac{K-1}{2}},\\
g_{3,\ell}^> &= \bsplit{&\kla*{\frac{\pi\alpha}{\i\beta u}}^{N+2K} \csch_-(y_{-\ell})^\frac{N}{2} \csch_-(y_\ell)^\frac{N}{2} \csch_-(y_{-\ell}')^\frac{K-1}{2} \csch_-(y_\ell')^\frac{K+1}{2} \\
&\times \csch_-(y_{-\ell}'-y_{-\ell})^\frac{K+1}{2} \csch_-(y_\ell'-y_\ell)^\frac{K-1}{2}}.
\end{align}
Let us drop the regularization for now.
Then we find
\begin{align}
\sum_{j=1}^3 \intx{t_{j-1}^>}{t_j^>}{g_{j,\ell}^>}{t'} &= \intr{g_{1,\ell}^>}{t'} + \kla*{\powe{\i\pi K} - 1} \intx{0}{t}{g_{1,\ell}^>}{t'}.
\end{align}
This gives rise to the decomposition $G^> = G^{>,\Text{I}} + G^{>,\Text{II}}$ where
\begin{align}
G_{1,\ell\ell'}^{>,\Text{I}}(\xi) &= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2 2u} \intrnd{2}{g_{1,\ell}^>}{\xi'}\\
&= -\frac{\delta_{\ell,-\ell'}}{(2\pi\alpha)^2 2u} \powe{-NF_1(\xi)} \kla*{\frac{\pi\alpha}{\i\beta u}}^{2K} \frac{\beta^2 u^2}{\pi^2} I_1\kla*{y_\ell, K} I_1\kla*{y_{-\ell}, K}
\end{align}
and we defined the integral
\begin{align}
I_1(y,K) &= \intr{\csch_-(y+y')^\frac{K+1}{2} \csch_-(y')^\frac{K-1}{2}}{y'} \approx \kla*{1 + \powe{\i\pi K}} \frac{8^\frac{K-1}{2} B\kla*{\frac{K}{2}, 1-K} \sech(y)^\frac{K-1}{2}}{1 + \kla*{2^\frac{K-1}{2} - 1} \sech(y)^\frac{K+7}{10}}.
\end{align}
\QnA{Neglecting the regularization is not a good idea, because at the very least it has a dramatic effect on $I_1$.
Furthermore, we simply do not have an elegant way of treating the second contribution and it is also not small in any obvious way.
Integrating over a finite time-slice also means that the integral does not factorize into $\xi_\pm$ leaving us with a problem that appears to be just as hard as the Matsubara integral.
All the while we have neglected initial correlations, and while it is probably possible to reinsert them at first order by expanding $\powe{-\beta H_\text{int}}$, this seems to open yet another can of worms.}
