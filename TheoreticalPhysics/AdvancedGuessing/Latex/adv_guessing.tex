\chapter{Introduction}
\section{Definitions}
\definition{Asymptotic}{$f(x)$ is asymptotic to $g(x)$ as $x\rightarrow x_0$ iff $\lim_{x\rightarrow x_0} \frac{f(x)}{g(x)} = 1$; we write $f(x) \sim g(x)$ as $x\rightarrow x_0$.}
\definition{Big-$\mathcal{O}$}{We write $f(x) = \mathcal{O}\kla*{g(x)}$ as $x\rightarrow x_0$ iff $\frac{f(x)}{g(x)}$ is bounded.
\example{Example}{}{$\sin(x) = \mathcal{O}(x)$ and $5\sin(x) = \mathcal{O}(x)$.}
}
\definition{Little-$o$}{We write $f(x) = o\kla*{g(x)}$ as $x\rightarrow x_0$ if $\frac{f(x)}{g(x)} \rightarrow 0$, i.e. $f(x) \ll g(x)$.
}
\definition{Asymptotic Sequence}{$\klc*{\phi_n(x)}$ is an asymptotic sequence as $x\rightarrow x_0$ if $\phi_{n+1}(x) \ll \phi_n(x)$ for all $n$.
\example{Example}{}{$\klc*{\frac{1}{x^n}}_{n\in\mathbb{N}}$ is asymptotic as $x\rightarrow \infty$, because $\frac{1}{x} \gg \frac{1}{x^2} \gg \frac{1}{x^3} \gg \dots$ as $x\rightarrow \infty$}
}
\definition{Asymptotic Expansion (Poincare)}{$f(x)$ has the expansion $f(x) \sim \sum_{n\ge 0} a_n \phi_n(x)$ as $x\rightarrow x_0$ if 
\begin{enumerate}
\item $\klc*{\phi_n(x)}$ is asymptotic
\item $R_N(x) = f(x) - \sum_{n=0}^N a_n\phi_n(x) \ll \phi_N(x)$ for all $N$ (equivalently $R_N \sim a_{N+1} \phi_{N+1}$)
\end{enumerate}
\note{$a_{N+1} = \lim_{x\rightarrow x_0} \frac{R_N}{\phi_{N+1}}$ is unique for a given sequence}
\example{Laplace Transform}{$F(x) = \intx{0}{\infty}{\frac{1}{1+t} \powe{-xt}}{t}$ for $x>0$}{The geometric series yields $F(x) = \intx{0}{\infty}{\powe{-xt} \kla*{1 - t + t^2 + \dots}}{t} = \frac{1}{x} - \frac{1}{x^2} + \frac{2!}{x^3} + \dots$, but this is not rigorous because the integral goes beyond the \gls{roc} of $t<1$.
However, $F(x) \sim \sum_{n\ge 0} \frac{(-1)^n n!}{x^{n+1}}$ is an asymptotic expansion as $x\rightarrow \infty$; one can show that $|R_n(x)| < \frac{N!}{x^{N+1}} \ll \frac{1}{x^N}$.}
}
\section{Properties of Asymptotic Expansions}
\definition{Distinction between convergent and asymptotic series}{Let $S_n(x) = \sum_{j=1}^n a_j \phi_j(x)$ be an approximation to some function $f(x)$.
Then we call $S_n$
\begin{itemize}
\item convergent, iff $S_n(x) \rightarrow f(x)$ as $n\rightarrow \infty$ for each fixed $x$
\item asymptotic, iff $S_n(x) \sim f(x)$ as $x\rightarrow x_0$ for each fixed $n$
\end{itemize}}
\definition{Uniqueness of an asymptotic expansion}{If $f(x) \sim \sum_{n\ge 0} a_n \phi_n(x)$, those $a_n$ are uniquely determined by the recursive formula 
\begin{align}
a_n &= \lim_{x\rightarrow x_0} \frac{f(x) - \sum_{j=0}^{n-1} a_j\phi_j(x)}{\phi_n(x)}.
\end{align}
\example{Example}{$\sqrt{9 + \epsilon} = a_1\phi_1 + a_2 \phi_2 + a_3\phi_3$ with $\phi_1=\cos\epsilon$, $\phi_2 = \sin\epsilon$, $\phi_3=\sin^2\epsilon$}{\begin{align}
a_1 &= \lim_{\epsilon\rightarrow 0} \frac{\sqrt{9+\epsilon}}{\cos \epsilon} = 3,\\
a_2 &= \lim_{\epsilon\rightarrow 0} \frac{\sqrt{9+\epsilon} - 3\cos\epsilon}{\sin\epsilon} = \frac{1}{6},\\
a_3 &= \lim_{\epsilon\rightarrow 0} \frac{\sqrt{9+\epsilon - 3\cos\epsilon - \frac{1}{6} \sin\epsilon}}{\sin^2\epsilon} = \dots = \frac{323}{216}.
\end{align}
This is consistent with the known expansion $\sqrt{9+\epsilon} = 3 + \frac{1}{6} \epsilon - \frac{1}{216}\epsilon^2 + \mathcal{O}\kla*{\epsilon^3}$ in the sense that $3\cos\epsilon + \frac{1}{6}\sin\epsilon + \frac{323}{216}\sin^2\epsilon$ gives the same result after Taylor expansion.
}
\note{Every convergent Taylor series is asymptotic to $f(x)$ as well, since $1\gg \kla*{x-x_0} \gg \kla*{x-x_0}^2 \gg \dots$ as $x\rightarrow x_0$.}
\note{If we use a different asymptotic sequence, we get different coefficients}
}
\definition{\gls{tst}}{$f(x)$ is \gls{tst} if $f(x) \ll \frac{1}{x^n}$ as $x\rightarrow \infty$ for any $n\ge 0$.
\note{We also say that $f(x)$ is \emph{subdominant} to $\frac{1}{x^n}$}
\example{Example}{$\powe{-x}$ is \gls{tst} as $x\rightarrow \infty$}{We have $\powe{-x}\ll \frac{1}{x^n}$ due to $\frac{x^n}{\powe{x}} \rightarrow 0$, and thus $\powe{-x} \sim 0 + \frac{0}{x} + \frac{0}{x^2} + \dots$.
This is a correct expansion based on the definition, but this raises series uniqueness concerns: $g(x) = f(x) + \powe{-x}$ have the same expansion.
The problem can be traced back to $x=\infty$ being an essential singularity for $\powe{-x}$}
}
\definition{Working with Asymptotic Series}{Addition, Subtraction, Multiplication, Division and even Integration can be performed term by term; but Substitution and Differentiation are more subtle.
\example{Substitution}{$f(x) = \powe{x^2}$ and $x(\epsilon) = \epsilon + \frac{1}{\epsilon}$ as $\epsilon\rightarrow 0$}{The exact result is $f\kla*{x(\epsilon)} = \powe{2 + \epsilon^2 + \frac{1}{\epsilon^2}} \sim \powe{2} \powe{\frac{1}{\epsilon^2}}$ with an essential singularity at $\epsilon=0$.

Wrong substitution of $x\sim \frac{1}{\epsilon}$  as $\epsilon\rightarrow 0$ instead only gives $f(x) \sim \powe{\frac{1}{\epsilon^2}}$, so the prefactor is incorrect.
}
\example{Differentiation}{$f(x)=x+\sin(x)$ as $x\rightarrow\infty$}{We have $f\sim x$, but $f'=1+\cos(x) \slashed{\sim} 1$.}
These troubles emerge from asymptotic relations in the complex plane: $f(z) \sim g(z)$ is defined by $\frac{f(z)}{g(z)} \rightarrow 1$ independent of the path taken from $z\rightarrow z_0$.
\example{Example}{$2\sinh(1/z)$ as $z\rightarrow 0$}{\begin{enumerate}
\item $2\sinh(1/z) = \sim \powe{1/z}$ for $|\arg z| < \frac{\pi}{2}$
\item $2\sinh(1/z) = \sim \powe{-1/z}$ for $\frac{\pi}{2} < \arg z < \frac{3\pi}{2}$
\end{enumerate}
The result depends on the direction, i.e. the choice of path.
}
}

\chapter{Local and Global Analysis}
\section{Methods of Perturbations and Dominant Balance}
\definition{Local Analysis}{Powerful, but cannot provide global information on the behavior of solutions at two distantly separated points.
\begin{itemize}
\item $\Rightarrow$ cannot predict how a change in \gls{ic} at $x=0$ will affect the asymptotic behavior as $x\rightarrow\infty$
\item $\Rightarrow$ we require a global analysis
\end{itemize}}
\definition{\gls{pt}}{\begin{enumerate}
\item identify a small parameter $\epsilon$ such that the \gls{hp} becomes soluble at $\epsilon=0$
\item perturbative solution is constructed by local analysis about $\epsilon=0$ as $y(x) = \sum_{n\ge 0} y_n(x) \epsilon^n$
\begin{itemize}
\item $\Rightarrow$ series is \emph{local in $\epsilon$}, but \emph{global in $x$}!
\item $\Rightarrow$ we may approximate the solution by a finite number of terms ($n\le N$)
\end{itemize}
\item restore the \gls{hp} by setting $\epsilon=1$; assuming this is within the \gls{roc}
\end{enumerate}}
\note{\gls{pt} is a large collection of iteration methods.}
\example{Cubic Equation}{$x^3 - 4.001x + 0.002 = 0$}{\begin{enumerate}
\item write $x^3 - \kla*{4+\epsilon}x + 2\epsilon=0$ and note that $\epsilon=0$ gives $x^3=4x$ with $x(\epsilon=0) = a_0 \in \klc*{0,\pm 2}$ (zeroth order)
\item $x(\epsilon) = \sum_{n\ge 0} a_n \epsilon^n$, let us focus on $x_1 = -2 + \sum_{n\ge 1} a_n \epsilon^n$
\begin{itemize}
\item $\epsilon^1$: $0 = 4 + 8a_1$ gives $a_1 = -\frac{1}{2}$
\item $\epsilon^2$: $0 = 8a_2 - a_1 - 6a_1^2 = 8a_2 + \frac{1}{2} - \frac{3}{2}$ gives $a_2 = \frac{1}{8}$
\item $\Rightarrow$ $x_1 = -2 - \frac{1}{2}\epsilon + \frac{1}{8} \epsilon^2 + \dots$
\end{itemize}
\item set $\epsilon=0.001$; if the series converges, the sum is the answer. Otherwise, many divergent series are asymptotic (see later)
\end{enumerate}
}
\example{ODE}{$y'' = f(x)y$ with $y(0)=1$ and $y'(0)=1$}{\begin{enumerate}
\item $y''=\epsilon f(x)y$
\item let $y(x) = \sum_{n\ge 0} \epsilon^n y_n(x)$ with $y_n(0)=0=y_n'(0)$ for $n\ge 1$
\begin{itemize}
\item $\epsilon^0$: $y_0''=0$ gives $y_0(x)=1+x$
\item $\epsilon^n$: $y_n''=y_{n-1} f$ gives $y_n(x) = \intx{0}{x}{\intx{0}{t}{f(s)y_{n-1}(s)}{s}}{t}$
\item $y(x) = 1 + x + \epsilon\intx{0}{x}{\intx{0}{t}{(1+s)f(s)}{s}}{t} + \epsilon^2 (\dots)$
\item note that the $n$-th term is bounded by $\epsilon^n K(x)^n \frac{1+|x|}{(2n)!} x^{2n}$
\end{itemize}
\end{enumerate}}

Consider the algebraic equation $\epsilon^2 x^6 - \epsilon x^4 - x^3 + 8 = 0$ with a small parameter $\epsilon$.

For $\epsilon=0$, the equation simplifies to $x^3=8$ and has the three (complex) solutions $x_n = 2 \powe{\i \frac{2\pi}{3} n}$ for $n=0,1,2$.
The original equation is of sixth order and must therefore have six solutions.
Presumably, these roots continuously merge into the three simple $x_n$ as $\epsilon\rightarrow 0$.
\definition{Dominant Balance}{The Method of Dominant Balance in the context of equations such as the one above generally concerns the problem
\begin{align}
\sum_{q=1}^Q T_q\kla*{x} &= 0,
\end{align}
where $Q$ is the number of terms and $x\kla*{\epsilon}$ is a solution for $\epsilon\rightarrow 0$.
Asymptotically solving the reduced equation for only two of these terms, e.g. $T_q(x) + T_p(x) = 0$, yields some solution $x_{q,p}\kla*{\epsilon}$.
This solution is \emph{consistent} if it is \emph{dominant}, meaning that the sum over the remaining terms, $R_{q,p}(x)$, is ``much smaller'' than $T_q(x)$ and $T_p(x)$.}
In our case we have the four terms $T_1(x) = \epsilon^2 x^6$, $T_2(x) = -\epsilon x^4$, $T_3(x) = -x^3$ and $T_4(x)=8$.
\begin{itemize}
\item $q=1,2$: $\epsilon^2 x^6 = \epsilon x^4$ gives $\epsilon x^2 = 1$ and $\tilde{x}^2 = 1$ after the scale transformation $\tilde{x} = \sqrt{\epsilon} x$. The solution is $\tilde{x} = \pm 1$. We have $\epsilon^{-1} \tilde{x}^6 \ll 8-\epsilon^{-\frac{3}{2}} \tilde{x}^3$ as $\epsilon\rightarrow 0$, so this solution is not consistent.
\item $q=1,3$: $\epsilon^2 x^6 = x^3$ gives $\epsilon^2 x^3 = 1$ and $\tilde{x}^3=1$ after the scale transformation $\tilde{x} = \epsilon^\frac{2}{3} x$. The solution is $\tilde{x} = \powe{\i \frac{2\pi}{3} n}$ for $n=0,1,2$. We have $\epsilon^{-2} \tilde{x}^6 \gg 8 - \epsilon^{-\frac{5}{3}} \tilde{x}^4$, so this solution is consistent (checking the other term is not required, they must scale identically by construction).
\item $q=1,4$: $\epsilon^2 x^6 = -8$ gives $\tilde{x}^6 = -8$ after the scale transformation $\tilde{x} = \epsilon^\frac{1}{3} x$. The solution is $\tilde{x} = 8^\frac{1}{6} \powe{\i \pi \frac{2n+1}{6}}$ for $n=0,\dots,5$. We have $8 \ll \epsilon^{-1} \tilde{x}^3$, so this is not consistent.
\item $q=2,3$: $\epsilon x^4 = -x^3$ gives $\epsilon x = -1$ and $\tilde{x} = -1$ after the scale transformation $\tilde{x} = \epsilon x$. We have $\epsilon^{-3} \tilde{x}^3 \ll \epsilon^{-4} \tilde{x}^6$, so this is not consistent.
\item $q=2,4$: $\epsilon x^4 = 8$ gives $\tilde{x}^4 = 8$ after the scale transformation $\tilde{x} = \epsilon^\frac{1}{4} x$. The solution is $\tilde{x} = 8^\frac{1}{4} \powe{\i \frac{2\pi}{4} n}$ for $n=0,\dots,3$. We have $8 \ll \epsilon^{-\frac{3}{4}} \tilde{x}^3$, so this is not consistent.
\item $q=3,4$: $x^3=8$ has been solved already; $8 \gg \epsilon^2 x^6 - \epsilon x^4$ as $\epsilon\rightarrow 0$, so this is consistent.
\end{itemize}
To summarize, aside from the solutions $x_n = 2\powe{\i \frac{2\pi}{3} n}$ for $n=0,1,2$ we have only found $x_n\kla*{\epsilon} = \epsilon^{-\frac{2}{3}} \powe{\i \frac{2\pi}{3}n}$ as consistent solutions.
This sheds some light on the disappearance of the zeros: apparently each root splits into exactly two, since the asymptotic roots satisfy the same three-fold symmetry.

Introducing $x = \epsilon^{-\frac{2}{3}} y$ gives $\epsilon^{-2} y^6 - \epsilon^{-\frac{5}{3}} y^4 - \epsilon^{-2} y^3 + 8 = 0$.
The new equation is equivalent to $y^6-y^3+8\epsilon^2 - \epsilon^{\frac{1}{3}} y^4 = 0$, which can also be interpreted as a regular perturbation problem with $\epsilon' = \epsilon^\frac{1}{3}$ (in this notation, the small parameter only appears in integer powers).
Equivalently, we may also write $y = \sum_{n\ge 0} y_n \kla[\big]{\epsilon^\frac{1}{3}}^n$ for all of the six unperturbed solutions $0$ (3 times) and $\powe{\i\frac{2\pi}{3}n}$ for $n=0,1,2$.

\example{Singular Perturbation}{$\epsilon y'' = y'$ with $y(0)=0$ and $y(1)=1$}{\note{$\epsilon$ is put with the highest order, so $\epsilon=0$ is likely a singular point}
The unperturbed problem is $y'=0$ and has no solution consistent with the \gls{ic}, so this is a singular perturbation.
\begin{itemize}
\item $\Rightarrow$ the solution cannot be $y(x) = \sum_{n\ge 0} y_n(x)\epsilon^n$, because $y_0(x)$ does not exist
\end{itemize}
\definition{Boundary Layer Theory}{The exact solution is $y(x) = \frac{\powe{x/\epsilon}-1}{\powe{1/\epsilon} - 1}$.
For small $\epsilon>0$, there exists a boundary layer (later!); the solution varies rapidly near $x=1$ and develops a discontinuity for $\epsilon\rightarrow 0^+$.}
}
\example{\gls{ivp}}{$y''+(1-\epsilon x)y=0$ with $y(0)=1$ and $y'(0)=0$}{This is a regular perturbation in $\epsilon$ over a finite interval $x\in\klb*{0,L}$.
\begin{itemize}
\item $\Rightarrow$ $y(x) = \cos(x) + \epsilon \kla*{\frac{1}{4}x^2 \sin(x) + \frac{1}{4}x\cos (x) - \frac{1}{4}\sin(x)} + \dots$ converges for all $x$ and $\epsilon$
\item however, for $L\rightarrow \infty$ the problem is singular: the exact solution approaches $y_0$ for fixed $x$, but not uniformly for all $x$ (multiple timescale method)
\end{itemize}
}
\section{Perturbation Method vs. Iteration Method}
\example{A regular perturbation}{$x^2+\epsilon x-1=0$}{Exact solution is $x=-\frac{1}{2}\epsilon \pm \sqrt{1 + \frac{1}{4}\epsilon^2} = \pm 1 - \frac{1}{2} \epsilon \pm \frac{1}{8} \epsilon^2$ (converges for $|\epsilon| < 2$).
\note{Binomial series expansion $\kla*{1+x}^\alpha = 1 + \alpha x + \frac{\alpha (\alpha - 1)}{2!} x^2 + \dots$ for $|x|<1$}

First Method: Perturbation Series.
\begin{itemize}
\item $\epsilon=0$: $x^2=1$ gives $x=\pm 1$
\item let $x(\epsilon) = \sum_{n\ge 0} a_n \epsilon^n$ and expand $1 = \kla*{1+a_1\epsilon + \dots}^2 + \epsilon\kla*{1 + a_1\epsilon + \dots}$ (analogous for $a_0=-1$)
\item $\epsilon^1$: $0=1+2a_1$ gives $a_1=-\frac{1}{2}$
\item $\epsilon^2$: $a_1^2 + 2a_2 + a_1=0$ gives $a_2=\frac{1}{8}$ (we recover the binomial expansion)
\end{itemize}
Second Method: Iteration.
\begin{itemize}
\item $x^2=1-\epsilon x$, so try $x_{n+1} = \pm\sqrt{1 - \epsilon x_n}$ with $x_0=\pm 1$ as \gls{ic}
\item $x_1 = \sqrt{1 - \epsilon} \sim 1 - \frac{1}{2} \epsilon - \frac{1}{8}\epsilon^2$ (note the different sign of the $\epsilon^2$-term!), truncate at linear order
\item $x_2=\sqrt{1 - \epsilon x_1} \sim \sqrt{1 - \epsilon + \frac{1}{2}\epsilon^2} \sim 1 - \frac{\epsilon}{2} + \frac{1}{8} \epsilon^2$ (now it is correct)
\end{itemize}
}
\example{A singular perturbation}{$\epsilon x^2 + x - 1 = 0$}{The unperturbed problem only has a single root, the other disappears to infinity as $\epsilon\rightarrow 0$ (as usual).
Dominant Balance:
\begin{itemize}
\item $\epsilon x^2 \sim 1$ gives $x\sim \epsilon^{-\frac{1}{2}}$, but $1 \ll x$ is inconsistent
\item $\epsilon x^2 \sim -x$ gives $x\sim -\epsilon^{-1}$ and $x \gg 1$ is consistent
\end{itemize}
The scale transformation $y = \epsilon x$ gives $y^2 + y = \epsilon$ (once again a regular perturbation), and we have $y(\epsilon) = \sum_{n\ge 0} a_n \epsilon^n$ with $a_0=-1$.
\begin{align*}
\Rightarrow x&\sim \frac{1}{\epsilon} \kla*{-1 - \epsilon + \epsilon^2 - 2\epsilon^3 + \dots} = -\frac{1}{\epsilon} - 1 + \epsilon - 2\epsilon^2 + \dots
\end{align*}
}
\example{Another quadratic}{$(1-\epsilon)x^2 - 2x + 1 = 0$}{Exact solution is $x=\frac{1 \pm \sqrt{\epsilon}}{1 - \epsilon}$ and at $\epsilon=0$, $x=1$ is a double root.
The naive expansion $x=1+\epsilon a_1+\dots$ fails at first order; instead, $x\sim 1 + \sqrt{\epsilon} + \epsilon + \epsilon^{3/2} + \dots$ works (using Dominant Balance; this recovers the geometric series).
}
\example{Non-linear equation}{$x\powe{-x} = \epsilon$}{As $\epsilon\rightarrow 0$ we must have that one solution satisfies $x(\epsilon) \rightarrow \infty$ while the other satisfies $x(\epsilon) \rightarrow 0$.
Also, as $x\powe{-x}$ is bounded on the relevant interval $x\in\kla*{0,\infty}$, a solution can only exist for some finite range of $\epsilon$.

Let us try to first find the one close to the origin.
The iteration method $x_{n+1} = \epsilon \powe{x_n}$ with $x_0=0$ gives
\begin{itemize}
\item $x_1=\epsilon$
\item $x_2 = \epsilon\powe{\epsilon} \sim \epsilon + \epsilon^2$
\item $x_3 = \epsilon\powe{\epsilon + \epsilon^2} \sim \epsilon + \epsilon^2 + \frac{3}{2}\epsilon^3$
\item $x_4 = \epsilon\powe{x_3} \sim \epsilon + \epsilon^2 + \frac{3}{2}\epsilon^3 + \frac{8}{3}\epsilon^4$
\end{itemize}
The other solution is far away, so let us start by rewriting the equation as $x - \log x = -\log \epsilon$.
Since $x\gg \log x$, we have $x_0 = -\log \epsilon$ as a reasonable first guess.
Rewriting as a fixpoint problem, we have $x = \log \frac{x}{\epsilon}$.
Using $x_0$ as a starting point, we obtain the iteration
\begin{align}
x_0 &= \log\frac{1}{\epsilon}, \quad x_1 = \log \frac{-\log \epsilon}{\epsilon} = \log\log \frac{1}{\epsilon} + \log \frac{1}{\epsilon}, \quad x_2 = \log \frac{\log\log \frac{1}{\epsilon} + \log \frac{1}{\epsilon}}{\epsilon}.
\end{align}
}
\example{A projectile motion}{$\ddot{x} = -\frac{GM}{\kla*{x+R}^2}$ with $x(0)=0$, $\dot{x}(0)=v_0$ and $x(t)\ll R$}{Introducing $g = \frac{GM}{R^2}$ yields
\begin{align}
\ddot{x} &= -\frac{g}{\kla*{1+x/R}^2}.
\end{align}
The naive solution is $\ddot{x} \sim -g$, giving $x=-\frac{1}{2}gt^2 + v_0t$ and the recurrence time $T=\frac{2v_0}{g}$.

The peak of the trajectory may be used as a timescale for the expansion, so introduce $t_0=\frac{v_0}{g}$ and $x_0=x(t_0) = \frac{v_0^2}{g}$.
Then, writing $\tau=\frac{t}{t_0}$ and $y = \frac{x}{x_0}$, we obtain
\begin{align}
\ddot{x} &= \frac{x_0}{t_0^2} \frac{\d^2 y}{\d \tau^2},\\
y'' &= -g \frac{t_0^2}{x_0} \frac{1}{\kla*{1 + \epsilon y}^2}
\end{align}
with the small parameter $\epsilon = \frac{x_0}{R} = \frac{v_0^2}{gR}$, and $y(0)=0$ as well as $y'(0) = v_0 \frac{t_0}{x_0} = 1$.
Expanding $y=y_0+\epsilon y_1$ (we only want the leading order correction here) gives
\begin{align}
y_0'' + \epsilon y_1'' &= -\frac{1}{\kla*{1 + \epsilon y_0}^2} = -\frac{1}{1 + 2\epsilon y_0} = -1 + 2\epsilon y_0.
\end{align}
The zeroth order is $y_0=-\frac{1}{2}\tau^2 + \tau$ and $\tau=2$ gives $T_0= 2t_0$.
The first order gives $y_1'' = 2y_0 = 2\tau - \tau^2$, and integrating yields $y_1 = \frac{1}{3}\tau^3 - \frac{1}{12} \tau^4 + c_0 + c_1\tau$ with $c_0=0$ and $c_1=0$ form the \gls{ic}.
Thus, $y = \tau - \frac{1}{2}\tau^2 + \frac{\epsilon}{12} \tau^3 \kla*{4 - \tau} + \mathcal{O}\kla*{\epsilon^2}$.
Writing $\tau = 2 + \epsilon\tau_1$, we find $\tau_1 = \frac{4}{3}$.
Therefore, $\tau = 2 + \frac{4}{3}\epsilon$ and $T = \frac{2v_0}{g} + \frac{4}{3} \frac{v_0^2}{g^2R}$.

\section{Logarithmic Decaying}
Consider the ODE $y''y = -1$ with $y(0)=y(1)=0$; what is local behaviour of $y(x)$ as $x\rightarrow 0^+$?

Let us start with the assumption of an algebraic decay, $y\sim Ax^b$.
Inserting gives
\begin{align}
Ab(b-1)x^{b-2} Ax^b &= Ab(b-1) x^{2b-2} \sim -1,
\end{align}
which leads to $b=1$ -- unfortunately, this trivializes the LHS and is thus not allowed ($y=x$ is not a good approximation as $y''=0$).

Next, let us try a logarithmic decay $y\sim Ax \kla*{-\log x}^b$.
This gives
\begin{align}
y' &\sim \frac{y}{x} + Ax b \kla*{-\log x}^{b-1} \frac{-1}{x} = \frac{y}{x} + b\frac{y}{x\log x} = \frac{y}{x} \kla*{1 + \frac{b}{\log x}},\\
y'' &\sim -\frac{y}{x^2} \kla*{1 + \frac{b}{\log x}} + \frac{y}{x^2} \kla*{1 + \frac{b}{\log x}}^2 - \frac{y}{x^2} \frac{b}{\kla*{\log x}^2}.
\end{align}
Plugging this into the equation, we find
\begin{align}
-1 &\sim y'' y \sim \frac{y^2}{x^2} \klc*{1 + \frac{2b}{\log x} + \frac{b^2}{\kla*{\log x}^2} - 1 - \frac{b}{\log x} - \frac{b}{\kla*{\log x}^2}}\\
&\sim A^2 \klc*{b(b-1) \kla*{-\log x}^{b-2} - b\kla*{-\log x}^{b-1}}.
\end{align}
Setting $b=1$ and $A=1$ satisfies this, so the asymptotic behaviour is $y(x) \sim -x\log x$.
\section{Perturbation Methods for Linear Eigenvalue Problems}
\definition{Schrödinger Equation}{A Schrödinger problem is the ODE
\begin{align}
\kla*{V(x) + W(x) - E - \partial_x^2} y(x) &= 0
\end{align}
with \gls{bc} $y(x) \rightarrow 0$ for $|x|\rightarrow \infty$.
Further, we assume that $V$ and $W$ are continuous and unbounded for large $x$, and that the problem is soluble for $W=0$.}
\begin{enumerate}
\item insert $\epsilon$, i.e. $W\rightarrow \epsilon W$; e.g. \gls{ho} $V(x)=\frac{1}{4}x^2$ with $E_n= n+\frac{1}{2}$ and $y_n(x) = \powe{-x^2/4} H_n(x)$ (Hermite polynomials)
\item find unperturbed solutions $y_0(x)$ and $E_0$, then seek solution in the form $E=\sum_{n\ge 0} E_n\epsilon^n$ and $y(x) = \sum_{n\ge 0} y_n(x)\epsilon^n$.
\begin{itemize}
\item $\Rightarrow$ $\epsilon^n$: $\kla*{V(x) - E_0 - \partial_x^2}y_n(x) = -W(x)y_{n-1}(x) + \sum_{j=1}^n E_j y_{n-j}(x)$ with $y_n(x) \rightarrow 0$ as $|x|\rightarrow \infty$
\end{itemize}
\end{enumerate}
\example{Method of Reduction}{$y_n(x) = y_0(x)F_n(x)$ with $F_0(x)=1$}{Inserting this into the $n$-th order equation gives
\begin{align}
\partial_x \kla*{y_0F_n} &= y_0'F_n + y_0F_n',
\partial_x^2 \kla*{y_0F_n} &= y_0'' F_n + 2y_0'F_n' + y_0F_n'',\\
\partial_x \kla*{y_0^2 F_n'} &= 2y_0y_0'F_n' + y_0^2 F_n'',\\
\Rightarrow y_0 \partial_x^2 \kla*{y_0 F_n} &= y_0y_0'' F_n + 2y_0y_0'F_n' + y_0^2F_n'' = y_0y_0''F_n + \partial_x\kla*{y_0^2 F_n'},\\
\Rightarrow \partial_x \kla*{y_0^2F_n'} &= y_0\partial_x^2 \kla*{y_0F_n} - y_0y_0''F_n = y_0\kla*{\partial_x^2 - V+E_0} y_0F_n\\
&= y_0^2 \klb[\bigg]{WF_{n-1} - \sum_{j=1}^n E_jF_{n-j}}.
\end{align}
Due to the \gls{bc}, the integral over the LHS vanishes, which leaves us with
\begin{align}
\intr{E_n y_0^2}{x} &= \intr{y_0^2 \klb[\bigg]{WF_{n-1} - \sum_{j=1}^{n-1} E_j F_{n-j}}}{x},\\
E_n &= \frac{1}{\intr{y_0^2}{x}} \intr{y_0 \klb[\bigg]{Wy_{n-1} - \sum_{j=1}^{n-1} E_j y_{n-j}}}{x}.
\end{align}
Knowing the energy, we can obtain the function $F_n$ by integrating the LHS,
\begin{align}
y_0^2 F_n'(x) &= \intx{-\infty}{x}{y_0(s) \klb[\bigg]{W(s)y_{n-1}(s) - \sum_{j=1}^{n} E_j y_{n-j}(s)}}{s},\\

\end{align}
Here, $a$ is chosen such that $y_n(a)=0$ (but $y_0(a)\neq 0$).

For $W(x)=\frac{1}{4}x^4$, the ground state has $E(\epsilon) \sim \frac{1}{2} + \frac{3}{4}\epsilon - \frac{21}{8} \epsilon^2 + \frac{333}{16}\epsilon^3 + \dots$, which is divergent for all $\epsilon> 0$, so this is a singular perturbation problem.
\note{Reason: $y_0 \sim \powe{-x^2/4}$, but $y(\epsilon > 0) \sim \powe{-x^3 \sqrt{\epsilon} / 6}$, which constitutes an abrupt qualitative change; $x^4\epsilon$ is not small compared to $V(x)$ for large $x$.}

\subsection{Time-independent Perturbation Theory in QM}
Assume that our Hamiltonian can be written as $H=H_0+\epsilon V$, where $\epsilon\ll 1$ and the unperturbed problem can be solved exactly, i.e. $H_0\Ket{n^{(0)}} = E_n^{(0)} \Ket{n^{(0)}}$ for $n\ge 0$.
Further, assume that the eigenvalues are non-degenerate (the degenerate version of this PT requires substantially more effort), and that the eigenstates form an ONB.
Using PT to second order, solve $H\Ket{n} = E_n\Ket{n}$.

Let $\Ket{n} = \sum_{l\ge 0} \epsilon^l \Ket{n^{(l)}}$ and $E_n = \sum_{l\ge 0} \epsilon^l E_n^{(l)}$.
Inserting this into the Schrödinger equation gives
\begin{align}
H_0\sum_{l\ge 0} \epsilon^l \Ket{n^{(l)}} + V \sum_{l\ge 1} \epsilon^{l} \Ket{n^{(l-1)}} &=  \sum_{l\ge 0} \epsilon^l E_n^{(l)} \sum_{m\ge 0} \epsilon^m \Ket{n^{(m)}} = \sum_{l\ge 0} \epsilon^l \sum_{m=0}^l E_n^{(l-m)} \Ket{n^{(m)}}.
\end{align}
The PT equation at order $l$ then read
\begin{align}
H_0\Ket{n^{(l)}} + V\Ket{n^{(l-1)}} &= \sum_{m=0}^l E_n^{(l-m)} \Ket{n^{(m)}}.
\end{align}
For $l=1$, we obtain $H_0 \Ket{n^{(1)}} + V\Ket{n^{(0)}} = E_n^{(1)} \Ket{n^{(0)}} + E_n^{(0)} \Ket{n^{(1)}}$.
The energy correction can be obtained by projecting onto $\Ket{n^{(0)}}$, giving
\begin{align}
E_n^{(1)} &= \brc{n^{(0)}}{H_0}{n^{(1)}} + \brc{n^{(0)}}{V}{n^{(0)}} - E_n^{(0)} \brb{n^{(0)}}{n^{(1)}} = \brc{n^{(0)}}{V}{n^{(0)}}.
\end{align}
The states can be found similarly by projecting onto $\Ket{m^{(0)}}$ ($m\neq n$), giving
\begin{align}
E_m^{(0)} \brb{m^{(0)}}{n^{(1)}} + \brc{m^{(0)}}{V}{n^{(0)}} &= E_n^{(1)} \brb{m^{(0)}}{n^{(0)}} + E_n^{(0)} \brb{m^{(0)}}{n^{(1)}}.
\end{align}
Note that we may expand the correction states as\footnote{We can exclude $m=n$, because $\brb{n^{(0)}}{n} = \brb{n^{(0)}}{n^{(0)}} + \epsilon \brb{n^{(0)}}{n^{(1)}} + \dots = 1 + \epsilon \brb{n^{(0)}}{n^{(1)}} + \dots \overset{!}{\le 1}$ requires $\brb{n^{(0)}}{n^{(1)}} = 0$ (there may only be some phase, see notes).} $\Ket{n^{(1)}} = \sum_{m\neq n} \Ket{m^{(0)}} \brb{m^{(0)}}{n^{(1)}}$, and we already found the coefficients to be
\begin{align}
\brb{m^{(0)}}{n^{(1)}} &= \frac{\brc{m^{(0)}}{V}{n^{(0)}}}{E_n^{(0)} - E_m^{(0)}}.
\end{align}
To second order, we find
\begin{align}
H_0\Ket{n^{(2)}} + V\Ket{n^{(1)}} &= E_n^{(2)} \Ket{n^{(0)}} + E_n^{(1)} \Ket{n^{(1)}} + E_n^{(0)} \Ket{n^{(2)}},
\end{align}
and once again projecting gives
\begin{align}
E_n^{(0)} \brb{n^{(0)}}{n^{(2)}} + \brc{n^{(0)}}{V}{n^{(1)}} &= E_n^{(2)} + E_n^{(1)} \brb{n^{(0)}}{n^{(1)}} + E_n^{(0)} \brb{n^{(0)}}{n^{(2)}}.
\end{align}
Thus $E_n^{(2)} = \brc{n^{(0)}}{V}{n^{(1)}}$, which expands to
\begin{align}
E_n^{(2)} &= \sum_{m\neq n} \frac{\abs*{\brc{n^{(0)}}{V}{m^{(0)}}}^2}{E_n^{(0)} - E_m^{(0)}}.
\end{align}
For the anharmonic oscillator, we have $H_0 = \frac{1}{2m}p^2 + \frac{1}{2} m\omega^2x^2$ and $V=x^4$.
Using $x = \sqrt{\frac{\hbar}{2m\omega}} \kla*{a + a^\dagger}$, we find ($\Ket{n} \equiv \Ket{n^{(0)}}$)
\begin{align}
E_n^{(1)} &= \brc{n}{V}{n} = \frac{\hbar^2}{4m^2\omega^2} \brc{n}{\kla{a+a^\dagger}^4}{n} \\
&= \frac{\hbar^2}{4m^2\omega^2} \brc{n}{\klc{a^2 \kla{a^\dagger}^2 + a \kla{a^\dagger}^2 a + \kla{a^\dagger}^2 a^2 + aa^\dagger aa^\dagger + a^\dagger a^2 a^\dagger + a^\dagger aa^\dagger a}}{n}\\
&= \frac{\hbar^2}{4m^2\omega^2} \brc{n}{\klc{\kla*{n+1}\kla*{n+2} + (n+1)n + n(n-1) + \kla*{n+1}^2 + n(n+1) + n^2}}{n}\\
&= \frac{\hbar^2}{4m^2\omega^2} \kla*{3 + 6n + 6n^2}.
\end{align}
\definition{General Method}{Consider $\kla*{H-E_n^{(0)}} \Ket{n} = \kla*{E_n - E_n^{(0)}}\Ket{n}$ and define $\Delta E_n = E_n - E_n^{(0)}$.
Then the homogeneous solution of is $\Ket{n^{(0)}}$, since $\kla*{E_n^{(0)} - H_0}\Ket{n^{(0)}} = 0$.

For the particular solution, note that projecting onto the unperturbed states yields
\begin{align}
\brc{n^{(0)}}{\kla*{E_n^{(0)} - H_0}}{n} &= 0 = \brc{n^{(0)}}{\kla*{\epsilon V - \Delta E_n}}{n}.
\end{align}
Thus, $\Ket{n^{(0)}}$ are orthogonal to $\kla*{\epsilon V - \Delta E_n}\Ket{n}$ and
\begin{align}
\kla*{\epsilon V - \Delta E_n}\Ket{n} &= \sum_{m\neq n} \oper{m^{(0)}}{m^{(0)}} \kla*{\epsilon V - \Delta E_n} \Ket{n} = Q_n \kla*{\epsilon V - \Delta E_n} \Ket{n}
\end{align}
with the projector $Q_n = I - \oper{n^{(0)}}{n^{(0)}}$.
\begin{itemize}
\item the particular solution is $\Ket{n} = \frac{1}{E_n^{(0)} - H_0} Q_n \kla*{\epsilon V - \Delta E_n} \Ket{n}$
\item the general solution is $\Ket{n} = c_0 \Ket{n^{(0)}} + \frac{1}{E_n^{(0)} - H_0} Q_n \kla*{\epsilon V - \Delta E_n} \Ket{n}$

\end{align} with $\brb{n^{(0)}}{n} = 1$ and $\brb*{n}{n}=1$
\end{itemize}
\note{$\Ket{n} \rightarrow \Ket{n^{(0)}}$ as $\epsilon\rightarrow 0$, so this is a regular perturbation.
Also, $c_0\rightarrow 1$, so we \emph{set} $c_0=1$ (renormalization problem)}
Computing the energies can now be done by unraveling $$\Delta E_n = \epsilon \brc{n^{(0)}}{V}{n} = \epsilon \brc{n^{(0)}}{\kla*{V + V \epsilon \frac{Q_n}{E_n^{(0)} - H_0} V + \dots}}{n^{(0)}},$$ which yields
\begin{itemize}
\item $\epsilon^1$: $\Delta E_n = \epsilon \brc{n^{(0)}}{V}{n^{(0)}}$
\item $\epsilon^2$: $\Delta E_n = \epsilon^2 \brc{n^{(0)}}{V \frac{Q_n}{E_n^{(0)} - H_0} V}{n^{(0)}}$
\end{itemize}

\section{An Elementary Approach}
Consider the integral $I(x) = \intx{a}{b}{f(x,t)}{t}$ as $x\rightarrow x_0$.
\begin{enumerate}
\item if $f(t,x) \sim f_0(t)$ uniformly on $t\in\klb*{a,b}$, then $I(x) \sim \intx{a}{b}{f_0(t)}{t}$ (assuming finite and non-zero)
\item if $f(x,t) \sim \sum_{n\ge 0} f_n(t) \kla*{x-x_0}^{\alpha n}$ for some $\alpha > 0$ uniformly, then $I(x) \sim \sum_{n\ge 0} \kla*{x-x_0}^{\alpha n} \intx{a}{b}{f_n(t)}{t}$ (generalization of the first case)
\end{enumerate}
\example{Three examples}{}{\begin{enumerate}
\item $I(x) = \intx{0}{2}{\cos\klb*{\kla*{xt + x^2t}^{1/3}}}{t} \sim 2$ as $x\rightarrow 0$ (uniformly converges to the constant function $f_0(t)=2$)
\item $I(x) = \intx{0}{1}{\frac{\sin(tx)}{t}}{t} \sim \sum_{n\ge 0} \frac{(-1)^n}{(2n+1)!} x^{2n+1} \intx{0}{1}{t^{2n}}{t} \sim x - \frac{1}{18} x^3 + \frac{1}{600} x^5 + \dots$ as $x\rightarrow 0$ (uniformly converging series expansion of $f$)
\item $I(x) = \intx{0}{x}{t^{-1/2} \powe{-t}}{t}$ as $x\rightarrow 0^+$: $f(t) \sim t^{-1/2} - t^{1/2} + \frac{1}{2!} t^{3/2} - \dots$ is asymptotic uniformly on $\klb*{0,x}$, so $I(x) \sim 2x^{1/2} - \frac{2}{3} x^{3/2} + \frac{1}{5} x^{5/2} + \dots$
\end{enumerate}
}

\end{itemize}

\subsection{Laplace Integral}
\definition{Laplace Integral}{$I(x) = \intx{a}{b}{f(t) \powe{x\phi(t)}}{t}$, i.e. $x$ only enters via an exponential function in the integrand.
Note that
\begin{align}
I(x) &= \frac{1}{x} \intx{a}{b}{\frac{f(t)}{\phi'(t)} }{\powe{x\phi(t)}} = \frac{1}{x} \frac{f(t)}{\phi'(t)} \powe{x\phi(t)} \bigg\vert_a^b - \frac{1}{x} \intx{a}{b}{\powe{x\phi(t)} \frac{\d}{\d t} \frac{f(t)}{\phi'(t)}}{t}.
\end{align}
If the second term is dominated by the first, then
\begin{align}
I(x) &\koml{x\rightarrow\infty}{\sim} \frac{1}{x} \frac{f(b)}{\phi'(b)} \powe{x\phi(b)} - \frac{1}{x} \frac{f(a)}{\phi'(a)} \powe{x\phi(a)}.
\end{align}
\begin{enumerate}
\item $\phi'(t) \neq 0$ on $\klb*{a,b}$ and at least one of $f(a),f(b)$ is non-zero: $\frac{1}{x} \intx{a}{b}{\powe{x\phi(t)} \frac{\d}{\d t} \frac{f(t)}{\phi'(t)}}{t}$ exists and is dominated by the boundary terms
\item $\Re\phi(t) < \Re \phi(b)$ on $[a,b)$, $\Re\phi'(b) \neq 0$ and $f(b) \neq 0$: $I(x) \sim \frac{1}{x} \frac{f(b)}{\phi'(b)} \powe{x\phi(b)}$
\item $\Re\phi(t) < \Re \phi(a)$ on $(a,b]$, $\Re\phi'(a) \neq 0$ and $f(a) \neq 0$: $I(x) \sim -\frac{1}{x} \frac{f(a)}{\phi'(a)} \powe{x\phi(a)}$
\end{enumerate}
Continuing integration by parts also yields
\begin{align}
I(x) \koml{x\rightarrow\infty}{\sim} \powe{x\phi(b)} \sum_{n\ge 1} A_n \frac{1}{x^n}.
\end{align}
}
\note{Failure of \gls{ibp}:
\begin{enumerate}
\item it can only contain integer powers of $\frac{1}{x}$, so fractional exponents might cause problems
\item $\phi'(t)$ has a zero somewhere on $\klb*{a,b}$
\item when \gls{ibp} produces an integral that does not exist
\end{enumerate}
}
\example{Four examples}{}{\begin{enumerate}
\item $I(x) = \intx{0}{\infty}{\powe{-xt^2}}{t} = \frac{1}{2} \sqrt{\frac{\pi}{x}}$, but \gls{ibp} gives $I(x) = \intx{0}{\infty}{\frac{-1}{2xt}}{\powe{-xt^2}}$ which diverges at boundary (and remaining integral does not exist either)
\item $I(x)=\intx{0}{\infty}{\powe{-x\sinh(t)^2}}{t}$ has $\phi'(0)=0$; can be cured by Laplace's method
\item $I(x) = \intx{0}{\infty}{\log(1+t) \powe{-x\sinh(t)^2}}{t} = -\frac{1}{x} \frac{\log(1+t)}{\sinh(2t)} \powe{-x\sinh(t)^2} \Big\vert_0^\infty + \frac{1}{x} \intx{0}{\infty}{\powe{-x\sinh(t)^2} \frac{\d}{\d t} \frac{\log(1+t)}{\sinh(2t)}}{t}$ gives $I(x) = \frac{1}{2x} + \mathcal{O}\kla*{x^{-3/2}}$ (last term requires Laplace again, since non-integer power)
\item Stieltjes Integral: $I(x) = \intx{0}{\infty}{\frac{1}{1+xt} \powe{-t}}{t}$ as $x\rightarrow\infty$ (note: \gls{ibp} useful for small $x$) \begin{align}
I(x) &= -\intx{0}{\infty}{\frac{1}{1+tx}}{\powe{-t}} = -\frac{\powe{-t}}{1+tx}\bigg\vert_0^\infty - x \intx{0}{\infty}{\frac{1}{(1+xt)^2} \powe{-t}}{t} \\
&= \dots = 1 - x + 2!x^2 + \dots + (-1)^{n-1} (n-1)!x^{n-1} + (-1)^n n! x^n \intx{0}{\infty}{\frac{1}{(1+xt)^{n+1}} \powe{-t}}{t},
\end{align}
so $I(x) \sim \sum_{n\ge 0} (-1)^n n! x^n$ as $x\rightarrow 0^+$ (only asymptotic, does not converge anywhere); for $x\rightarrow\infty$, the local behavior of $x^2I'' + (1+3x)I'+I=0$ includes $\frac{\log x}{x}$ (not even a power, let alone an integral one)
\end{enumerate}
}
\section{Laplace's Method and Watson's Lemma}
Idea: if the $f,\phi$ are real continuous functions and $\phi$ has its maximum on the integration interval, say at $t=c\in\kla*{a,b}$, then only the neighborhood of $c$ contributes to the \emph{full asymptotic expansion} as $x\rightarrow\infty$.
Examples:
\begin{enumerate}
\item $I(x,\epsilon) = \intx{c-\epsilon}{c+\epsilon}{f(t) \powe{x\phi(t)}}{t}$ if $c\in\kla*{a,b}$
\item $I(x,\epsilon) = \intx{a}{a+\epsilon}{f(t)\powe{x\phi(t)}}{t}$ if $c=a$ (and similarly for $c=b$)
\end{enumerate}
\definition{Full Asymptotic Expansion of $I(x,\epsilon)$ as $x\rightarrow\infty$}{\begin{enumerate}
\item does not depend on $\epsilon$
\item is identical to the full expansion of $I(x)$, because $I$ over the remaining intervals is subdominant or \gls{tst} compared to $I(x)$
\end{enumerate}
}
\example{Stieltjes again}{$\intx{0}{10}{\frac{1}{1+t} \powe{-tx}}{t}$ as $x\rightarrow\infty$}{\begin{enumerate}
\item leading behavior: $\phi(t)=-t$ has maximum at $t=0$
\begin{itemize}
\item using $I(x,\epsilon) = \intx{0}{\epsilon}{(\dots)}{t} \sim \intx{0}{\epsilon}{\powe{-xt}}{t} \sim \frac{1}{x}$ only has exponentially small error 
\end{itemize}
\item full asym. expansion
\begin{itemize}
\item $I(x,\epsilon) = \sum_{n\ge 0} (-1)^n \intx{0}{\epsilon}{t^n \powe{-xt}}{t} \sim \sum_{n\ge 0} (-1)^n n! x^{-n-1}$
\item error term $\intx{\epsilon}{\infty}{(\dots)}{t} \sim (-\epsilon)^n \frac{1}{x} \powe{-\epsilon x} \ll \frac{1}{x}$ is exponentially small
\end{itemize}
\end{enumerate}
}
\example{Error Function}{$\intx{-10}{10}{\powe{-xt^2}}{t}$ as $x\rightarrow\infty$}{\gls{ibp} fails because $\phi'(0)=0$.

However, $I(x) \sim I(x,\epsilon) = \intx{-\epsilon}{\epsilon}{(\dots)}{t} \sim \intr{\powe{-xt^2}}{t} = \sqrt{\frac{\pi}{x}}$ with exponentially small error.
}
Summary:
\begin{enumerate}
\item approximate $I(x)$ by $I(x,\epsilon)$: restrict the original integration to a narrow region surrounding the maximum of $\phi(t)$
\item expand $f(t)$ and $\phi(t)$ in series (Taylor or asymptotic) valid near the peak
\note{$I(x,\epsilon)$ now becomes a series of integrals, so $\epsilon$ must be sufficiently ``small'' in order to have a valid expansion}
\item extend the integration range in each integral to infinity (or whatever else is useful)
\note{Changing integration limits only introduces exponentially small terms (\gls{tst})}
\end{enumerate}
\subsection{Watson's Lemma}
\definition{Watson's Lemma}{Consider $I(x) = \intx{0}{b}{f(t) \powe{-xt}}{t}$ for $b>0$ and $x\rightarrow \infty$.
If the function satisfies
\begin{itemize}
\item $f$ is continuous
\item $f(t) \sim t^\alpha \sum_{n\ge 0} a_n t^{\beta n}$ as $t\rightarrow 0^+$ with $\alpha > -1$ and $\beta > 0$,
\end{itemize}
\note{If $b=\infty$ we also need that $f(t)\ll \powe{ct}$ for some $c>0$ to ensure convergence}
then the \emph{full asymptotic expansion} of any integral of this type is given by
\begin{align}
I(x) &\koml{x\rightarrow\infty}{\sim} \sum_{n\ge 0} \frac{a_n \Gamma\kla*{\alpha + \beta n+ 1}}{x^{\alpha + \beta n + 1}}.
\end{align}
To show this, choose $\epsilon$ such that
\begin{align}
\abs*{f(t) - t^\alpha \sum_{n=0}^N a_n t^{\beta n}} &\le K t^{\alpha + \beta (N+1)} \quad \text{for} \quad 0\le t \le \epsilon.
\end{align}
Then the error introduced by replacing $f$ by this finite series is bounded by
\begin{align}
K\intx{0}{\epsilon}{t^{\alpha + \beta (N+1)} \powe{-xt}}{t} \le K \frac{\Gamma\kla*{\alpha + \beta(N+1)+1}}{x^{\alpha + \beta(N+1)+1}},
\end{align}
and $\abs*{I(x) - \sum_{n=0}^N a_n \frac{\Gamma\kla*{\alpha + \beta n + 1}}{x^{\alpha + \beta n + 1}}} \ll \frac{1}{x^{\alpha + \beta N + 1}}$ for all $N$.
}
\example{Yet another Stieltjes integral}{$I(x) = \intx{0}{5}{\frac{1}{1+t^2} \powe{-xt}}{t}$ as $x\rightarrow \infty$}{$c=0$, so $I(x,\epsilon) = \intx{0}{\epsilon}{\powe{-xt} \sum_{n\ge 0} (-1)^n t^{2n}}{t}$ for small $\epsilon$ and $I(x) \sim \sum_{n\ge 0} (-1)^n \frac{\Gamma(2n+1)}{x^{2n+1}}$ ($\alpha = 0$, $\beta = 2$)
}
\example{Modified Bessel function}{$K_0(x) = \intx{1}{\infty}{\kla*{s^2 - 1}^{-1/2} \powe{-xs}}{s}$}{With $t=s-1$ we find $I(x,\epsilon) = \intx{0}{\epsilon}{\powe{-x} \kla*{t^2+2t}^{-1/2} \powe{-xt} }{t}$.
\note{$\binom{x}{y} = \frac{\Gamma(x+1)}{\Gamma(y+1) \Gamma(x-y+1)}$ and $(-1)^k \binom{z}{k} = \binom{-z+k-1}{k}$ give $\binom{-1/2}{n} = (-1)^n \binom{n-1/2}{n} = (-1)^n \frac{\Gamma\kla*{n+\frac{1}{2}}}{n! \Gamma\kla*{\frac{1}{2}}}$}
For $|t|<2$, we have
\begin{align}
\kla*{t^2+2t}^{-1/2} = (2t)^{-1/2} \sum_{n\ge 0} \kla*{\frac{-t}{2}}^n \frac{\Gamma\kla*{n+\frac{1}{2}}}{n! \Gamma\kla*{\frac{1}{2}}},
\end{align}
so $\alpha = -\frac{1}{2}$, $\beta=1$ and thus
\begin{align}
K_0(x) &\koml{x\rightarrow\infty}{\sim} \powe{-x} \sum_{n\ge 0} (-1)^n \frac{\Gamma\kla*{n + \frac{1}{2}}^2}{2^{n+1/2} n! \Gamma\kla*{\frac{1}{2}}} \frac{1}{x^{n+1/2}}.
\end{align}
}
\example{Parabolic Cylinder function}{$D_\nu(x) = \frac{1}{\Gamma(-\nu)} \powe{-x^2/4} \intx{0}{\infty}{ t^{-\nu-1} \powe{-t^2/2} \powe{-xt}}{t}$ for $\Re \nu < 0$}{Here, $\powe{-t^2/2} = \sum_{n\ge 0} (-1)^n \frac{t^{2n}}{2^n n!}$, so $\alpha = -\nu-1$, $\beta=2$ and Watson's Lemma immediately gives
\begin{align}
D_\nu(x) &\sim \frac{1}{\Gamma(-\nu)} \powe{-x^2/4} \sum_{n\ge 0} (-1)^n \frac{\Gamma\kla*{2n-\nu}}{2^n n!} \frac{1}{x^{2n-\nu}}
\end{align}
}
\subsection{Beyond Watson's Lemma}
Watson's Lemma only applies to the special case $\phi(t) = -t$.
\definition{Indirect Watson's Lemma}{Let $s=-\phi(t)$ to get $I(x) = \intx{-\phi(a)}{-\phi(b)}{F(s) \powe{-xs}}{s}$ with $F(s) = \frac{-f(t)}{\phi'(t)}$ (and then apply Watson again, hoping that the conditions are met by the new function)
}
\example{Trigonometric Gauss integral}{$I(x) = \intx{0}{\pi/2}{\powe{-x\sin(t)^2}}{t}$ as $x\rightarrow\infty$}{$\phi(t) = \sin(t)^2$ gives $t=\arcsin\kla*{\sqrt{s}}$ and
\begin{align}
I(x) &= \intx{0}{1}{\powe{-xs} \frac{1}{2\sqrt{s}\sqrt{1-s}}}{s}
\end{align}
yields $\alpha = -\frac{1}{2}$, $\beta = 1$ and $a_n = \frac{\Gamma\kla*{n+\frac{1}{2}}}{n!\Gamma\kla*{\frac{1}{2}}}$, resulting in
\begin{align}
I(x) &\koml{x\rightarrow\infty}{\sim} \frac{1}{2} \sum_{n\ge 0} \frac{\Gamma\kla*{n+\frac{1}{2}}^2}{n! \Gamma\kla*{\frac{1}{2}}} \frac{1}{x^{n+1/2}}.
\end{align}
}
\definition{Direct asymptotic behavior of $I(x)$}{Replace $I(x)$ by $I(x,\epsilon)$ as usual and replace $\phi(t)$ by the first few terms in its Taylor series.
\begin{enumerate}
\item if $c$ is an endpoint and $\phi'(c)\neq 0$, then $\phi(t) \sim \phi(c) + (t-c)\phi'(c)$
\begin{itemize}
\item if $c=a$, then $\phi'(a) < 0$ and $I(x,\epsilon) \sim \intx{a}{a+\epsilon}{f(a) \powe{x\phi(a)} \powe{x(t-a) \phi'(a)}}{t} \sim f(a) \powe{x\phi(a)} \intx{a}{\infty}{\powe{x(t-a)\phi'(a)}}{t}$ and $I(x) \sim -\frac{f(a)}{x\phi'(a)} \powe{x\phi(a)}$
\item if $c=b$, then $\phi'(b)>0$ and $I(x) \sim \frac{f(b)}{x\phi'(b)} \powe{x\phi(b)}$
\end{itemize}
\item if $a<c<b$ and $\phi'(c)=0$ but $\phi''(c)\neq 0$, then $\phi(t) \sim \phi(c) + \frac{1}{2} (t-c)^2 \phi''(c)$
\begin{itemize}
\item we must have $\phi''(c) < 0$ for convergence
\item it follows that \begin{align}
I(x,\epsilon) &\sim \intx{c-\epsilon}{c+\epsilon}{f(c) \powe{x\phi(c)} \powe{\frac{1}{2}(t-c)^2 \phi''(c)}}{t} \sim f(c) \powe{x\phi(c)} \sqrt{\frac{2\pi}{x|\phi''(c)|}}
\end{align}
\note{If this happens at $t=a$ or $t=b$ (i.e. $c$ is at the boundary), then only half of the peak contributes and we need a factor of $\frac{1}{2}$ to correct this}
\end{itemize}
\item in general, if $\phi^{(p)}(c) \neq 0$ is the first non-vanishing term, then $\phi(t) \sim \phi(c) + \frac{1}{p!} (t-c)^p \phi^{(p)}(c)$
\begin{itemize}
\item if $a<c<b$, then $p$ must be even and $\phi^{(p)}(c) < 0$
\item it follows that \begin{align}
I(x,\epsilon) &\sim f(c) \powe{x\phi(c)} \intr{\powe{x (t-c)^p \frac{1}{p!} \phi^{(p)}(c)}}{t},
\end{align}
which gives $I(x) \sim f(c) \powe{x\phi(c)} \frac{2}{p} \Gamma\kla*{\frac{1}{p}} \kla*{\frac{p!}{x\abs*{\phi^{(p)}(c)}}}^{1/p}$ as $x\rightarrow\infty$
\end{itemize}
\end{enumerate}
}
\example{Several examples}{}{\begin{enumerate}
\item $\intx{0}{\pi/2}{\powe{-x\tan(t)}}{t} \sim -\frac{f(0)}{x\phi'(0)} \powe{x\phi(0)} = \frac{1}{x}$ because $\phi(t) = -\tan(t)$ gives $c=0$ and $\phi'(0)=-1 \neq 0$
\item $\intx{0}{\infty}{\powe{-x\sinh(t)^2}}{t}$: $c=0$ but $\phi'(0)=0$ and $\phi''(0) = 2$, so $I(x)\sim \frac{1}{2}\sqrt{\frac{\pi}{x}}$
\item $\intx{-1}{1}{\powe{-x\sin(t)^4}}{t}$ has $c=0$, $p=4$, $\phi^{(p)}(c) = -4!$, so $I(x) \sim \frac{1}{2} \Gamma\kla*{\frac{1}{4}} \frac{1}{x^{1/4}}$
\item $\intx{-\pi/2}{\pi/2}{(t+2) \powe{-x\cos(t)}}{t}$ has $c=\pm \frac{\pi}{2}$ and $\phi'(c) = \pm 1$, giving \begin{align}
I(x)&\sim \frac{f(b) \powe{x\phi(b)}}{x\phi'(b)} - \frac{f(a)}{x\phi'(a)} \powe{x\phi(a)} = \frac{\frac{\pi}{2} + 2}{x} + \frac{-\frac{\pi}{2} + 2}{x} = \frac{4}{x}
\end{align}
\item $\intx{0}{1}{\sin(t) \powe{-x\sinh(t)^4}}{t}$ has $c=0$ with $f(c)=0$, but Laplace is fine because $f$ vanishes \emph{algebraically}!
\note{Note that $\frac{\Gamma(a)}{x^a} = \intx{0}{\infty}{t^{a-1} \powe{-xt}}{t}$}
\begin{align}
I(x) &\sim \intx{0}{\epsilon}{t \powe{-xt^4}}{t} \sim \frac{\Gamma\kla*{\frac{1}{2}}}{4x^{1/2}}
\end{align}
\item $\intx{0}{\infty}{\sinh(t)^{-1/2} \powe{-x\cosh(t)}}{t}$ has $c=0$ with algebraically diverging $f(t)$, so \begin{align}
I(x) &\sim \intx{0}{\epsilon}{t^{-1/2} \powe{-x\kla*{1 + \frac{1}{2}t^2}}}{t} = \powe{-x} \frac{\Gamma\kla*{\frac{1}{4}}}{\kla*{8x}^{1/4}}
\end{align}
\end{enumerate}
}
\example{Modified Bessel function of second kind}{$K_\nu(x) = \intx{0}{\infty}{\powe{-x\cosh(t)} \cosh(\nu t)}{t}$ for $x>0$}{$c=0$ with $\phi''(0)=-1$, so $K_\nu(x) \sim \frac{1}{2} f(0) \powe{x\phi(0)} \sqrt{\frac{2\pi}{-x\phi''(0)}} = \sqrt{\frac{\pi}{2x}}\powe{-x}$
}
\example{Modified Bessel function of first kind}{$I_\nu(x) = \frac{1}{\pi} \intx{0}{\pi}{\powe{x\cos(t)} \cos(\nu t)}{t}$ as $x\rightarrow\infty$}{\begin{enumerate}
\item direct local analysis of $x^2 y''+xy'-\kla*{x^2+\nu^2}y=0$ with $y=\powe{S}$
\begin{itemize}
\item $S'' + \kla*{S'}^2 + \frac{1}{x} S' - \kla*{1 + \frac{\nu^2}{x^2}} = 0$
\item \gls{db} gives $\kla*{S'}^2 \sim 1$, so $S(x) = x+C(x)$
\item $C'' + \kla*{1 + C'}^2 + \frac{1+C}{x} - \kla*{1 + \frac{\nu^2}{x^2}} = 0$
\item \gls{db} gives $C' \sim -\frac{1}{2x}$, so $C(x) \sim -\frac{1}{2} \log(x)$
\item $y(x) \sim C_1 x^{-1/2} \powe{x}$ and we cannot determine $C_1$
\end{itemize}
\item integral representation has $c=0$ and $\phi''(0)=-1$, so $I_\nu(x) \sim \frac{1}{2} \frac{\powe{x}}{\pi} \sqrt{\frac{2\pi}{x}} = \frac{1}{\sqrt{2\pi x}} \powe{x}$
\note{The leading order does not depend on $\nu$, but the higher orders do}
\end{enumerate}
}
\subsection{Laplace's Method: higher-order terms}
Consider $c\in\kla*{a,b}$, $\phi'(c)=0$ and $\phi''(c)<0$ with $f(c)\neq 0$ (and finite of course).
Then as $x\rightarrow\infty$,
\begin{align}
I(x) &\sim \intx{c-\epsilon}{c+\epsilon}{f(t) \powe{x\phi(t)}}{t}\\
&\sim \intx{c-\epsilon}{c+\epsilon}{\klb*{f(c) + f'(c)(t-c) + \frac{1}{2} f''(c)(t-c)^2} \powe{x\klb*{\phi(c) + \frac{1}{2} \phi''(c) (t-c)^2 + \dots}}}{t}.
\end{align}
Expanding the terms from the exponent beyond second order, we have
\begin{align}
\powe{x\klb*{\dots}} &= 1 + x \klb*{\frac{1}{3!} (t-c)^3 \phi'''(c) + \frac{1}{4!} (t-c)^4 \phi^{(4)}(c) + \dots} + \frac{1}{2! 3!^2} x^2 (t-c)^6 \kla*{\phi'''(c)}^2 + \dots
\end{align}
and inserting this into the integral with $s=\sqrt{x}(t-c)$ yields (odd powers of $(t-c)$ vanish!)
\begin{align}
I(x) &\sim \frac{1}{\sqrt{x}} \powe{x\phi(c)} \intx{c-\epsilon}{c+\epsilon}{\powe{\frac{1}{2} s^2\phi''(c)} \klc*{f(c) + \frac{1}{x} \klb*{\frac{1}{2}s^2 f''(c) + \frac{1}{4!} s^4 f(c) \phi^{(4)}(c) + \frac{1}{3!} s^4 f'(c) \phi'''(c) + \frac{1}{2!3!^2} s^6 f(c) \kla*{\phi'''(c)}^2 + \dots}} }{s}\\
&\sim \sqrt{\frac{2\pi}{-\phi''(c)x}} \powe{x\phi(c)} \klc*{f(c) + \frac{1}{x}\klb*{\frac{-f''(c)}{2\phi''(c)} + \frac{f(c)\phi^{(4)}(c)}{8\kla*{\phi''(c)}^2} + \frac{f'(c)\phi'''(c)}{2\kla*{\phi''(c)}^2} - \frac{5\kla*{\phi'''(c)}^2 f(c)}{4! \kla*{\phi''(c)}^3} + \dots}}.
\end{align}
\note{$\intr{\powe{-s^2/2} s^{2n}}{s} = \sqrt{2\pi} (2n-1)!!$}
\note{\begin{itemize}
\item expanding $\powe{\frac{1}{2}(t-c)^2\phi''(c)}$ would make the result depend on $\epsilon$ (need the ``re-summation'' from the full gaussian)
\item analogously, \emph{not} expanding the higher order terms would give divergent contributions when sending the limits to infinity (if e.g. $\phi^{(4)}(c) > 0$)
\end{itemize}}
\example{Exponential trig integral}{$I(x) = \intx{0}{\pi/2}{\powe{-x\sin(t)^2}}{t}$ as $x\rightarrow\infty$}{\begin{align}
I(x) &\sim \intx{0}{\epsilon}{\powe{-x\kla*{t^2-\frac{1}{3}t^4}}}{t} \sim \intx{0}{\epsilon}{\powe{-xt^2} \kla*{1 + \frac{xt^4}{3}}}{t}\\
&\sim \intx{0}{\infty}{\powe{-xt^2} \kla*{1 + \frac{xt^4}{3}}}{t} = \frac{1}{2} \sqrt{\frac{\pi}{x}} \kla*{1 + \frac{1}{4x}}
\end{align}
}
\example{Exponential hyperbolic integral}{$I(x) = \intr{\powe{-x\cosh(t)}}{t}$ as $x\rightarrow\infty$}{Peak is at $c=0$ and $u = \sqrt{\cosh(t)-1}$ gives $I(x) = \intr{\powe{-x(1+u^2)} \frac{2u}{u\sqrt{2+u^2}}}{u} = \sqrt{2} \powe{-x} \intr{\powe{-xu^2} \kla*{1 + \frac{1}{2}u^2}^{-1/2}}{u}$.
Then, Laplace's method gives
\begin{align}
I(x) &\sim \sqrt{2}\powe{-x} \intx{-\epsilon}{\epsilon}{\powe{-xu^2} \kla*{1 - \frac{1}{4}u^2 + \frac{3}{32} u^4 + \dots}}{u} \sim \sqrt{2}\powe{-x} \intr{\powe{-xu^2} \kla*{\dots}}{u}\\
&= \sqrt{2}\powe{-x} \sqrt{\frac{\pi}{x}} \kla*{1 - \frac{1}{8x} + \frac{1}{128x^2} + \dots}
\end{align}
}
\subsection{Movable Maxima}
We have to consider the indirect application of Laplace's method when
\begin{enumerate}
\item $f(t)$ vanishes exponentially fast at $t=c$
\item the Laplace integral converges while $\phi(c)=\infty$
\end{enumerate}
\example{Watson falls short}{$I(x) = \intx{0}{\infty}{\powe{-xt - 1/t}}{t}$ as $x\rightarrow\infty$}{$\phi(t)=-t$ has its peak at $c=0$, but $f(t)=\powe{-1/t}$ vanishes exponentially fast, so Watson's Lemma fails (asymptotic series is just zeros + \gls{tst}).

The maximum of the full function $f(t)\powe{x\phi(t)}$ is given by $0=-x+\frac{1}{t^2}$, i.e. $c=\frac{1}{\sqrt{x}}$ depends on $x$ -- this is movable!
We can introduce a scale transformation via $s=\frac{t}{\sqrt{x}}$, giving $I(x) = \frac{1}{\sqrt{x}} \intx{0}{\infty}{\powe{-\sqrt{x} \kla*{s + 1/s}}}{s}$.
Now $c=1$ is not problematic for $\tilde{f}\equiv 1$, so
\begin{align}
I(x) &\sim \frac{1}{\sqrt{x}} \intx{1-\epsilon}{1+\epsilon}{\powe{-\sqrt{x}\kla*{2 + (s-1)^2}}}{s} \sim \powe{-2\sqrt{x}} \frac{1}{\sqrt{x}} \intr{\powe{-\sqrt{x} (s-1)^2}}{s} = \sqrt{\pi} x^{-3/4} \powe{-2\sqrt{x}}.
\end{align}
}
\example{Stirling Formula}{$x! = \intx{0}{\infty}{t^x \powe{-t}}{t}$ as $x\rightarrow\infty$}{Maximum of the integrand is at the maximum of $-t+x\log(t)$, i.e. $0=-1+\frac{x}{t}$, so $c=x$ is again movable.
Transforming into the moving frame with $s=\frac{t}{x}$, we now have $c=1$ and
\begin{align}
x! &= x^{x+1} \intx{0}{\infty}{\powe{x \kla*{\log(s)-s}}}{s} \sim x^{x+1} \intx{1-\epsilon}{1+\epsilon}{\powe{-x\kla*{1 + \frac{1}{2}(1-s)^2 + \dots}}}{s}\\
&\sim x^{x+1} \powe{-x} \intr{\powe{-\frac{x}{2}(1-s)^2}}{s} = x^{x+1} \powe{-x} \sqrt{\frac{2\pi}{x}}.
\end{align}
The higher-order terms follow from $s-\log(s) = 1+u^2$, with $u^2 = \sum_{n\ge 2} \frac{1}{n}(1-s)^n$ for $|s-1|<1$.
Inverting the series yields
\begin{align}
s &= 1 + \sqrt{2}u + \frac{2}{3}u^2 + \frac{1}{9\sqrt{2}} u^3 + \dots,
\end{align}
and inserting into our integral gives
\begin{align}
x! &\sim x^{x+1} \powe{-x} \intr{\powe{-xu^2} \kla*{\sqrt{2} + \frac{4}{3}u + \frac{1}{3\sqrt{2}} u^2 + \dots} }{u}\\
&\sim \sqrt{2} x^{x+1} \powe{-x} \intr{\powe{-xu^2} \kla*{1 + \frac{u^2}{6} + \frac{u^4}{270} + \dots}}{u} = \sqrt{2\pi x} x^x \powe{-x} \kla*{1 + \frac{1}{12x} + \frac{1}{288x^2} + \dots}
\end{align}
}
\section{Method of Stationary Phase}
One generalization of Laplace's method is $\phi(t) \rightarrow\i\psi(t)$ with $\psi(t)\in\mathbb{R}$.
\definition{Generalized Fourier Integral}{$I(x) = \intx{a}{b}{f(t) \powe{\i x\psi(t)}}{t}$ where $f,\psi,a,b,x$ are real.}
\example{Integration by Parts}{$I(x) = \intx{0}{1}{\frac{\powe{\i xt}}{1+t}}{t}$}{Sometimes \gls{ibp} works,
\begin{align}
I(x) &= \intx{0}{1}{\frac{1}{1+t} \frac{1}{\i x}}{\kla*{\powe{\i xt}}} = \frac{1}{\i x} \frac{\powe{\i xt}}{1+t}\bigg\vert_0^1 - \frac{1}{\i x} \intx{0}{1}{\frac{-1}{(1+t)^2} \powe{\i xt}}{t}\\
&= \frac{1}{2\i x} \powe{\i x} - \frac{1}{\i x} - \frac{\i}{x} \intx{0}{1}{\frac{\powe{\i xt}}{1 + t^2}}{t}.
\end{align}
Repeating this process gives the complete asymptotic expansion of $I(x)$,
\begin{align}
I(x) &\sim \frac{\powe{\i x}}{2\i x} - \frac{1}{\i x} + \frac{1}{x^2} \frac{\powe{\i xt}}{(1+t)^2} \bigg\vert_0^1 - \frac{1}{x^2} \intx{0}{1}{\frac{-2}{(1+t)^3} \powe{\i xt}}{t} = \dots = \frac{1}{\i x} \kla*{\frac{1}{2} \powe{\i x} - 1} + \frac{1}{(\i x)^2} \kla*{\frac{1}{4} \powe{\i x} - 1} + \dots
\end{align}
}
\example{\gls{ibp} fails}{$I(x) = \intx{0}{1}{\sqrt{t} \powe{\i xt}}{t}$}{This time we get $\frac{1}{\i x} \powe{\i x} - \frac{1}{2\i x} \intx{0}{1}{\frac{1}{\sqrt{t}} \powe{\i xt}}{t}$, where the remaining integral diverges at the next iteration.
\note{$\intx{0}{\infty}{\powe{\i s} s^{\alpha - 1}}{s} = \powe{\i \frac{\pi}{2}\alpha} \Gamma(\alpha)$ for $\Re(\alpha) \in \kla*{0,1}$}
Cure: $s=xt$ and $\frac{1}{2\i x} = \frac{1}{2\i x^{3/2}} \intx{0}{x}{\powe{\i s} s^{-1/2}}{s} \sim \frac{1}{2\i} x^{-3/2} \sqrt{\pi} \powe{\i \pi/4}$.
The fractional power can not come from \gls{ibp}!

The full asymptotic expansion be obtained from $\intx{0}{1}{t^{-1/2} \powe{\i xt}}{t} = \intx{0}{\infty}{(\dots)}{t} - \intx{1}{\infty}{(\dots)}{t}$, giving
\begin{align}
I(x) &\sim \frac{\i \sqrt{\pi}}{2x^{3/2}} \powe{\i\pi /4} + \frac{1}{\i x\sqrt{\pi}} \kla*{1 + \frac{1}{2} \sum_{n\ge 0} \frac{\Gamma\kla*{n + \frac{1}{2}}}{(\i x)^{n+1}}}
\end{align}
as $x\rightarrow\infty$.
}
\subsection{Motivation: Rapid oscillations}
We have $I(\omega) = \intx{a}{b}{\cos(\omega t)}{t} = \mathcal{O}\kla*{\frac{1}{\omega}}$ as $\omega\rightarrow\infty$.
Similarly,
\begin{align}
\intx{a}{b}{f(t)\cos(\omega t)}{t} &= \frac{1}{\omega} \sin(\omega t) f(t) \Big\vert_a^b - \frac{1}{\omega} \intx{a}{b}{f'(t)\sin(\omega t)}{t} = \mathcal{O}\kla*{\frac{1}{\omega}} + o\kla*{\frac{1}{\omega}}
\end{align}
if $f(a),f(b)$ are both finite and non-zero.
This can be justified by the Riemann-Lebesgue Lemma.
\definition{Riemann-Lebesgue Lemma}{If $\intx{a}{b}{\abs*{f(t)}}{t}$ exists, then $\intx{a}{b}{f(t)\powe{\i\omega t}}{t} \rightarrow 0$ as $\omega\rightarrow\infty$.
Furthermore, if $f'$ is finite on $\klb*{a,b}$, then $\intx{a}{b}{f(t) \powe{\i\omega t}}{t} = \mathcal{O}\kla*{\frac{1}{\omega}}$.
}
\example{A generalized Fourier integral}{$I(x) = \intx{a}{b}{f(t) \powe{\i x\psi(t)}}{t}$ as $x\rightarrow\infty$}{Use \gls{ibp} to get $I(x) = \frac{f(t)}{\i x\phi'(t)} \powe{\i x\psi(t)} \Big\vert_a^b - \frac{1}{\i x} \intx{a}{b}{\frac{\d}{\d t} \kla*{\frac{f(t)}{\psi'(t)}} \powe{\i x\psi(t)}}{t}$.
This gives the leading behavior for large $x$, provided that $\frac{f(t)}{\psi'(t)}$ is smooth for $t\in\klb*{a,b}$ and non-zero at one of the endpoints.

The Riemann-Lebesgue lemma shows that the remaining integral vanishes more rapidly than $\frac{1}{x}$, therefore $I(x) \sim \frac{f(t)}{\i x\phi'(t)} \powe{\i x\psi(t)} \Big\vert_a^b $.
\note{Potential issue: \gls{ibp} may fail if $\psi'(t)=0$ for some $t\in\klb*{a,b}$, corresponding to a stationary point.
We will show that $I(x)$ still vanishes, but usually less rapidly than $\frac{1}{x}$.}
}
The major contribution to $I(x)$ comes from a neighborhood of points with stationary phase (assuming $f\neq 0$ there), since the oscillatory cancellation effect disappears near stationary points.

The stationary phase method gives the leading behavior of the generalized Fourier integrals having stationary points, i.e. analogous to the Laplace method, we may focus on $\klb*{c-\epsilon,c+\epsilon}$ where $c$ is a stationary point.
\definition{Stationary Phase Method}{Consider $t=a$ is a stationary point.
Then
\begin{align}
I(x) &= \intx{a}{a+\epsilon}{f(t)\powe{\i x\psi(t)}}{t} + \intx{a+\epsilon}{b}{f(t)\powe{\i x\psi(t)}}{t},
\end{align}
where the second integral vanishes $\sim \frac{1}{x}$ since there are no stationary points in the interval.
Suppose that $\psi^{(p)}(a)\neq 0$ is the first non-zero derivative.
Then
\begin{align}
I(x) &\sim \intx{a}{a+\epsilon}{f(a) \powe{\i x\psi(a) + \i x\frac{1}{p!} \psi^{(p)}(a) (t-a)^p}}{t}\\
&\sim f(a) \powe{\i x\psi(a)} \intx{0}{\infty}{\powe{\frac{\i x}{p!} \psi^{(p)}(a) s^p}}{s}.
\end{align}
If $\psi^{(p)}(a)>0$, we may rotate the contour by the angle $\frac{\pi}{2p}$, and similarly $-\frac{\pi}{2p}$ if the derivative is negative (this will turn $\i s^p$ into $-s^p$ later, essentially this is a generalized Wick rotation).
Define $s=\powe{\pm \i \frac{\pi}{2p}} \kla*{\frac{p! u}{\pm x\psi^{(p)}(a)}}^{1/p}$ with $u\in\mathbb{R}$.
Then we find
\begin{align}
I(x) &\sim f(a) \powe{\i x\psi(a) \pm \i \frac{\pi}{2p}} \kla*{\frac{p!}{x\abs*{\psi^{(p)}(a)}}}^{1/p} \Gamma\kla*{\frac{1}{p}+1}.
\end{align}
If $f(a)=0$, it is necessary to investigate whether the contribution from the stationary point still dominates the leading behavior; it might be that $\frac{1}{x}$ is more relevant then.
\note{The Stationary Phase Method gives \emph{only} the leading order.
For higher-order terms, we need to be very careful about $\intx{a+\epsilon}{b}{(\dots)}{t}$, as it only decays algebraically.
In Laplace's method, the same integral vanishes exponentially. }
}
\example{Quadratic Phase Shift}{$I(x) = \intx{0}{\infty}{\cos\kla*{xt^2-t}}{t}$ as $x\rightarrow\infty$}{It is helpful to rewrite this as the real part of a pure exponential.
Then identifying $\psi(t)=t^2$, we have a stationary point at $c=0$ with $\psi'(c)=0$ and $\psi''(c)=2$.
Thus, $I(x) \sim \Re \klc*{\Gamma\kla*{\frac{3}{2}} f(0) \powe{\i \frac{\pi}{4}} \kla*{\frac{2}{2x}}^{1/2} } = \frac{1}{2} \sqrt{\frac{\pi}{2x}}$.
}
\example{Bessel functions}{$J_n(x) = \frac{1}{\pi} \intx{0}{\pi}{\cos\kla*{x\sin(t) - nt}}{t}$ as $x\rightarrow\infty$}{We have $J_n(n) = \Re \frac{1}{\pi} \intx{0}{\pi}{\powe{\i n\psi(t)}}{t}$ with $\psi(t)=\sin(t)-t$.
This is stationary at $t=0$ with $\psi'(0)=0$, $\psi''(0)=0$ and $\psi'''(0)=-1$.
Thus,
\begin{align}
J_n(n) &\sim \frac{1}{\pi}\Re \klc*{\Gamma\kla*{\frac{4}{3}} \powe{-\i \frac{\pi}{6}} \kla*{\frac{3!}{n}}^{1/3}} = \frac{1}{\pi} \Gamma\kla*{\frac{4}{3}} 2^{-2/3} 3^{5/6} n^{-1/3}.
\end{align}
\note{This stroboscopic behavior may in principle deviate significantly from the intermediate ranges.}
}
\example{Bessel function}{$J_0(x) = \frac{1}{\pi} \intx{0}{\pi}{\cos\kla*{x\sin(t)}}{t}$ as $x\rightarrow\infty$}{Here, $\psi(t)=\sin(t)$, $f(t)=1$ and $\psi'(c)=0$ for $c=\frac{\pi}{2}$.
Thus, the stationary phase method yields
\begin{align}
J_0(x) &\sim 2\cdot \frac{1}{\pi} \Re \klc*{\Gamma\kla*{\frac{3}{2}} \powe{-\i \frac{\pi}{4}} \powe{\i x} \sqrt{\frac{2}{x}}} = \sqrt{\frac{2}{\pi x}} \cos\kla*{x - \frac{\pi}{4}}.
\end{align}
\note{The extra factor of 2 originates from $c$ being an interior point.
This is analogous (but inverse) to the factor of $\frac{1}{2}$ for endpoints in the Laplace method.
For consistency, it would perhaps be better to include the factor in the formula.}
}
\subsection{How to derive an integral representation}
Bessel's equation is $xy''+y'+xy = 0$ and has linearly independent solutions $J_0$ and $Y_0$.
We know that $J_0(0)=1$ while $Y_0(0)=-\infty$.
The key idea is to make an integral ansatz $y(x) = \int_C \powe{xz}p(z)\,\d z$ with some function $p$ and contour $C$.
The ODE is then equivalent to $\int_C \kla*{xz^2+z+x}p(z) \powe{xz}\,\d z = 0$ for all $x$.

If we could rewrite the integrand as $\frac{\d}{\d z} \kla*{\powe{x z} S(z)}$, we would have $0 = \powe{xz} S(z)\Big\vert_{\partial C}$.
It would then be sufficient to find a contour for which this holds.
Here,
\begin{align}
\frac{\d}{\d z} \kla*{\powe{xz} S(z)} &= x\powe{xz}S(z) + \powe{xz} S'(z) = \kla*{xz^2+z+x}p(z) \powe{xz}
\end{align}
leads to $x\kla*{z^2+1}p(z) + zp(z) = xS(z) + S'(z)$.
This must hold for all $x$, and thus $S(z)=\kla*{z^2+1}p(z)$ and $S'(z) = zp(z)$.
From $\frac{S'}{S} = \frac{z}{z^2+1}$ we obtain $S(z)=\sqrt{z^2+1}$ and $p(z) = \frac{S(z)}{z^2+1} = \frac{1}{\sqrt{z^2+1}}$.
Consequently, $y(x) = \int_C \powe{xz} \frac{1}{\sqrt{z^2+1}} \,\d z$ satisfy the Bessel equation if $\powe{xz}\sqrt{z^2+1}\Big\vert_{\partial C}$ for all $x$.
Here, we can accomplish this by choosing a contour from $z=-\i$ to $z=\i$, e.g.
\begin{align}
y(x) &= \intx{-\i}{\i}{\frac{\powe{xz}}{\sqrt{z^2+1}}}{z} = \intx{-1}{1}{\frac{\powe{\i ux}}{\sqrt{1-u^2}}\i}{u}.
\end{align}
Using $y(0)=\i\pi$, we find $J_0(x) = \frac{1}{A}y(x)$ with $A=\i\pi$, and thus $J_0(x) = \frac{1}{\pi} \intx{-1}{1}{\frac{\powe{\i ux}}{\sqrt{1-u^2}}}{u}$.
The imaginary part vanishes due to symmetry, and substituting $u=\sin(t)$ yields $J_0(x) = \frac{1}{\pi} \intx{0}{\pi}{\cos\kla*{x\sin(t)}}{t}$ (we also shifted the region by $\frac{\pi}{2}$).
\section{Steepest Descent Method}
Goal: Generalization of Laplace and stationary phase methods.
Note:
\begin{enumerate}
\item Laplace Method $\intx{a}{b}{f(t)\powe{x\phi(t)}}{t}$ can give the full asymptotic expansion as $x\rightarrow\infty$, i.e. errors are exponentially small (\gls{tst})
\item stationary phase for $\intx{a}{b}{f(t) \powe{\i\psi(t)}}{t}$ only gives the leading term, errors are $\mathcal{O}\kla*{\frac{1}{x}}$
\end{enumerate}
Consider $I(x) = \intx{C}{ }{h(t)\powe{x\rho(t)}}{t}$ as $x\rightarrow\infty$ with some integration contour $C\subset \mathbb{C}$ and analytic functions $h,\rho$.
The key idea is to deform $C$ into a new contour $C'$, on which $\rho$ has (piecewise) constant imaginary part.
Then applying Laplace's method to $J(x) = \powe{\i x \Im \rho} \intx{C'}{ }{h(t)\powe{x\Re \rho(t)}}{t}$.
To motivate this, notice that the rapid oscillations in the integrand vanish for constant imaginary part.
\note{One could also deform $C$ such that the real part is constant and then apply the stationary phase method. But this only gives the leading term (it is a less powerful method).}
\definition{Steepest Descent Path}{If $f(u,v)$ is a differentiable function of two variables, then the gradient $\nabla f = \kla*{\frac{\partial f}{\partial u} ,  \frac{\partial f}{\partial v}}^\transpose$ points in the direction of the most rapid change of $f$ in the neighborhood of the point $(u,v)$.
\note{The directional derivative $\frac{\d f}{\d s} = \textbf{n} \cdot \nabla f$ gives the rate of change of $f$ in direction $\textbf{n}$ (unit vector).
The largest directional derivative is $\frac{\d f}{\d s} = \abs*{\nabla f}$ for $\textbf{n} = \frac{\nabla f}{\abs*{\nabla f}}$}
Note that 
\begin{itemize}
\item $\nabla f\perp$ level curves of $f$
\item $\frac{\d f}{\d s}$ along tangents to level curves is zero
\end{itemize}
Let $\rho(t) = \phi(t) + \i\psi(t)$ be an analytic function of $t=u+\i v$ and $\rho'(t)\neq 0$ in some region.
Then the steepest contour is defined as a contour whose tangent is parallel to $\nabla \abs*{\powe{x\rho(t)}} = \nabla \powe{x\phi(t)}$, which is parallel to $\nabla \phi$.
Therefore, a steepest contour is one on which the magnitude of $\powe{x\phi(t)}$ is changing most rapidly with $t$.
}
Claim: If $\rho(t)$ is analytic, then contours of constant $\psi$ (i.e. constant imaginary part) are steepest contours.
\definition{Cauchy-Riemann}{We know that $\partial_u \phi = \partial_v\psi$ and $\partial_v \phi = -\partial_u\psi$, which gives
\begin{align}
\nabla\phi \cdot \nabla\psi &= \kla*{\partial_u \phi} \kla*{\partial_u \psi} + \kla*{\partial_v \phi} \kla*{\partial_v \psi} = 0.
\end{align}
Thus, the directional derivative $\textbf{n}\propto \nabla\phi$ satisfies $\frac{\d \psi}{\d s} = 0$ and hence
\begin{itemize}
\item $\psi$ is constant on contours where tangents are parallel to $\nabla\phi$
\item contours of constant $\psi$ are steepest contours
\end{itemize}
}
\example{Full asymptotic expansion}{$I(x) = \intx{0}{1}{\powe{\i xt^2}}{t}$ as $x\rightarrow\infty$}{Let $\rho(t)=\i t^2$ with $\psi(t) = \Im \rho(t) = u^2-v^2$ and $\phi(t) = \Re \rho(t) = -2uv$.
The steepest path requires $\psi=\const$, and at $t=0$ this corresponds to $\Im\rho(0)=0=u^2-v^2$.
This has the two solutions $v(u) = \pm u$, which corresponds to $\phi(u) = \mp 2u^2$.
We are interested in the steepest \emph{descent}, i.e. negative $\phi$, so we choose $v(u)=+u$.
In particular, along the trajectory $t(u) = u + \i v(u) = (1+\i)u$, the maximum of $\powe{x\phi}$ now occurs at the point we started from, $t=0$.

At $t=1$, we have $\Im\rho(1)=1=u^2-v^2$, so $v=\pm\sqrt{u^2-1}$.
We want $\phi(u) = \mp 2u \sqrt{u^2-1} \le 0$, so we again choose the positive solution and note that $u\ge 1$.

On the bridge connecting the two contours deep in the imaginary part of the complex plane we do not have $\psi=\const$, but here the integrand is simply suppressed by $\phi$.
Explicitly, we have the bound
\begin{align}
\abs[\bigg]{\intx{C_\text{bridge}}{}{\powe{x \rho(t)}}{t}} \le \intx{u_1}{u_2}{\powe{x\phi(u)} f(u)}{u},
\end{align}
where $f(u)$ is some remnant from the derivative of the path.
As $V$, the upper limit on the imaginary part of the bridge contour, goes to infinity, this integral vanishes assuming that $\phi(u,V) \rightarrow -\infty$ and $v(u) \rightarrow\infty$ somewhere on $\klb*{u_1,u_2}$.

Putting all ingredients together, we find that because there are no poles contained in the contour, $I=I_1+I_2$, where
\begin{align}
I_1(x) &= (1+\i) \intx{0}{\infty}{\powe{-2xu^2}}{u} = \frac{1}{2} \sqrt{\frac{\pi}{x}} \powe{\i\frac{\pi}{4}},\\
I_2(x) &= \powe{\i x} \intx{\infty}{1}{\powe{-2xu\sqrt{u^2-1}} \kla*{1 + \i \frac{u}{\sqrt{u^2-1}}} }{u}.
\end{align}%s=2u\sqrt{u^2-1}, s^2/4 = u^4-u^2, u^2 = 1/2 +- \sqrt{1 + s^2}/2, u = +- 1/\sqrt{2} \sqrt{1 +- \sqrt{1 + s^2}} ... = \powe{\i x} \intx{0}{\infty}{\powe{-xs} \kla*{1 + \i \frac{1 \pm \sqrt{1+s^2}}{s}} \frac{1}{\sqrt{2}} \frac{\pm \frac{s}{\sqrt{1+s^2}}}{\sqrt{1 \pm \sqrt{1 + s^2}}} }{s}
The second integral can be simplified tremendously by a reparameterization of the contour.
Writing $\rho(t) = \i t^2 = \rho(1) - s$, we obtain $t = \sqrt{1 + \i s}$ where $s\in\kla*{0,\infty}$.
Inserting this into the integral yields
\begin{align}
I_2(x) &= \powe{\i x} \intx{\infty}{0}{\powe{-xs} \frac{\i}{2\sqrt{1 + \i s}}}{s}.
\end{align}
Here, we can apply Watson's Lemma with $\kla*{1+x}^{-1/2} = \sum_{n\ge 0} (-x)^n \frac{\Gamma\kla*{n+\frac{1}{2}}}{n! \Gamma\kla*{\frac{1}{2}}}$, giving
\begin{align}
I_2(x) &\sim -\frac{1}{2} \i \powe{\i x} \sum_{n\ge 0} \frac{\Gamma\kla*{n+\frac{1}{2}}}{n! \Gamma\kla*{\frac{1}{2}}} (-\i)^n \frac{\Gamma\kla*{n+1}}{x^{n+1}} = \frac{1}{2} \powe{\i x} \sum_{n\ge 0} \frac{\Gamma\kla*{n+\frac{1}{2}}}{\Gamma\kla*{\frac{1}{2}} (\i x)^{n+1}}.
\end{align}
\note{Alternative: We know that the dominant contribution of the Laplace integral comes from the neighborhood near the maximum of $\phi(t)$ at $t=1$.
Only considering the vertical path $t=1+\i s$ actually gives the same result!}
}
\example{Logarithmic Integral}{$I(x) = \intx{0}{1}{\log(t)\powe{\i xt}}{t}$ as $x\rightarrow\infty$}{Note that the stationary phase method fails because there is no stationary point, and \gls{ibp} fails because the logarithm diverges at the lower bound.

We have $\rho(t)=\i t = -v+\i u$, and at $t=0$ constant phase is equivalent to $\psi(0)=0=u$.
Steepest descent for $v>0$, so the first contour is the positive imaginary axis and we get
\begin{align}
I_1 &= \intx{0}{\infty}{\log(\i v)\powe{- xv} \i}{v} = \frac{\i}{x} \log(\i) - \i \frac{\log(x)}{x} + \frac{\i}{x} \intx{0}{\infty}{\log(s)\powe{-s}}{s} = -\i \frac{\log(x)}{x} - \frac{\i\gamma}{x} - \frac{\pi}{2x}.
\end{align}
At $t=1$ we have $\psi(1)=1=u$, so the second contour is parallel to the first.
This part yields
\begin{align}
I_2 &= \intx{\infty}{0}{\log\kla*{1 + \i v} \powe{-xv + \i x} \i}{v} = -\i\powe{\i x} \sum_{n\ge 1} \frac{(-\i)^n (n-1)!}{x^{n+1}}.
\end{align}
Combining both, we find
\begin{align}
I(x) &\sim I_1+I_2 \sim -\i \frac{\log(x)}{x} - \frac{\i\gamma + \pi/2}{x} - \powe{\i x} \sum_{n\ge 0} \frac{n!}{(\i x)^{n+2}}.
\end{align}
}
\section{Saddle Points}
Consider $I(x) = \intx{a}{b}{h(t) \powe{x\rho(t)}}{t}$ as $x\rightarrow\infty$ where $\rho(t) = \phi(t) + \i\psi(t)$ is an analytic function of $t=u+\i v$.
Along contours of constant $\psi$, points of maximal $\phi$ make dominant contributions to $I(x)$.
In our previous discussion, these points were always endpoints -- what happens if $\phi$ has an interior maximum?
\begin{enumerate}
\item if $\phi(t)$ has an interior maximum along a constant phase contour, the directional derivative vanishes: $\frac{\d \phi}{\d s} = \abs*{\nabla\phi}=0$ (in addition to the usual $\nabla\psi=0$). Thus, $\rho'(t)=0$ at an interior maximum
\item a point at which $\rho'(t)=0$ is called Saddle Point
\item saddle points are special: it is only at such a point that two distinct steepest curves can intersect. When $\rho'(t_0)\neq 0$, the steepest curve passing through $t_0$ tangent to $\nabla\phi$ is unique
\end{enumerate}
\example{Gaussian}{$\rho(t)=t^2$}{$\rho'(t)=2t$, so $t=0$ is a saddle point.
We have $\rho(t)=u^2-v^2+2\i uv$, so $\phi(t)=u^2-v^2$ and $\psi(t)=2uv$.
At the saddle point, $\rho(0)=0$ and the steepest curves are determined by $0=2uv$, i.e. $u=0$ or $v=0$, but only $u=0$ leads to steepest descent.
}
\example{Hyperbolic}{$\rho(t)=\i\cosh(t)$}{Here, $\rho'(t)=\i \sinh(t)$ has a saddle point at $t=0$.
We have $\rho(t) = \i \cosh(u)\cosh(\i v) + \i \sinh(u)\sinh(\i v) = \i \cosh(u) \cos(v) - \sinh(u)\sin(v)$.
At $t=0$, $\psi(0)=1=\cosh(u)\cos(v)$.
}
\example{Combined Linear}{$\rho(t)=\sinh(t)-t$}{Here, $\rho(t)=\i \cosh(u)\sin(v) - \i v + \sinh(u)\cos(v) - u$ and $\rho'(t)=\cosh(t)-1$ has a saddle point at $t=0$.
Note that $\rho''(0)=0$ (but $\rho'''(0)\neq 0$).
At $t=0$, $\psi(0)=0=\cosh(u)\sin(v)-v$, so $u(v) = \arcosh\kla*{\frac{v}{\sin(v)}}$ or $v=0$, and there are 6 steepest paths separated by $60^\circ$ rotations.
}
\example{Combined Quadratic}{$\rho(t)=\cosh(t) - \frac{1}{2}t^2$}{$\phi(t)=\cosh(u)\cos(v) + \frac{1}{2}\kla*{u^2-v^2}$ and $\psi(t) = \sinh(u)\sin(v) - uv$ with a saddle point at $t=0$ and $\rho^{(4)}(0)\neq 0$ being the first non-zero derivative.
At $t=0$, $\psi(0)=0=\sinh(u)\sin(v)-uv$, so $v=0$ or $u=0$ or $\frac{\sinh(u)}{u} = \frac{v}{\sin(v)}$.
This gives 8 steepest paths separated by $45^\circ$ rotations.
}
\example{Bessel Function}{$J_0(x) = \Re \klc*{\frac{1}{\i \pi} \intx{-\i\pi/2}{\i\pi/2}{\powe{\i x\cosh(t)}}{t}}$ as $x\rightarrow\infty$}{Here, $\rho(t)=\i\cosh(t) = \i\cosh(u)\cos(v) - \sinh(u)\sin(v)$ and the saddle point is at $t=0$.
This lies in the center of the original contour, and we have $\cos(v) = \frac{1}{\cosh(u)}$ as a consequence of constant $\psi$ at $t=0$.
The two horizontal pieces $\pm\i \frac{\pi}{2} \pm u$ ($u\in\kla*{0,\infty}$) leave us with a contour that connects the two points $\pm \kla*{\infty + \i \frac{\pi}{2}}$, called \emph{Sommerfeld contour}.
Among all of the possible contours, there is exactly one curve that has constant $\psi$ and passes through our saddle point at $t=0$.
By Laplace's method, the full asymptotic expansion is determined by a small neighborhood around the saddle point, so we can linearize the contour to $v=u$ ($\phi<0$ requires $uv>0$, so we need positive slope for steepest descent).
Using $t=(1+\i)r$ with $r\in\kla*{-\epsilon,\epsilon}$ and $\cosh(t)\sim 1 + \i r^2$, we obtain
\begin{align}
J_0(x) &\sim \Re \klc*{\frac{1}{\i\pi} \intx{-\epsilon}{\epsilon}{\powe{\i x \kla*{1+\i r^2}}(1+\i)}{r}} \sim \frac{1}{\pi} \Re \klc*{\powe{\i x}(1-\i)} 2\intx{0}{\infty}{\powe{-xr^2}}{r} = \sqrt{\frac{2}{\pi x}} \cos\kla*{x-\frac{\pi}{4}}.
\end{align}
Watson's Lemma can give us the full expansion.
Along the contour, we have $\rho(t)=\i - \eta(t)$ with $\eta(t)=\i-\i\cosh(t)$ and $\d\eta=-\i\sinh(t)\d t$.
Thus,
\begin{align}
J_0(x) &= 2\cdot \Re\klc*{\frac{1}{\i\pi} \intx{0}{\infty}{\powe{x\kla*{\i - \eta}} \frac{\i}{\sqrt{-\eta^2+2\i\eta}}}{\eta}}.
\end{align}
The expansion yields $\alpha = -\frac{1}{2}$, $\beta=1$ and $c_n=\frac{\Gamma\kla*{n+\frac{1}{2}}}{(2\i)^n \Gamma\kla*{\frac{1}{2}} n!}$, so
\begin{align}
J_0(x) &\sim \frac{\sqrt{2}}{\pi} \Re 
\klc*{\powe{\i \kla*{x-\pi/4}} \sum_{n\ge 0} \frac{\Gamma\kla*{n+\frac{1}{2}}^2}{(2\i x)^n n! \sqrt{x} \Gamma\kla*{\frac{1}{2}}}} = \sqrt{\frac{2}{\pi x}} \klc*{\alpha(x) \cos\kla*{x-\frac{\pi}{4}} + \beta(x) \sin\kla*{x-\frac{\pi}{4}}}
\end{align}
with $\alpha(x) = \sum_{k\ge 0} (-1)^k \frac{\Gamma\kla*{2k+\frac{1}{2}}^2}{\pi (2k)!(2x)^{2k}} $ and $\beta(x) = \sum_{k\ge 0} (-1)^k \frac{\Gamma\kla*{2k+\frac{3}{2}}^2}{\pi (2k+1)!(2x)^{2k+1}}$ as $x\rightarrow\infty$.
}
\example{Trigonometric Gauss integral}{$I(x)=\intx{0}{1}{\powe{-4xt^2} \cos\kla*{5xt-xt^3}}{t}$ as $x\rightarrow\infty$}{We can rewrite this as $I(x)=\frac{1}{2} \powe{-2x} \Re \klc*{\intx{-1}{1}{\powe{x\rho(t)}}{t}}$ with
\begin{align}
\rho(t) &= -\kla*{t-\i}^2-\i \kla*{t-\i}^3 = -\klb*{u^2-(v-1)^2 + 2\i u(v-1)} \kla*{2-v + \i u}\\
&= (v-2)\kla*{u^2-v^2+2v-1} + 2u^2(v-1) + \i \klb*{u\kla*{u^2-v^2+2v-1} - 2u(v-1)(v-2)}
\end{align}
(note: there seem to be some typos in the script here. Main point is that there are two saddle points at $t=\i$ and $t=\i \frac{5}{3}$. Dominant term from $t=\i$, and leading order just has $\rho(t)\sim -u^2$ as $u\rightarrow 0$, giving a simple Gauss with $\sqrt{\frac{\pi}{x}}$)
}
\example{Hyperbolic linear shift}{$I(x)=\frac{1}{2\pi\i}\intx{C}{ }{\powe{x\kla*{\sinh(t)-t}}}{t}$ as $x\rightarrow\infty$}{Here, $C$ is a Sommerfeld contour from $\infty-\i\pi$ to $\infty+\i\pi$ that passes through the saddle point $t=0$.
We once again only care about the local behavior, where $\psi(0)=0=\cosh(u)\sin(v)-v$ simplifies to $v=\pm \sqrt{3}u$.
Thus, $t=r\powe{\pm\i \pi/3}$ and
\begin{align}
I(x) &\sim \frac{1}{2\pi\i} \intx{0}{\epsilon}{ \powe{-x\frac{1}{6}r^3} \powe{\i\pi/3}}{r} + \frac{1}{2\pi\i} \intx{\epsilon}{0}{\powe{-x\frac{1}{6}r^3} \powe{-\i\pi/3}}{r} \sim \frac{1}{2\pi} \intx{0}{\infty}{\powe{-\frac{x}{6}r^3}}{r} \propto x^{-1/3}.
\end{align}
}

\section{Homogeneous linear differential equations}
The general form of such ODEs is
\begin{align}
0 &= y^{(n)}(x) + P_{n-1}(x)y^{(n-1)}(x) + \dots + P_1(x) y'(x) + P_0(x)y(x).
\end{align}
\note{Example:$y''+P(x)y'+Q(x)y=0$}
\subsection{Classification of singular points}
\definition{Ordinary Points}{The point $x_0\neq \infty$ is called ordinary, if the coefficient functions $P_0,\dots,P_{n-1}$ are all analytic in a neighborhood of $x_0$ in the complex plane.

\textsc{Fuchs} (1866):
\begin{enumerate}
\item all $N$ linearly independent solutions of the ODE are analytic in a neighborhood of an ordinary point
\item if any solution is expanded in a Taylor series around $x_0$, then the \gls{roc} is at least as large as the distance of $x_0$ to the nearest singularity of the coefficient functions in the complex plane
\item the location of a singularity of a solution must coincide with the location of a singularity of a coefficient function. The solution of a linear equation cannot have singularities at any other point.
\end{enumerate}
}
\example{Linear}{$\kla*{x^2+1}y'+2xy=0$}{This ODE has an ordinary point at $x=0$.
The solution $y(x)=\frac{c}{1+x^2}$ can be expanded into a series with \gls{roc} 1, which equals the distance to the coefficient singularities at $x=\pm i$.
}
\definition{Regular Singular Points}{The point $x_0\neq\infty$ is called regular singular if not all of $P_0,\dots,P_{n-1}$ are analytic while $(x-x_0)^nP_0,\dots (x-x_0)P_{n-1}$ are analytic.
\note{A solution may be analytic at a regular singular point. If it is not analytic, its singularity must either be a pole or an algebraic or logarithmic branch point.}
\textsc{Fuchs}:
\begin{enumerate}
\item there is always at least one solution of the form $y(x)=(x-x_0)^\alpha A(x)$ ($\alpha$ is the \emph{indicial exponent}) where $A(x)$ is analytic at $x_0$
\item the Taylor series of $A$ converges at least up to the nearest singularity of the coefficient functions
\end{enumerate}
\note{Example: $y'=\frac{y}{\sinh(x)}$ has solution $y(x)=c\tanh\kla*{\frac{x}{2}}$ and is analytic at $x=0$, so this is a regular singular point.
The radius of convergence is $\pi$ (poles at $\pm \i\pi$).}
}
\definition{Irregular Singular Points}{The point $x_0\neq \infty$ is called irregular singular if it is neither ordinary nor regular singular.
\note{Typically, at an irregular point, all solutions exhibit an essential singularity, often in combination with a pole or an algebraic or logarithmic branch point. But this is not always the case, some may even be analytic at $x_0$.}
}
\definition{Classification of $x_0=\infty$}{Use $x=\frac{1}{t}$ and $\frac{\d}{\d x} = -t^2\frac{\d}{\d t}$ etc. to then classify the point $t=0$.
}
\example{First order examples}{near $x=0$}{$y'=\frac{y}{2}$ has an ordinary point at $x=0$; $y(x)=c\powe{x/2}$ has a Taylor series at $x=0$.

$y'=\frac{y}{2x}$ has a regular singular point at $x=0$; $y(x)=\sqrt{x}$ has a Frobenius series with $\alpha=\frac{1}{2}$ at $x=0$.

$y'=\frac{y}{2x^2}$ has an irregular singular point at $x=0$; $y(x)=c\powe{-
1/(2x)}$ has an essential singularity at $x=0$ (and therefore neither Taylor nor Frobenius series)
}
\subsection{Local behavior near ordinary points}
If we do not have a simple closed-form solution, it is very useful to know how to obtain answers to hard problems in terms of infinite series.
Furthermore, the first few terms are usually sufficient to give a very accurate approximation (REF adv guessing I).
A perturbation series with a small parameter $\epsilon=(x-x_0)$ gives more accurate approximation as $\epsilon$ becomes smaller or $n$ increases (or both).
\example{Gauss ODE}{$y'=2xy$ with $y(0)=1$}{This has an ordinary point at $x=0$.
Let $y(x)=\sum a_n x^n$.
Then
\begin{align}
0 &= \sum_{n\ge 0} a_n \klc*{nx^{n-1} - 2x^{n+1}} = \sum_{n\ge 0} x^n \klc*{a_{n+1}(n+1) - 2a_{n-1}}.
\end{align}
This yields $\frac{a_{n+2}}{a_n} = \frac{2}{n+2}$.
Because $a_{-1}=0$, all odd terms vanish, and the even terms give $a_n=\frac{2}{n} \frac{2}{n-2} \dots 2a_0$.
We have $a_0=0$ and thus $y(x)=\sum_{n\ge 0} \frac{1}{n!}x^{2n} = \powe{x^2}$.
}
\definition{Gamma Function}{The Gamma function admits the product representation
\begin{align}
\Gamma(z) &= \frac{1}{z} \prod_{n\ge 1} \kla*{1 + \frac{1}{n}}^z \frac{1}{1+\frac{z}{n}}.
\end{align}
Using the fundamental property $\Gamma(z+1)=z\Gamma(z)$, the Pochhammer function $(z)_n = \frac{\Gamma(z+n)}{\Gamma(z)}$ can be written as
\begin{align}
(z)_n &= (z+n-1)(z+n-2) \dots z = \prod_{l=0}^{n-1} (z+l).
\end{align}
This is very helpful for analyzing repeated products of the form
\begin{align}
(kn-k+j)(kn-2k+j)\dots &= \prod_{l=0}^{n-1} (j+kl) = k^n \kla*{\frac{j}{k}}_n
\end{align}
}
\example{Airy equation}{$y''=xy$ near $x=0$}{This is an ordinary point.
Series expansion gives
\begin{align}
0 &= \sum x^n \klc*{a_{n+2}(n+1)(n+2)-a_{n-1}},
\end{align}
i.e. $a_n = \frac{1}{n(n-1)}a_{n-3}$ with $a_{-1},a_{-2},\dots=0$ giving $a_{3n+2}=0$.
