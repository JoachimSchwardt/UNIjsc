\section{Stieltjes Integral}
Consider a Stieltjes integral $y(x) = \intx{0}{\infty}{\frac{\powe{-t}}{1+xt}}{t}$ for $x>0$.
We have
\begin{align}
y' &= \intx{0}{\infty}{\frac{-t\powe{-t}}{\kla*{1+xt}^2}}{t} = -\frac{1}{x}\intx{0}{\infty}{\frac{1+xt-1}{\kla*{1+xt}^2}\powe{-t}}{t} = -\frac{y}{x} + \frac{1}{x}\intx{0}{\infty}{\frac{\powe{-t}}{\kla*{1+xt}^2}}{t},\\
&= -\frac{y}{x} + \frac{-1}{x^2} \frac{\powe{-t}}{1+xt} \bigg\vert_0^\infty - \frac{1}{x^2} \intx{0}{\infty}{\frac{\powe{-t}}{1+xt}}{t} = -\frac{y}{x} - \frac{1+y}{x^2}\\
y'' &= -\frac{y'}{x} + \frac{y}{x^2} - \frac{y'}{x^2} + 2\frac{1+y}{x^3} = \frac{y}{x^2} + 2\frac{1+y}{x^3} + \kla*{\frac{1}{x} + \frac{1}{x^2}} \kla*{\frac{y}{x} + \frac{1+y}{x^2}}\\
&= \frac{2y}{x^2} + \frac{3}{x^3} + \frac{1}{x^4} + \frac{4y}{x^3} + \frac{y}{x^4}.
\end{align}
Thus, $x^2y'' + (1+3x)y' = 2y+\frac{3}{x} + \frac{1}{x^2} + \frac{4}{x}y + \frac{y}{x^2} - \frac{y}{x} - 3y - \frac{1}{x^2} - \frac{3}{x} - \frac{y}{x^2} - \frac{3y}{x} = -y$ is an ODE satisfied by $y$.

Let $y(x) \sim \sum_{n\ge 0} n! (-x)^n$ as $x\rightarrow 0^+$.
Assume that $y(x) = x^\alpha \sum_{n\ge 0} a_n x^n$ with $a_0=1$.
To simplify the calculation, let us first define $y(x) = x^\alpha q(x)$.
This yields
\begin{align}
y' &= \alpha x^{\alpha-1} q + x^\alpha q',\\
y'' &= \alpha (\alpha-1) x^{\alpha-2} q + 2\alpha x^{\alpha-1} q' + x^\alpha q'',
\end{align}
and inserting into the ODE leads to
\begin{align}
0 &= y + (1+3x)y'+x^2y''\\
&= x^\alpha q + (1+3x) \alpha x^{\alpha-1} q + (1+3x) x^\alpha q' + \alpha (\alpha-1)x^\alpha q + 2\alpha x^{\alpha+1} q' + x^{\alpha+2} q''\\
&= \klb*{\kla*{1+\alpha}^2 x^\alpha + \alpha x^{\alpha-1}}q + \klb*{x^\alpha + (3+2\alpha) x^{\alpha+1}}q' + x^{\alpha+2} q'' \\
&= \sum_{n\ge 0} a_n \klc*{\klb*{\kla*{1+\alpha}^2 x^\alpha + \alpha x^{\alpha-1}} x^n + \klb*{x^\alpha + (3+2\alpha) x^{\alpha+1}}nx^{n-1} + x^{\alpha+2} n(n-1)x^{n-2}}\\
&= \sum_{n\ge 0} \klc*{ a_n\klb*{\kla*{1+\alpha}^2 + (3+2\alpha)n + n(n-1)} + a_{n+1} \klb*{\alpha + (n+1)} }x^{n+\alpha}.
\end{align}
Simplifying gives $\frac{a_{n+1}}{a_n} = -\frac{(1+\alpha)^2 + (2 + 2\alpha  + n)n}{\alpha + n + 1}$.
Also note that the coefficient of $x^{\alpha-1}$ is $a_0\alpha$, so $\alpha=0$.
This reduces the recursion to $\frac{a_{n+1}}{a_n} = - \frac{1 + 2n + n^2}{n+1} = -(n+1)$.
Resolving it gives $a_{n+1} = -(n+1)a_n = (-1)^2 (n+1)na_{n-1} = \dots = (-1)^{n+1} (n+1)! a_0$.
\definition{Radius of Convergence (RoC)}{
The radius of convergence of a series can be computed as
\begin{align}
r &= \frac{1}{\limsup_{n\rightarrow \infty} \sqrt[n]{|a_n|}}
\end{align}
or
\begin{align}
r &= \limsup_{n\rightarrow \infty} \abs*{\frac{a_n}{a_{n+1}}}.
\end{align}
}
The radius of convergence is zero, because $\abs*{\frac{a_n}{a_{n+1}}} = \frac{n!}{(n+1)!} = \frac{1}{n+1}\rightarrow 0$ as $n\rightarrow\infty$.
The root formula also works, because $\sqrt[n]{n!} \sim \klb*{n^n \powe{-n}}^{1/n} = \frac{n}{\e} \rightarrow\infty$.

Naive Mathematics: Using the geometric series, we can rewrite the integral as
\begin{align}
y(x) &= \intx{0}{\infty}{ \powe{-t} \sum_{n\ge 0} (-1)^n x^n t^n}{t} \koml{\text{RoC violated!}}{=} \sum_{n\ge 0} (-1)^n x^n \intx{0}{\infty}{\powe{-t} t^n}{t} = \sum_{n\ge 0} (-1)^n n! x^n.
\end{align}
The resulting series matches our asymptotic analysis as $x\rightarrow 0^+$ from above.
This makes sense, insofar as replacing the upper limit by $\frac{\epsilon}{x}$ with $\epsilon < 1$ then allows us to interchange summation and integration without violating the RoC.
As $x$ gets small, the limit tends to infinity, so intuitively our series expansion should get better and better for fixed $n$.

\section{Asymptotic Expansion of Integrals}
Find the asymptotic expansion of a given integral up to at least second order (or all orders, if possible).
\definition{Watson's Lemma}{Consider $I(x) = \intx{0}{b}{f(t) \powe{-xt}}{t}$ for $b>0$ and $x\rightarrow \infty$.
If the function satisfies
\begin{itemize}
\item $f$ is continuous
\item $f(t) \sim t^\alpha \sum_{n\ge 0} a_n t^{\beta n}$ as $t\rightarrow 0^+$ with $\alpha > -1$ and $\beta > 0$,
\end{itemize}
\note{If $b=\infty$ we also need that $f(t)\ll \powe{ct}$ for some $c>0$ to ensure convergence}
then the \emph{full asymptotic expansion} of any integral of this type is given by
\begin{align}
I(x) &\koml{x\rightarrow\infty}{\sim} \sum_{n\ge 0} \frac{a_n \Gamma\kla*{\alpha + \beta n+ 1}}{x^{\alpha + \beta n + 1}}.
\end{align}
To show this, choose $\epsilon$ such that
\begin{align}
\abs*{f(t) - t^\alpha \sum_{n=0}^N a_n t^{\beta n}} &\le K t^{\alpha + \beta (N+1)} \quad \text{for} \quad 0\le t \le \epsilon.
\end{align}
Then the error introduced by replacing $f$ by this finite series is bounded by
\begin{align}
K\intx{0}{\epsilon}{t^{\alpha + \beta (N+1)} \powe{-xt}}{t} \le K \frac{\Gamma\kla*{\alpha + \beta(N+1)+1}}{x^{\alpha + \beta(N+1)+1}},
\end{align}
and $\abs*{I(x) - \sum_{n=0}^N a_n \frac{\Gamma\kla*{\alpha + \beta n + 1}}{x^{\alpha + \beta n + 1}}} \ll \frac{1}{x^{\alpha + \beta N + 1}}$ for all $N$.
}
\definition{Direct asymptotic behavior of $I(x)$}{Replace $I(x)$ by $I(x,\epsilon)$ as usual and replace $\phi(t)$ by the first few terms in its Taylor series.
\begin{enumerate}
\item if $c$ is an endpoint and $\phi'(c)\neq 0$, then $\phi(t) \sim \phi(c) + (t-c)\phi'(c)$
\begin{itemize}
\item if $c=a$, then $\phi'(a) < 0$ and $I(x,\epsilon) \sim \intx{a}{a+\epsilon}{f(a) \powe{x\phi(a)} \powe{x(t-a) \phi'(a)}}{t} \sim f(a) \powe{x\phi(a)} \intx{a}{\infty}{\powe{x(t-a)\phi'(a)}}{t}$ and $I(x) \sim -\frac{f(a)}{x\phi'(a)} \powe{x\phi(a)}$
\item if $c=b$, then $\phi'(b)>0$ and $I(x) \sim \frac{f(b)}{x\phi'(b)} \powe{x\phi(b)}$
\end{itemize}
\item if $a<c<b$ and $\phi'(c)=0$ but $\phi''(c)\neq 0$, then $\phi(t) \sim \phi(c) + \frac{1}{2} (t-c)^2 \phi''(c)$
\begin{itemize}
\item we must have $\phi''(c) < 0$ for convergence
\item it follows that \begin{align}
I(x,\epsilon) &\sim \intx{c-\epsilon}{c+\epsilon}{f(c) \powe{x\phi(c)} \powe{\frac{1}{2}(t-c)^2 \phi''(c)}}{t} \sim f(c) \powe{x\phi(c)} \sqrt{\frac{2\pi}{x|\phi''(c)|}}
\end{align}
\note{If this happens at $t=a$ or $t=b$ (i.e. $c$ is at the boundary), then only half of the peak contributes and we need a factor of $\frac{1}{2}$ to correct this}
\end{itemize}
\item in general, if $\phi^{(p)}(c) \neq 0$ is the first non-vanishing term, then $\phi(t) \sim \phi(c) + \frac{1}{p!} (t-c)^p \phi^{(p)}(c)$
\begin{itemize}
\item if $a<c<b$, then $p$ must be even and $\phi^{(p)}(c) < 0$
\item it follows that \begin{align}
I(x,\epsilon) &\sim f(c) \powe{x\phi(c)} \intr{\powe{x (t-c)^p \frac{1}{p!} \phi^{(p)}(c)}}{t},
\end{align}
which gives $I(x) \sim f(c) \powe{x\phi(c)} \frac{2}{p} \Gamma\kla*{\frac{1}{p}} \kla*{\frac{p!}{x\abs*{\phi^{(p)}(c)}}}^{1/p}$ as $x\rightarrow\infty$
\end{itemize}
\end{enumerate}
}

\subsection{Cubic phase oscillation}
Consider $I(x) = \intx{0}{1}{\powe{\i xt^3}}{t}$ as $x\rightarrow\infty$.
We will use the residue theorem with a contour $\Gamma$ consisting of three components
\begin{itemize}
\item $z(s)=s$ with $s\in\kla*{0,1}$
\item $z(s)=\powe{\i s}$ with $s\in\kla*{0,\frac{\pi}{6}}$
\item $z(s)=\powe{\i \frac{\pi}{6}}s$ with $s\in\kla*{0,1}$.
\end{itemize}
Since the function does not have poles within $\Gamma$, we have $\oint_\Gamma f(z)\,\d z=0$.
The first part is just the original integral while the third gives
\begin{align}
I_3 &= \intx{1}{0}{ \powe{\i x \kla*{\powe{\i\frac{\pi}{6}}s}^3} \powe{\i\frac{\pi}{6}} }{s} = -\powe{\i\frac{\pi}{6}}\intx{0}{1}{\powe{-xs^3}}{s} = -\powe{\i\frac{\pi}{6}} \intx{0}{x}{\powe{-t} \frac{1}{3} x^{-1/3} t^{-2/3} }{t}\\
&= -\powe{\i\frac{\pi}{6}} \frac{1}{3} x^{-1/3} \sum_{n\ge 0} \frac{(-1)^n}{n!} \intx{0}{x}{ t^{n-2/3} }{t} = -\powe{\i\frac{\pi}{6}} \frac{1}{3} x^{-1/3} \sum_{n\ge 0} \frac{(-1)^n}{n!} \frac{x^{n+1/3}}{n+\frac{1}{3}}.
\end{align}
The integral over the arc gives $I_2 = \intx{0}{\frac{\pi}{6}}{\powe{\i x\kla*{\powe{\i s}}^3} \i\powe{\i s}}{s}$.
Its magnitude can be bounded by
\begin{align}
|I_2| &\le \intx{0}{\frac{\pi}{6}}{ \powe{-x\sin(3s)} }{s} \le \intx{0}{\frac{\pi}{6}}{\powe{-x \frac{6s}{\pi}}}{s} = \frac{\pi}{6x} \kla*{\powe{-x} - 1}.
\end{align}
Unfortunately this only gives $I_1 = -I_3 + \mathcal{O}\kla*{\frac{1}{x}}$, which is not the full expansion...
