\chapter{Lie-Groups and Representations}
\section{Definition of a Group}
A group $(G,\circ)$ is a set $G=\klc*{g_i}$ with an operation $\circ$ that admits the following rules:
\begin{enumerate}
\item closure: $g_i\circ g_j=g_k\in G$ for all $i,j$ (i.e. the group operation always remains in the group)
\item associative: $g_i\circ \kla*{g_j\circ g_k} = \kla*{g_i\circ g_j}\circ g_k$
\item identity: $e\circ g_i=g_i=g_i\circ e$ for all $i$
\item (unique) inverse: $g_i\circ g_i^{-1}=e=g_i^{-1}\circ g_i$ for all $i$; the inverse element $g_i^{-1}\in G$ is unique
\end{enumerate}
If $g_i\circ g_j=g_j\circ g_i$ for all $i,j$, i.e. the group elements commute, the group is called Abelian.
\example{Some Groups}{}{
$\kla*{\klc*{1},\cdot}$, $\kla*{\klc*{\pm 1}, \cdot}$, $\kla*{\klc*{0,1},\text{XOR}}$, $\kla*{\klc*{\sigma_0,\sigma_j},\cdot}$.

Cyclic group $Z_n=\klc*{1,\omega_n,\omega_n^2,\dots,\omega_n^{n-1}}$ with $\omega_n = \powe{\i 2\pi/n}$ (the multiplication operation is implied and often omitted).
}
\section{Representations}
A representation is a mapping of group elements $g\mapsto \mathcal{R}(g)$ such that $g_1\circ g_2 = g_3$ implies $\mathcal{R}(g_1) \cdot \mathcal{R}(g_2) = \mathcal{R}(g_3)$, where $\cdot$ is a potentially distinct operation adequate on the space of the representation (e.g. matrix multiplication).
There are special representations:
\begin{enumerate}
\item \emph{fundamental} repr.: all other representations can be generated from this repr. by direct products (e.g. angular momentum)
\item \emph{adjoint} repr.: constructed from the \emph{structure constants} of the group, $\klb*{T^a,T^b} = \i f_{abc} T^c$ (Lie-Algebra; summation over repeated indexes is implied). Explicitly, $\kla*{\tilde{T}^a}_{bc} = -\i f_{abc}$ are the components
\item \emph{equivalent} repr.: $T^i$ and $\tilde{T}^i$ are equivalent if they are related by a single similarity transform, i.e. $\tilde{T}^i = UT^i U^{-1}$ for all $i$
\end{enumerate}
\section{Lie-Groups}
In \emph{continuous} groups, elements are labeled by $r$ real parameters $\bm{\alpha}=(\alpha_1,\dots,\alpha_r)$ such that $g(\bm{0})=e$ and $g(\bm{\alpha}) \in G$.
This implies
\begin{align}
g\kla*{\bm{\alpha}} g\kla*{\bm{\beta}} &= g\kla*{\bm{\gamma}\kla*{\bm{\alpha},\bm{\alpha}}}
\end{align}
with some analytic function $\bm{\gamma}$.

Consider for example a coordinate transformation (or Lorentz transform in case of an internal parameter space).
Then we have $x_i' = f_i\kla*{x;\bm{\alpha}}$ with $x=\kla*{x_1,\dots,x_N}$ and $f_i\kla*{x;\bm{0}} = x_i$, i.e. an infinitesimal transform is a small deviation from the identity.
Explicitly, writing $x_i'=x_i+\d x_i$ gives
\begin{align}
\d x_i &= \sum_{\rho=1}^r \frac{\partial f_i}{\partial \alpha_\rho} \d \alpha_\rho = \bm{\alpha} \cdot \nabla_{\bm{\alpha}} f_i.
\end{align}
A function $F(x)$ then transforms to $F'=F+\d F$ with $\d F=\sum_{i=1}^N \frac{\partial F}{\partial x_i} \d x_i$.
We can express this as
\begin{align}
F' &= \kla*{1 - \i \d\bm{\alpha} \cdot \bm{T}} F,
\end{align}
where $T_\rho = \i\sum_{i=1}^N \frac{\partial f_i}{\partial\alpha_\rho} \frac{\partial F}{\partial x_i}$ are the \emph{generators} of the infinitesimal transformation.

By expanding around the identity, the group elements in a Lie-Group can be written as
\begin{align}
g\kla*{\bm{\alpha}} &= 1 - \i \alpha_a T^a + \mathcal{O}\kla*{\alpha^2} = \powe{-\i\alpha_a T^a} = \lim_{k\rightarrow\infty} \kla*{1 - \i \frac{\alpha_a T^a}{k}}^k.
\end{align}
Considering the commutation of two transformations, we also find the relation
\begin{align}
g(\theta)g(\phi)g(\theta)^{-1} g(\phi)^{-1} &= 1 + g(\theta)\klb*{g(\phi), g(\theta)^{-1}} g(\phi)^{-1} = 1 + g(\theta) \i\theta_a (-\i)\phi_b \klb*{T^a,T^b} g(\phi)^{-1}\\
&= 1 + \theta_a \phi_b\klb*{T^a,T^b} + \mathcal{O}\kla*{\theta^2,\phi^2} \overset{!}{=} g(\xi) = 1 - \i \xi_c T^c + \mathcal{O}\kla*{\xi^2}.
\end{align}
From this we may read off $\xi_c = C_c^{ab}\theta_a\phi_b$ with $\i C_c^{ab} T^c = \klb*{T^a,T^b}$.
Clearly $C_c^{ab}=-C_c^{ba}$, and in general the coefficients define the group multiplication rules.
\section{Compact Lie-Groups}
If the volume of the parametrization is finite, i.e. the set $A=\klc*{\bm{\alpha}}$ has finite measure, a Lie-Group is compact (e.g. $\powe{\i\theta \tau}$ with $\theta\in [0,2\pi)$ and some generator $\tau$).
\definition{Representation Theorem}{
Any repr. of a compact Lie-Group is equivalent to a repr. by unitary operators, and any repr. satisfies the same fundamental algebra, i.e. $\klb*{T^a_\mathcal{R}, T^b_\mathcal{R}} = \i f_{abc} T^c_\mathcal{R}$.

Explicitly, this means that all group elements are of the form $U=\powe{\i \theta_a T^a}$.
}
The linear combination $\theta_a T^a$ spans a vector-space with (finite) basis $\klc*{T^a}$ and $\klb*{T^a,T^b} = \i f_{abc} T^c$.
\subsection{Properties}
Normalization: we have $\tr\kla*{\klb*{T^a,T^b}}=0$ for all $a,b$ due to the cyclicity of the trace.
Using the Lie-Algebra, this gives $f_{abc} \tr\kla*{T^c}=0$, and because this holds for all combinations, we can conclude $\tr\kla*{T^c}=0$ unless $f_{abc}\equiv 0$ (e.g. for an Abelian group).
\note{One can show that $\tr\kla*{T^aT^b}=\lambda \delta_{ab}$ for some fixed constant $\lambda$, and by convention generators are normalized such that $\lambda=\frac{1}{2}$.
}
\subsection{Examples}
Some compact simple Lie-Groups are $SU(N)$ or $SO(N)$.
Simple means that no subgroup commutes with the rest, i.e. for every invertible subgroup $H\subset G$ one has $g_jh_ig_j^{-1}\in H$ for all $h_i\in H$ and $g_j\in G$.
\note{There are two trivial invertible subgroups: the group itself and the trivial group $\klc*{e}$.}
Semi-simple LGs are direct products of simple groups, e.g. $SU(2)\times SU(3)$ (non semi-simple may be $SU(2)\times SU(2) \times SU(3)$).

Non-compact LGs are e.g. the Lorentz group or the Poincare group.
\subsection{Special Unitary Group}
The groups $SU(N)$ with elements $U\in\mathbb{C}^{N\times N}$ and $U^\dagger U=UU^\dagger = \mathbbm{1}$ and $\det(U)=+1$ have $r=N^2-1$ generators.
This is derived from $2N^2$ total real parameters, minus $N(N-1)$ from the off-diagonal symmetry and minus $N$ from the real-valued diagonal.
A final $-1$ comes from the special property, i.e. $\det(U)=+1$ rather than $\det(U)=\pm 1$.

One may write $U=\powe{\i \theta_a T^a}$ with $a=1,\dots,r$ and the expansion of $U^\dagger U = \mathbbm{1} + \i\theta_a \kla*{T^a - T^{a\dagger}} + \dots$ gives $T^a = \kla*{T^a}^\dagger$, i.e. all generators are Hermitian.
Similarly, using 
\begin{align}
\det \exp A &= \exp \tr A,
\end{align}
one finds that $\det(U)=1$ implies $\tr\kla*{T^a}=0$.
\example{$SU(2)$}{\textsc{Pauli} matrices}{
The fundamental repr. is $T^a = \frac{1}{2}\sigma^a$ with $f_{abc}=\epsilon_{abc}$ and $\sigma^1 = \begin{psmallmatrix} 0 & 1 \\ 1 & 0 \end{psmallmatrix}$, $\sigma^2 = \begin{psmallmatrix} 0 & -\i \\ \i & 0 \end{psmallmatrix}$ and $\sigma^3 = \begin{psmallmatrix} 1 & 0 \\ 0 & -1 \end{psmallmatrix}$.
}
\example{$SU(3)$}{\textsc{Gellmann} matrices}{
The fundamental repr. is $T^a = \frac{1}{2}\lambda^a$ ($3\times 3$-matrices) with $f_{abc}=???$ and $\lambda^i = \begin{psmallmatrix} \sigma^i & 0 \\ 0 & 0 \end{psmallmatrix}$ for $i=1,2,3$ and $\lambda^4=\begin{psmallmatrix} 0 & 0 & 1 \\ 0 & 0 & 0 \\ 1 & 0 & 0 \end{psmallmatrix}$, $\lambda^5=\begin{psmallmatrix} 0 & 0 & -\i \\ 0 & 0 & 0 \\ \i & 0 & 0 \end{psmallmatrix}$, $\lambda^6 = \begin{psmallmatrix} 0 & 0 \\ 0 & \sigma^1 \end{psmallmatrix}$, $\lambda^7 = \begin{psmallmatrix} 0 & 0 \\ 0 & \sigma^2 \end{psmallmatrix}$ and $\lambda^8=\begin{psmallmatrix} 1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & -2 \end{psmallmatrix}$.

One can show that $\klc*{\lambda^a,\lambda^b} = \frac{4}{3} \delta_{ab} + 2d_{abc}\lambda^c$ with the totally symmetric $d_{abc}=\frac{1}{4} \tr \kla*{\lambda^a\klc*{\lambda^b,\lambda^c}}$ complementing the totally anti-symmetric $f_{abc} = \frac{1}{4} \tr
\kla*{\lambda^a \klb*{\lambda^b,\lambda^c}}$.
Combined, this gives
\begin{align}
\lambda^a\lambda^b &= \frac{2}{3} \delta_{ab} + \kla*{d_{abc} + \i f_{abc}}\lambda^c.
\end{align}
}
\example{Relation between $SU(2)$ and $SO(3)$}{Rotations}{
Both groups have 3 generators, and we may relate them by defining $X=\bm{x}\cdot \bm{\sigma}$ with $\bm{x} \in \mathbb{R}^3$ and $X\in\mathbb{C}^{2\times 2}$.
Explicitly, $X = \begin{psmallmatrix} z & x - \i y \\ x + \i y & -z \end{psmallmatrix}$ and $\det(X) = -\abs*{\bm{x}}^2$.

\chapter{Local Symmetry}
In the transformation $\psi'=\powe{-\i \theta_a T^a}\psi$, we can distinguish global ($\theta=\const$) and local transformations ($\theta_a(x)$ depends on space-time coordinates).
\section{Classical Electrodynamics}
Many scalar and vector potentials $\phi,\bm{A}$ lead to the same physical electromagnetic fields $\bm{E},\bm{B}$, so we may specify a specific gauge (e.g. Lorentz gauge, $\partial_\mu A^\mu=0$).
\begin{itemize}
\item electro-statics: $\partial_t \bm{E} = \partial_t \bm{B} = 0$ gives $\bm{E}=-\nabla\phi$, so $\phi \rightarrow \phi + \const$ changes nothing
\item magneto-statics: $\nabla\cdot \bm{B} = 0$ implies $\bm{B}=\nabla\times \bm{A}$ and thus $\bm{A}\rightarrow \bm{A} + \nabla \Lambda$ again changes nothing
\item electro-dynamics: $\nabla\times \bm{E} + \partial_t \bm{B} = 0$ gives $\bm{E}=-\nabla\phi - \partial_t\bm{A}$ (same transformations leave fields unchanged)
\end{itemize}
The field-strength tensor is defined as $F^{\mu\nu} = \partial^\mu A^\nu - \partial^\nu A^\mu$ with $A^\mu =(\phi,\bm{A})$ and the gauge transformation $A^\mu(x)\rightarrow A^\mu(x)+\partial^\mu \Lambda(x)$.
The Lagrangian is $\mathcal{L} = -\frac{1}{4} F^{\mu\nu} F_{\mu\nu} - J^\mu A_\mu$, such that the action $\mathcal{S} = \int \mathcal{L}\,\d^4x$ is gauge invariant.
The equations of motion are $\partial_\mu F^{\mu\nu}=J^\nu$ (\textsc{Maxwell}) and $\partial_\mu J^\mu = 0$ (continuity).
\section{Quantum Electrodynamics}
Introduce the \textsc{Dirac}-equation $\kla*{\i \slashed{\partial} - m}\psi=0$ with the  \textsc{Dirac}-matrices $\gamma^\mu$ with $\klc*{\gamma^\mu,\gamma^\nu } = 2g^{\mu\nu}$, $\slashed{A} = \gamma^\mu A_\mu$ and $\kla*{\gamma^\mu }^\dagger = \gamma^0 \gamma^\mu \gamma^0 = (\gamma^0, -\bm{\gamma}$).
The usual representation is $\gamma^\mu = \begin{psmallmatrix} 0 & \sigma_\mu \\ \sigma_\mu & 0 \end{psmallmatrix}$ for $\mu=0,1,2,3$.
One also defines $\gamma^5=\i\gamma^0\gamma^1\gamma^2\gamma^3$ with $\klb*{\gamma^5,\gamma^\mu }=0$, $\kla*{\gamma^5}^\dagger = \gamma^5$ and $\klc*{\gamma^5, S_{\mu\nu}}=0$, where
\begin{align}
S_{\mu\nu} &= \frac{\i}{4} \klb*{\gamma_\mu,\gamma_\nu }.
\end{align}
For an infinitesimal Lorentz transformation, we have $x'^\mu = \Lambda^\mu_{\ \nu} x^\nu = \kla*{\delta^\mu_{\ \nu} + \omega^\mu_{\ \nu}}x^\nu$, giving $\psi'=S(\Lambda)\psi$ with $S(\Lambda) = 1 - \frac{\i}{2} \omega^{\mu\nu} S_{\mu\nu}$.

The Dirac Lagrangian is $\mathcal{L} = \bar{\psi} \kla*{\i\slashed{\partial} - m}\psi$ with $\bar{\psi} = \psi^\dagger \gamma^0$, and the Dirac equation follows from the Euler-Lagrange equations $0=\frac{\partial\mathcal{L}}{\partial\bar{\psi}}$.
A local phase transform $\psi'=\powe{-\i g\theta(x)}\psi$ changes the Lagrangian, and one therefore introduces the \emph{covariant derivative}
\begin{align}
D^\mu &= \partial^\mu + \i g A^\mu(x)
\end{align}
to get the (invariant) modified Lagrangian $\mathcal{L} = \bar{\psi} \kla*{\i\slashed{D} - m}\psi$ with $A'^\mu=A^\mu + \partial^\mu \theta(x)$.
\note{$D'_\mu\psi' = \powe{-\i g\theta} \partial_\mu \psi - \i g\psi' \partial_\mu \theta + \i g A'_\mu \psi' = \powe{-\i \theta} D_\mu\psi$, so $D'_\mu U\psi = UD_\mu \psi$}
The full Lagrangian in QED is
\begin{align}
\mathcal{L} &= \bar{\psi} \kla*{\i\slashed{D} - m}\psi - \frac{1}{4} F^{\mu\nu}F_{\mu\nu}.
\end{align}
The gauge term describes the dynamics of the gauge fields $A^\mu$, the Dirac term describes the dynamics of $\psi$, and $\slashed{D}$ describes the interaction between both.
\section{Yang-Mills Theory}
Non-Abelian gauge theory for some $N$-component \emph{matter field} $\Phi=(\phi_1,\dots,\phi_N)$.
The transformation $\Phi'=U\kla*{\theta(x)} \Phi$ has $U=\powe{-\i \theta(x)}$ and $\theta(x)=\theta_a(x)T^a$ with $a=1,\dots,r$ generators, and each $T^a$ is an $N\times N$-matrix ($N$-dimensional representation of some simple compact LG).
Any field combinations that do not contain derivatives, e.g. mass terms $\Phi^\dagger \Phi$ or \textsc{Yukawa} interaction terms $y_{ijk} \phi_i\phi_j\phi_k$ are invariant under this transformation.
Kinetic terms are however not invariant and require the introduction of gauge fields.

We define $D_\mu=\partial_\mu + \i gA_\mu$ with $A^\mu = A^\mu_a T^a$, where $A^\mu_a$ are real-valued vector gauge fields and $g$ is an arbitrary coupling constant.
Expressed in terms of the components, we have
\begin{align}
\kla*{D_\mu \Phi}_i &= \kla*{\partial_\mu \delta_{ij} + \i g A_\mu^a \kla*{T^a}_{ij} }\Phi_j.
\end{align}
The covariant derivatives specify the gauge boson interactions with the matter fields, and to determine the corresponding transformation of the gauge fields we demand $\kla*{D_\mu \Phi}' = UD_\mu\Phi$ (same as in QED).
\note{Explicit calculation:
\begin{align}
\kla*{D^\mu \Phi}' &= \kla*{\partial^\mu + \i g A'^\mu } U\Phi = \kla*{\partial^\mu U}\Phi + U\partial^\mu \Phi + \i g UU^\dagger A'^\mu U\Phi \overset{!}{=} U \kla*{\partial^\mu + \i g A^\mu }\Phi,
\end{align}
so $A^\mu = U^\dagger A'^\mu U -\i \frac{1}{g} U^\dagger \kla*{\partial^\mu U}$.
}
One finds
\begin{align}
A'^\mu &= U A^\mu U^\dagger + \i \frac{1}{g} \kla*{\partial^\mu U} U^\dagger.
\end{align}
For infinitesimal $\theta$, we can simplify this relation by considering the individual components.
Using $U=1 - \i \theta$, we find
\begin{align}
A'^\mu &= A^\mu - \i \theta A^\mu + \i A^\mu \theta + \frac{1}{g} \partial^\mu \theta = A^\mu_a \kla*{T^a - \i \theta_b \klb*{T^b,T^a}} + \frac{1}{g}\partial^\mu \theta_a T^a,
\end{align}
so $A'^\mu_a = A^\mu_a + \theta_b A^\mu_c f_{bca} T^a + \frac{1}{g} \partial^\mu \theta_a$ (or with $f_{abc}$).
The general field-strength tensors
\begin{align}
F^{\mu\nu} &= \frac{1}{\i g} \klb*{D^\mu,D^\nu} = \partial^\mu A^\nu - \partial^\nu A^\mu +\i g \klb*{A^\mu,A^\nu }
\end{align}
is then gauge covariant, i.e. $F'^{\mu\nu} = UF^{\mu\nu}U^\dagger$.
The gauge Lagrangian $\mathcal{L} = -\frac{1}{4} F^{\mu\nu}_a F^a_{\mu\nu} = -\frac{1}{2} \tr\kla*{F^{\mu\nu} F_{\mu\nu}}$ is then the direct generalization of the one in QED (note that $F^{\mu\nu} = F^{\mu\nu}_a T^a$).
\subsection{Parallel Transport}
An equivalent derivation of the covariant derivative comes from the idea of parallel transport: the phase at $x$ is \enquote{parallel} to the phase at $x+\d x$ if the phases differ by $gA_\mu(x)\d x^\mu$.
Defining the Parallel Transport of a spinor as $\hat{\psi}(x;\d x)$, we then want that the act of transforming $\psi\rightarrow\psi'=U\psi$ commutes with the PT, i.e.
\begin{align}
\hat{\psi}'(x;\d x) &= U(x+\d x) \hat{\psi}(x; \d x)\overset{!}{=} \powe{\i g A_\mu \d x^\mu} U(x)\psi.
\end{align}
Expanding to leading order in $\d x$ gives
\begin{align}
\kla*{U + \partial_\mu U \d x^\mu }\kla*{1 + \i g A_\mu \d x^\mu } &= U + \partial_\mu U \d x^\mu + \i g U A_\mu \d x^\mu \overset{!}{=}  \kla*{1 + \i g A'_\mu \d x^\mu} U, 
\end{align}
so $A'_\mu = U A_\mu U^\dagger - \i \frac{1}{g} \kla*{\partial_\mu U}U^\dagger$.
\chapter{Quantum Field Theory}
Quantum Mechanics + Special Relativity + Field Theory and the rules for creating field excitations (as well as describe their dynamics).
\section{Field description in QFT}
A field $\phi(x)$ has a conjugate momentum $\Pi(x)$ with $\klb*{\phi(\bm{x},t),\Pi(\bm{y},t)} = \i\delta(\bm{x}-\bm{y})$.
\subsection{Complex Scalar field}
A real scalar field is
\begin{align}
\phi_i(x) &= \int \kla*{a_i(\bm{k}) \powe{-\i kx} + a_i^\dagger(\bm{k}) \powe{\i kx}} \,\tilde{\d} k
\end{align}
with $\tilde{\d}k = \frac{\d^3k}{(2\pi)^3 2\omega_k}$ and $\omega_k^2 = \bm{k}^2 + M^2$ with some particle mass $M$ and $i=1,2$.
A possible Lagrangian is $\mathcal{L} = \sum_i \frac{1}{2} \partial_\mu\phi_i \partial^\mu\phi_i - \frac{1}{2} M^2\phi_i$.
\note{$\klb*{a_i(\bm{k}),a_j^\dagger(\bm{k}')} = \delta_{ij} (2\pi)^3 \delta(\bm{k}-\bm{k}')$}
We can construct a complex scalar field $\phi=\frac{1}{\sqrt{2}} \kla*{\phi_1 - \i \phi_2}$ with $\phi^\dagger = \kla*{\phi}^\dagger$ and
\begin{align}
\phi(x) &= \int \kla*{a(\bm{k}) \powe{-\i kx} + b^\dagger(\bm{k}) \powe{\i kx}} \,\tilde{\d} k,
\end{align}
where $a=\frac{a_1-\i a_2}{\sqrt{2}}$ and $b^\dagger=\frac{a_1^\dagger - \i a_2^\dagger}{\sqrt{2}}$.
A possible Lagrangian is $\mathcal{L} = \partial_\mu\phi^\dagger \partial^\mu\phi - M^2\phi^\dagger \phi$ and the Hamiltonian is $H=\int \kla*{a(\bm{k})^\dagger a(\bm{k}) + b(\bm{k})^\dagger b(\bm{k})}\,\frac{d^3k}{(2\pi)^3}$.
\note{$\klb*{a(\bm{k}),a^\dagger(\bm{k}')} = (2\pi)^3 \delta(\bm{k}-\bm{k}') = \klb*{b(\bm{k}),b^\dagger(\bm{k}')}$}
The phase transform $\phi'=\powe{-\i\alpha}\phi$ leads to the \textsc{Noether} current $J^\mu=\frac{\partial\mathcal{L}}{\partial \partial_\mu\phi} \delta \phi$, i.e. $N_\phi^\mu = \i \kla*{\phi^\dagger \partial^\mu \phi - \phi \partial^\mu \phi^\dagger}$ and the conserved charge $N_\phi = \int \kla*{a^\dagger a - b^\dagger b}$ (i.e. difference between particles and anti-particles is conserved).
\subsection{Dirac fields}
The conjugate field is $\Pi=\frac{\partial\mathcal{L}}{\partial \dot{\psi}} = \i \psi^\dagger$ and $\klc*{\psi(\bm{x},t),\Pi(\bm{y},t)} = \i\delta\kla*{\bm{x} - \bm{y}}$.
\definition{Chirality}{
Define the projectors $P_L = \frac{1}{2}\kla*{1 - \gamma^5}$ and $P_R= \frac{1}{2}\kla*{1 + \gamma^5}$ and split $\psi=P_L\psi + P_R\psi = \psi_L + \psi_R$.
Then the two fields are eigenstates of $\gamma^5$ with $\gamma^5\psi_L=-\psi_L$ and $\gamma^5\psi_R=+\psi_R$.
}
A Lorentz transformation $S(\Lambda)$ has $S(\Lambda_1\Lambda_2) = S(\Lambda_1)S(\Lambda_2)$ and $\psi(x)\rightarrow \psi'(\Lambda x) = S(\Lambda) \psi(x)$; similarly, $\bar{\psi}(x) \rightarrow \bar{\psi}'(\Lambda x) = \bar{\psi}(x) S(\Lambda)^{-1}$.
Then $\bar{\psi}\psi$ is a Lorentz scalar (i.e. invariant under $S$), and $\bar{\psi} \gamma^\mu \psi$ is a Lorentz vector (i.e. $v'^\mu = \Lambda^\mu_{\ \nu} v^\nu$).
Meanwhile $\bar{\psi}\gamma^5\psi$ is a pseudo-scalar (also transforms trivially) and $\bar{\psi}\gamma^5\gamma^\mu\psi$ is a pseudo-vector (also transforms like a vector).

To construct the field, one can make an ansatz (for spin-$\frac{1}{2}$)
\begin{align}
\psi &= \int \sum_{s=1,2} \kla*{c_s(\bm{k}) u\kla*{\bm{k},s} \powe{-\i kx} + d_s^\dagger\kla*{\bm{k}} v\kla*{\bm{k},s}\powe{\i kx}}\,\tilde{d}k.
\end{align}
Pauli's exclusion principle states that no two fermions may be in the same state, which requires anti-symmetric creation and annihilation operators.

The Standard Model (SM) is an indirect product group
\begin{align}
\mathcal{G}_\text{SM} &= \underbrace{SU(3)_c}_\text{QCD} \times \underbrace{SU(2)_L \times U(1)_Y}_\text{electroweak}.
\end{align}
The covariant derivative is given by\footnote{The couplings are also sometimes written as $g_1\equiv g_Y$, $g_2\equiv g_W$ and $g_3\equiv g_S$.}

\end{align}

\section{Standard Model Lagrangian}

\end{align}

\begin{align}
B^{\mu\nu} &= \partial^\mu B^\nu - \partial^\nu B^\mu,\\
W_a^{\mu\nu} &= \partial^\mu W_a^\nu - \partial^\nu W_a^\mu - g_2\epsilon_{abc} W_b^\mu W_c^\nu,\\

\end{align}

\section{One-loop correction examples}
We will perform the one-loop calculations using dimensional regularization.
As a precursor, the key idea is to integrate over the loop momentum $k$ not in 4-dimensional but rather $D=4-2\epsilon$ dimensions\footnote{this $\epsilon$ is of course different from the one in the propagators}.
To keep the dimension intact, we then introduce a mass-scale $\mu$ to scale the integration measure $\frac{\d^4k}{(2\pi)^4} \rightarrow \mu^{4-D}\frac{\d^Dk}{(2\pi)^D}$.
For divergent integrals, we then split the result into a finite part and a collection of poles, which may then later be canceled by the introduction of counter-terms.
\subsection{Master integral}
The so-called Master integral is defined as
\begin{align}
I=\mu^{4-D}\intrndpi{D}{\frac{(n-1)!}{\kla*{k^2-Q+\i\epsilon}^n}}{k}.
\end{align}
First, consider the $k^0$-integral.
Its poles are at $\pm\sqrt{|\bm{k}|^2+Q}\mp\i\epsilon$ (do not think too hard about $\bm{k}$; it has a non-integer number of entries, but fortunately its precise form is irrelevant due to the arbitrary $Q$).
Both poles are avoided in the following contour, so
\begin{align}
0 &= \int_{\mathbb{R}} \kla*{k^0=t} + \int_0^{\pi/2} \kla*{k^0=R\powe{\i t}} + \int_{\mathbb{R}} \kla*{k^0=-\i t} + \int_{\pi/2}^\pi \kla*{k^0=R\powe{-\i t}}.
\end{align}
Both of the arcs go to zero as $R\rightarrow\infty$, although this involves some more work in the case of $n=1$.
Rearranging, we find $\int_{\mathbb{R}} \kla*{k^0=t} = \i\int_{\mathbb{R}} \kla*{k^0=-\i t}$, which completes the \textsc{Wick} rotation.
Crucially, we now have an effectively Euclidean momentum $k_E=(-\i k^0,\bm{k})$ with magnitude $k_E^2=-(k^0)^2-\bm{k}^2$.
Thus, the master integral can be resolved in spherical coordinates,
\begin{align}
I &= \i\mu^{4-D} \intx{0}{\infty}{ \frac{1}{(2\pi)^D} \int\d\Omega_D \frac{(n-1)!}{\kla*{-k^2-Q}^n} k^{D-1} }{k} \\
&= \i\mu^{4-D}\frac{S_D}{(2\pi)^D}(n-1)!(-1)^n Q^{D/2-n} \intx{0}{\infty}{\frac{t^{D/2-1}}{2\kla*{t+Q}^n}}{t}.
\end{align}
Here, $S_D=\int\Omega_D$ can be computed by considering the Gaussian integral
\begin{align}
\intrnd{D}{\powe{-k^2}}{k} &= \sqrt{\pi}^D = \int\Omega_D \intx{0}{\infty}{\powe{-k^2} k^{D-1}}{k} = S_D \frac{1}{2}\intx{0}{\infty}{\powe{-t} t^{D/2-1}}{t},
\end{align}
giving $S_D = \frac{2\pi^{D/2}}{\Gamma(D/2)}$.
Using the Beta function
\begin{align}
B(x,y) &= \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)} = \intx{0}{1}{t^{x-1}(1-t)^{y-1}}{t} = \intx{0}{\infty}{\frac{t^{x-1}}{(1+t)^{x+y}}}{t}, % = 2\intx{0}{\pi/2}{\sin(t)^{2x-1}\cos(t)^{2y-1}}{t},
\end{align}
we can then resolve the remaining integral to
\begin{align}
I &= \i\mu^{4-D}\frac{\pi^{D/2}}{(2\pi)^D \Gamma(D/2)}(n-1)!(-1)^n B\kla*{\frac{D}{2}, n-\frac{D}{2}} \\
&= (-1)^n \i \frac{1}{(4\pi)^{D/2}} \mu^{4-D} \frac{\Gamma\kla*{n-D/2}}{Q^{n-D/2}} = (-1)^n \frac{\i}{16\pi^2} \kla*{4\pi\frac{\mu^2}{Q}}^\epsilon \frac{\Gamma\kla*{n-2+\epsilon}}{Q^{n-2}}.
\end{align}
By expanding in leading order of $\epsilon$, we can simplify this to
\begin{align}
I &= (-1)^n \frac{\i}{16\pi^2} Q^{2-n} \kla*{1 + \epsilon\log\klb*{4\pi \frac{\mu^2}{Q}}} \Gamma(n-2+\epsilon).
\end{align}
Using $\Gamma(\epsilon)=\frac{1}{\epsilon}\Gamma(1+\epsilon)=\frac{1}{\epsilon}\kla*{1 + \epsilon\Gamma'(1)} = \frac{1}{\epsilon}-\gamma$, we then get (for $n=2$)
\begin{align}
I &= \frac{\i}{16\pi^2} \kla*{\Delta + \log\klb*{\frac{\mu^2}{Q}}}
\end{align}
with $\Delta = \frac{1}{\epsilon}-\gamma + \log(4\pi)$, and for $n=1$
\begin{align}
I &= -\frac{\i}{16\pi^2} Q \kla*{1 + \epsilon\log\klb*{4\pi\frac{\mu^2}{Q}}}\frac{\Gamma(\epsilon)}{\epsilon-1} = \frac{\i}{16\pi^2} Q \kla*{1 + \epsilon\log\klb*{4\pi\frac{\mu^2}{Q}}} (1+\epsilon) \kla*{\frac{1}{\epsilon} - \gamma}\\
&= \frac{\i}{16\pi^2} Q \kla*{\Delta+1 + \log\klb*{\frac{\mu^2}{Q}}}.
\end{align}
For $n\ge 3$ the integral is convergent and we simply get the scale-independent
\begin{align}
I &= (-1)^n \frac{\i}{16\pi^2} \frac{(n-3)!}{Q^{n-2}}.
\end{align}

Specifically,
\begin{align}
A_0(m) &= \frac{16\pi^2}{\i} \int \frac{1}{\kla*{k^2-m^2}} = m^2 \kla*{\Delta+1 + \log\klb*{\frac{\mu^2}{m^2}}},\\
B_0(p,m_1,m_2) &= \frac{16\pi^2}{\i} \int \frac{1}{\kla*{k^2-m_1^2}} \frac{1}{\klb*{\kla*{k+p}^2-m_2^2}} \\
&= \frac{16\pi^2}{\i} \int \frac{1}{\klb*{k^2-Q(x)}^2} = \int m^2\kla*{\Delta + \log\klb*{\frac{\mu^2}{Q(x)}}}
\end{align}
with $Q(x)=p^2x(x-1)+m_1^2(1-x)+m_2^2x$.
To understand how the second (and also higher order terms) can be reduced to this form, we need to understand the Feynman parametrization formula.
\section{Feynman parametrization formula}
The general theorem we will derive here is the remarkable identity
\begin{align}
\prod_{j=1}^n \frac{1}{A_j^{\alpha_j}} &= \frac{\Gamma\kla*{\sum \alpha_j}}{\prod \Gamma\kla*{\alpha_j}} \intx{0}{1}{ \dots \intx{0}{1}{ \delta\kla*{1-\sum x_j} \frac{\prod x_j^{\alpha_j-1}}{\klb*{\sum x_jA_j}^{\sum \alpha_j}} }{x_1} \dots }{x_n}
\end{align}
for any complex numbers $A_j$ for which zero is not contained in their \emph{convex hull}, and provided that $\Re\alpha_j>0$ for all $j=1,\dots,n$.
We can derive this in a constructive manner.
Let us start with the simpler identity
\begin{align}
\frac{1}{A^\alpha} &= \frac{1}{\Gamma(\alpha)} \intx{0}{\infty}{s^{\alpha-1}\powe{-sA}}{s} = \frac{1}{\Gamma(\alpha)} \partial_{-A}^{\alpha-1} \intx{0}{\infty}{\powe{-sA}}{s}.
\end{align}
Then the product over all terms can be written as some higher order derivative of
\begin{align}
\intx{0}{\infty}{\dots \intx{0}{\infty}{ \powe{-\kla*{s_1A_1+\dots+s_nA_n}} }{s_1}}{s_n}.
\end{align}
Let us now perform the substitutions $x=s_1+\dots+s_n$ and $x_j=\frac{s_j}{x}$ for $j=1,\dots,n-1$.
After the first substitution, the integration measure is still the same, and from the perspective of the other integrals $x$ can be viewed as a constant.
Then each of the other substitutions gives a factor of $x$, so we find (writing $x_n=1-x_1-\dots-x_{n-1}$)
\begin{align}
\intx{0}{1}{\dots \intx{0}{1}{ \intx{0}{\infty}{ x^{n-1} \powe{-x\kla*{x_1A_1+\dots + x_nA_n}} }{x}}{x_1}\dots}{x_{n-1}}.
\end{align}
The integral over $x$ can now be performed using the definition of the $\Gamma$-function and gives
\begin{align}
\intx{0}{1}{\dots \intx{0}{1}{ \frac{\Gamma(n)}{\klb*{\sum x_jA_j}^n} }{x_1} \dots}{x_{n-1}} &= \Gamma(n) \intx{0}{1}{\dots \intx{0}{1}{ \delta\kla*{1 - \sum x_j} \frac{1}{\klb*{\sum x_jA_j}^n} }{x_1} \dots}{x_{n}}.
\end{align}
But in order to get the general formula with $\alpha_j\neq 1$ we should first perform the derivatives to get
\begin{align}
\intx{0}{1}{\dots \intx{0}{1}{ \intx{0}{\infty}{ x^{n-1} (xx_1)^{\alpha_1-1} \dots (xx_n)^{\alpha_n-1} \powe{-x\kla*{x_1A_1+\dots + x_nA_n}} }{x}}{x_1}\dots}{x_{n-1}}.
\end{align}
We can still perform the $x$-integral and introduce the Dirac delta, so
\begin{align}
\Gamma\kla*{n + \sum(\alpha_j-1)} \intx{0}{1}{\dots \intx{0}{1}{ \delta\kla*{1 - \sum x_j} \frac{\prod x_j^{\alpha_j-1}}{\klb*{\sum x_jA_j}^{n + \sum(\alpha_j-1)}} }{x_1} \dots}{x_{n}}.
\end{align}
The $n$ cancels with the $-1$ contributions from all the $(\alpha_j-1)$ terms, and this therefore completes the proof.
It is crucial that $\sum x_jA_j > 0$ for all $x$, which relates to the requirement about the convex hull of the $A_j$.
\section{Anomalous magnetic moment}
One of the most famous results from the SM is the extremely accurate calculation of the anomalous magnetic moment $a_\mu=\frac{g-2}{2}$.
Of course not only QED contributes to this calculation, but the leading order correction (and by far the largest contribution over all) is determined by it.
The process involves the scattering $\gamma+\ell \rightarrow\bar{\ell}$, which can be shown to admit the general form
\begin{align}
-\i e \bar{u}(p') \klb*{\gamma^\mu F_1\kla*{q^2} + \frac{\i \sigma^{\mu\nu}}{2m} q_\nu F_2\kla*{q^2}}u(p),
\end{align}
where $q=p'-p$ is the exchange momentum.
By writing $\sigma^{\mu\nu}q_\nu = \frac{\i}{2}\klb*{\gamma^\mu\gamma^\nu - \gamma^\nu\gamma^\mu }q_\nu = \frac{\i}{2} \klb*{\gamma^\mu \slashed{q} - \slashed{q}\gamma^\mu }$ we can obtain (apply the Dirac equations $\slashed{p}u(p)=mu(p)$ etc.)
\begin{align}
\frac{\i \sigma^{\mu\nu}}{2m} q_\nu F_2\kla*{q^2} &= -\frac{1}{4m} F_2\kla*{q^2} \klb*{ \gamma^\mu \slashed{p}' - \gamma^\mu \slashed{p} - \slashed{p}'\gamma^\mu + \slashed{p}\gamma^\mu }\\
&= -\frac{1}{4m} F_2\kla*{q^2} \klb*{ \gamma^\mu \gamma^\nu p_\nu' + p_\nu \gamma^\nu \gamma^\mu - 2m\gamma^\mu}\\
&= -\frac{1}{4m} F_2\kla*{q^2} \klb*{ 2p'^\mu + 2p^\mu - 4m\gamma^\mu}.
\end{align}
The last term is of the $F_1$-form and can thus be absorbed in that function.
The only pure $F_2$-contribution is therefore of the form $(p+p')^\mu$, and of course the physical result should use $q^2\simeq 0$ since it is a photon (i.e. has no mass), so $a_\mu=F_2(0)$. (TODO: perhaps some more information about why these terms take this form and why only the $F_2$-term is relevant for $g-2$).
\subsection{Setting up the diagram}
The only one-loop diagram for this process is given by
\begin{align}
\ell_L(p) \rightarrow \ell(k) + \gamma(k-p) \rightarrow \gamma(q) + \ell(k+q) + \gamma(k-p) \rightarrow \gamma(q) + \ell_R(p').
\end{align}
The matrix element is therefore given by
\begin{align}

\end{align}
Let us introduce $\kappa=k+q$ and define $A=\kappa^2-m^2$, $B=k^2-m^2$ and $C=(k-p)^2$.
Then applying the Feynman parametrization yields
\begin{align}

\end{align}
Let us first evaluate the denominator,
\begin{align}
xA+yB+zC &= x\kappa^2-xm^2 + yk^2-ym^2 + zk^2+zp^2-2zkp\\

\end{align}
Using $p^2=m^2$ and $(q+p)^2=q^2+2qp+p^2=p'^2$ gives $qp=-\frac{1}{2}q^2$, and completing the square gives

Next we have to treat the numerator.
Note that the denominator is symmetric in $\tilde{k} = k+xq-zp$, so any terms linear in $\tilde{k}$ can simply be set to zero.

\example{Lie-Hadamard Formula}{$\powe{A}B\powe{-A}$}{
Define $F(s)=\powe{sA}B\powe{-sA}$.
Because the series expansion of the operator-valued exponential converges uniformly, we can write $F(s)=\sum_{n\ge 0} \frac{1}{n!}s^n\frac{\d^n F}{\d s^n} \Big\vert_{s=0}$.
Computing the derivatives is done recursively, i.e.
\begin{align}
F'(s) &= AF(s) - F(s)A = \klb*{A,F(s)},\\
F''(s) &= \klb*{A,F'(s)} = \klb*{A,\klb*{A,F(s)}},\\
\frac{\d^n F(s)}{\d s^n} &= \klb*{A,\klb*{A,\dots,\klb*{A,F(s)}}} \equiv \klb*{A,F(s)}_n,
\end{align}
where the iterated commutator admits $\klb*{A,B}_0=B$ and $\klb*{A,B}_{n+1}=\klb*{A,\klb*{A,B}_n}$.
Because $F(0)=B$, we then find the Lie-Hadamard formula
\begin{align}
F(s) &= \powe{sA}B\powe{-sA} = \sum_{n\ge 0} \frac{s^n}{n!} \klb*{A,B}_n = B + s\klb*{A,B} + \frac{s^2}{2!} \klb*{A,\klb*{A,B}} + \dots.
\end{align}
}

The adjoint representation is defined as $\kla*{\tilde{T}_a}_{bc}=-\i f_{abc}$, where $\klb*{T_a,T_b}=\i f_{abc}T_c$ is the Lie-Algebra for the generators.

and define $\Pi=\frac{1}{\sqrt{2}}\bm{\pi}\cdot\bm{\sigma}$.

Notice that $\tr\Pi=0$ because $\tr\sigma_i=0$.
Using $2\Pi^2=\kla*{\bm{\pi}\cdot\bm{\sigma}}^2=\bm{\pi}^2\sigma_0$, we also find $\bm{\pi}^2=\frac{1}{2}\tr\Pi^2$ and $\kla*{\partial_\mu\bm{\pi}}^2=\frac{1}{2}\tr\kla*{\partial_\mu\Pi}^2$.
Thus,
\begin{align}
\mathcal{L}_0 &= \bar{\psi}\kla*{\i\slashed{\partial}-m + \i g_\pi \sqrt{2}\gamma^5 \Pi}\psi + \frac{1}{4}\tr \klb*{\kla*{\partial_\mu\Pi}^2 - \mu^2 \Pi^2} - \lambda \frac{1}{4} \klb*{\tr\Pi^2}^2.
\end{align}
Because of the cyclicity of the trace, all of the latter terms are invariant under a global symmetry transformation with $U=\powe{\i\bm{\beta}\cdot\bm{\sigma}/2}$ with constant $\bm{\beta}$.
Likewise, all the factors of $U$ and $U^\dagger$ also cancel in the first term.

If we add the term $\mathcal{L}_1=-\epsilon\bar{\psi}\sigma_3\psi$ to the Lagrangian, then this is no longer the case.
Let us compute the transformed Lagrangian explicitly.
We have
\begin{align}
\bar{\psi}U^\dagger \sigma_3 U\psi &\equiv \bar{\psi} M \psi,
\end{align}
where using $\powe{\i x\bm{n}\cdot\bm{\sigma}} = \cos(x) + \i \sin(x) \bm{n}\cdot\bm{\sigma}$ we find
\begin{align}
M &= U^\dagger U\sigma_3 + U^\dagger \klb*{\sigma_3,U} = \sigma_3 + U^\dagger \i\sin(x) \bm{n}\cdot \klb*{\sigma_3,\bm{\sigma}}\\
&= \sigma_3 + U^\dagger \i \sin(x)n_k 2\i\epsilon_{3kl}\sigma_l = \sigma_3 - 2 U^\dagger \sin(x) \kla*{\bm{n}\times \bm{\sigma}}_3.
\end{align}
Here, $x=\frac{\beta}{2}$ and $\bm{\beta}=\beta\bm{n}$.
Crucially, this is different from $\mathcal{L}_1$ whenever $\bm{\beta}$ is not parallel to $\bm{e}_3$, i.e. for rotations that are not around the $z$-axis.

Finally, note that the Isospin is given by $T_3$.
For the nucleons, we find $T_3^p=+\frac{1}{2}$ and $T_3^n=-\frac{1}{2}$ (proton and neutron).
For the pions, we find $T_3^{\pi^0}=0$ and $T_3^{\pi^\pm}=\pm 1$.
}
\section{Non-Abelian Gauge Transformations}
A multiplet of fermion fields $\psi(x)$ transforms as
\begin{align}
\psi(x) &\rightarrow\psi'(x)=U\kla*{\theta(x)} \psi(x),
\end{align}
where $U\kla*{\theta(x)}=\exp\kla*{-\i\theta(x)}$ with $\theta(x)=\theta_a(x)T_a$.
We have $\klb*{T_a,T_b}=\i c_{abc}T_c$ and $\tr\kla*{T_aT_b}=\frac{1}{2}\delta_{ab}$.
The covariant derivative is given by
\begin{align}
D_\mu &= \partial_\mu + \i gA_\mu
\end{align}
with a gauge field matrix $A_\mu(x)=A_\mu^a(x)T_a$.
\example{Gauge Transform}{Multiplet Fermion Fields}{
Let us explicitly compute
\begin{align}
\kla*{D_\mu \psi}' &= \kla*{\partial_\mu + \i gA_\mu'} U\psi = U\kla*{\partial_\mu + \i g U^\dagger A_\mu' U}\psi + \kla*{\partial_\mu U}\psi\\
&= U \kla*{D_\mu + \i g \kla*{U^\dagger A_\mu' U - A_\mu } + U^\dagger \kla*{\partial_\mu U}}\psi.
\end{align}
We want this to have the form $UD_\mu\psi$, which we can achieve by letting the gauge fields transform as
\begin{align}
A_\mu' &= U A_\mu U^\dagger + \frac{\i}{g} \kla*{\partial_\mu U} U^\dagger.
\end{align}
For infinitesimal $\theta(x)$, we have $U = 1 -\i\theta_aT_a$ and thus
\begin{align}
A_\mu'^a &= 2\tr\klb*{A_\mu' T^a} = 2\tr\klb*{UA_\mu U^\dagger + \frac{\i}{g} \kla*{\partial_\mu U}U^\dagger}\\
&= 2\tr\klb*{A_\mu T_a + A_\mu \i \theta_bT_b T_a - \i\theta_bT_b A_\mu T_a + \frac{1}{g} \partial_\mu \theta_bT_b T_a}\\
&= A_\mu^a + 2\i\theta_b A_\mu^c\tr\klb*{\klb*{T_c,T_b} T_a} + \frac{1}{g} \partial_\mu \theta_a\\
&= A_\mu^a + \frac{1}{g} \partial_\mu \theta_a + 2c_{bcd} \theta_b A_\mu^c\tr\klb*{T_d T_a}\\
&= A_\mu^a + \frac{1}{g} \partial_\mu \theta_a + c_{bca} \theta_b A_\mu^c.
\end{align}
}
For a global transform, $\partial_\mu\theta=0$ such that $A_\mu'^a=A_\mu^a + c_{abc}\theta_bA_\mu^c$.
Since the adjoint representation is $\kla*{T_a}_{bc}=-\i c_{abc}$, this is equivalent to $A_\mu'=UA_\mu U^\dagger$.
This is precisely the way that matter fields transform, e.g. $\Phi'=U\Phi U^\dagger$.

Using $D_\mu=\partial_\mu + \i gA_\mu^a T_a = \partial_\mu + g\kla*{c_{abc}}_{bc} A_\mu^a$, we can compute
\begin{align}
\frac{1}{g} \kla*{D_\mu\theta}^a &= \frac{1}{g}\partial_\mu \theta^a + c_{cab} A_\mu^c \theta^b,
\end{align}
such that for local transformations in the adjoint representation we have
\begin{align}
A_\mu'^a &= A_\mu^a + \frac{1}{g} \kla*{D_\mu \theta}^a.
\end{align}
\example{Unitary Gauge}{Scalar Doublet}{
Consider a scalar doublet $\Phi = \powe{\i \bm{\sigma} \cdot \frac{1}{v} \bm{\chi}} \kla*{0, \frac{v+\eta(x)}{\sqrt{2}}}$, where $\chi^a(x)$ and $\eta(x)$ are real scalar fields with vanishing vacuum expectation values $\bra*{\eta}_0=\bra*{\chi^a}_0=0$.
We define another doublet as $\Phi^u = U(x)\Phi = \frac{1}{\sqrt{2}} \kla*{0, v + \eta(x)}$.
Then according to the formula we derived previously, we now need to transform the gauge fields to $B_\mu\equiv A_\mu'$ with
\begin{align}
\frac{\bm{\sigma}}{2}\cdot \bm{B}_\mu &= U \frac{\bm{\sigma}}{2}\cdot \bm{A}_\mu U^\dagger + \frac{\i}{g} \kla*{\partial_\mu U}U^\dagger.
\end{align}
Then $D_\mu\Phi^u = UD_\mu\Phi$ (this is exactly the same algebra as before, just different symbols).
}
