\section{Introducing the Hamiltonian}
Let $J_\textbf{k} := \frac{1}{2} J_0 \kla\left( \cos k_x + \cos k_y \right)$ for some constant $J_0$.
Then a ``certain interaction'' may be described by the effective Hamiltonian $\hat{H} = \hat{H}_0 + \hat{W}$ where
\begin{align}
\hat{H}_0 &= -\sum_{\textbf{k}s} \kla\left( \frac{1}{2} J_\textbf{k} + \mu  \right) \hat{c}_{\textbf{k}s}^\dagger  \hat{c}_{\textbf{k}s},\\
\hat{W} &= \frac{1}{2} \sum_{\textbf{k}_{1,2,3,4}} \kla\left( J_{\textbf{k}_1} + J_{\textbf{k}_2} + J_{\textbf{k}_3} + J_{\textbf{k}_4} \right)  \hat{c}_{\textbf{k}_1 \uparrow}^\dagger \hat{c}_{\textbf{k}_2 \downarrow}^\dagger \hat{c}_{\textbf{k}_3 \downarrow} \hat{c}_{\textbf{k}_4 \uparrow} \delta_{\textbf{k}_1 + \textbf{k}_2 + \textbf{k}_3 + \textbf{k}_4, \textbf{0} }.
\end{align}

\section{Mean-field approximation}
Let $\hat{b}_{\mathbf{k}} := \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow}$ be an effective bosonic operator, since
\footnote{Note that $[\hat{a},\hat{b}\hat{c}] = [\hat{a},\hat{b}]\hat{c} + \hat{b}[\hat{a},\hat{c}] = \{\hat{a},\hat{b}\}\hat{c} - \hat{b}\{\hat{a},\hat{c}\} $ and $[ \hat{a}\hat{b}, \hat{c} ] = [ \hat{a}, \hat{c} ]\hat{b} + \hat{a}[ \hat{b}, \hat{c} ] = \hat{a} \{\hat{b}, \hat{c} \} -  \{\hat{a}, \hat{c} \} \hat{b}$}
\begin{align}
\klam\left[ \hat{b}_{\mathbf{k}}, \hat{b}_{\mathbf{k}'} \right] &= \klam\left[ \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{-\mathbf{k}'\downarrow} \hat{c}_{\mathbf{k}'\uparrow}   \right] = \\
&= \hat{c}_{-\mathbf{k}\downarrow} \klam\left[ \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{-\mathbf{k}'\downarrow} \hat{c}_{\mathbf{k}'\uparrow}   \right] + \klam\left[ \hat{c}_{-\mathbf{k}\downarrow},  \hat{c}_{-\mathbf{k}'\downarrow} \hat{c}_{\mathbf{k}'\uparrow}   \right] \hat{c}_{\mathbf{k}\uparrow} = \\
&= \hat{c}_{-\mathbf{k}\downarrow} \klam\left\{ \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{-\mathbf{k}'\downarrow}  \right\} \hat{c}_{\mathbf{k}'\uparrow} - \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{-\mathbf{k}'\downarrow} \klam\left\{ \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{\mathbf{k}'\uparrow}   \right\} + \dots = 0,\\
\klam\left[ \hat{b}_{\mathbf{k}}, \hat{b}_{\mathbf{k}'}^\dagger \right] &= \klam\left[ \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{\mathbf{k}'\uparrow}^\dagger \hat{c}_{-\mathbf{k}'\downarrow}^\dagger  \right] = \\
&= \hat{c}_{-\mathbf{k}\downarrow} \klam\left[  \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{\mathbf{k}'\uparrow}^\dagger \hat{c}_{-\mathbf{k}'\downarrow}^\dagger  \right] + \klam\left[ \hat{c}_{-\mathbf{k}\downarrow} ,  \hat{c}_{\mathbf{k}'\uparrow}^\dagger \hat{c}_{-\mathbf{k}'\downarrow}^\dagger  \right] \hat{c}_{\mathbf{k}\uparrow} = \\
&= \hat{c}_{-\mathbf{k}\downarrow} \klam\left\{  \hat{c}_{\mathbf{k}\uparrow},  \hat{c}_{\mathbf{k}'\uparrow}^\dagger   \right\} \hat{c}_{-\mathbf{k}'\downarrow}^\dagger -  \hat{c}_{\mathbf{k}'\uparrow}^\dagger \klam\left\{ \hat{c}_{-\mathbf{k}\downarrow} ,  \hat{c}_{-\mathbf{k}'\downarrow}^\dagger  \right\} \hat{c}_{\mathbf{k}\uparrow} = \\
&= \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{-\mathbf{k}\downarrow}^\dagger \delta_{\mathbf{k}\mathbf{k}'} - \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{\mathbf{k}\uparrow} \delta_{\mathbf{k}\mathbf{k}'} = \\
&= \delta_{\mathbf{k}\mathbf{k}'} \kla\left( 1 - \hat{c}_{-\mathbf{k}\downarrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow} - \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{\mathbf{k}\uparrow} \right).
\end{align}
When applied to $\Bra\langle 0 |$ or $\Ket | 0 \rangle$ this commutator simplifies to $\left[ \hat{b}_{\mathbf{k}}, \hat{b}_{\mathbf{k}'}^\dagger \right] \equiv  \delta_{\mathbf{k}\mathbf{k}'}$, but this relation does not hold in general.
Let $\hat{A} = \bra\langle \hat{A} \rangle + \Delta\hat{A}$ and $\hat{B} = \bra\langle \hat{B} \rangle + \Delta\hat{B}$. 
Then the mean-field approximation consists of disregarding the term $\Delta\hat{A}\Delta\hat{B}$ in the product
\begin{align}
\hat{A}\hat{B} &= \bra\langle \hat{A} \rangle\bra\langle \hat{B} \rangle + \bra\langle \hat{A} \rangle\Delta\hat{B} + \bra\langle \hat{B} \rangle\Delta\hat{A} + \Delta\hat{A}\Delta\hat{B}.
\end{align}
Since $\Delta\hat{A} = \hat{A} - \bra\langle \hat{A} \rangle$, we may then write
\begin{align}
\hat{A}\hat{B} &\simeq \bra\langle \hat{A} \rangle\bra\langle \hat{B} \rangle + \bra\langle \hat{A} \rangle \kla\left( \hat{B} - \bra\langle \hat{B} \rangle \right) + \bra\langle \hat{B} \rangle\kla\left( \hat{A} - \bra\langle \hat{A} \rangle \right) = \\
&= \bra\langle \hat{A} \rangle\hat{B} + \bra\langle \hat{B} \rangle\hat{A} - \bra\langle \hat{A} \rangle\bra\langle \hat{B} \rangle.
\end{align}
Let us assume that $\bra\langle \hat{c}_{\textbf{k}_1\downarrow} \hat{c}_{\textbf{k}_2\uparrow} \rangle \propto \delta_{\textbf{k}_1, -\textbf{k}_2}$.
Then the mean-field approximation for $\hat{A} = \hat{c}_{\textbf{k}_1\uparrow}^\dagger \hat{c}_{\textbf{k}_2\downarrow}^\dagger$ and $\hat{B} = \hat{c}_{\textbf{k}_3\downarrow} \hat{c}_{\textbf{k}_4\uparrow}$ is
\begin{align}
\hat{A}\hat{B} &\simeq \bra\langle {\hat{b}_{\textbf{k}_1}^\dagger }\rangle \delta_{\textbf{k}_1,- \textbf{k}_2} \hat{c}_{\textbf{k}_3\downarrow} \hat{c}_{\textbf{k}_4\uparrow} + \bra\langle {\hat{b}_{\textbf{k}_4} }\rangle  \delta_{\textbf{k}_3,- \textbf{k}_4} \hat{c}_{\textbf{k}_1\uparrow}^\dagger \hat{c}_{\textbf{k}_2\downarrow}^\dagger - \bra\langle {\hat{b}_{\textbf{k}_1}^\dagger }\rangle \bra\langle {\hat{b}_{\textbf{k}_4}  }\rangle  \delta_{\textbf{k}_1,- \textbf{k}_2} \delta_{\textbf{k}_3,- \textbf{k}_4}.
\end{align}
Inserting into the interaction term with $\textbf{k}_1 \rightarrow \textbf{k}$ and $\textbf{k}_4 \rightarrow \textbf{k}'$ yields
\begin{align}
\hat{W} &\simeq \frac{1}{2} \sum_{\textbf{k}, \textbf{k}'} \kla\left( {2J_{\textbf{k}} + 2J_{\textbf{k}'} } \right) \klam\left[ \langle \hat{b}_{\textbf{k}}^\dagger  \rangle \hat{c}_{-\mathbf{k}'\downarrow} \hat{c}_{\mathbf{k}'\uparrow} + \bra\langle \hat{b}_{\textbf{k}'} \rangle \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger  - \bra\langle \hat{b}_{\mathbf{k}'} \rangle\bra\langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right].
\end{align}
This leads to the mean-field Hamiltonian
\begin{align}
\hat{H} &= \hat{H}_0 + \hat{W} \simeq \\
&\simeq \hat{H}_0 + \sum_{\mathbf{k}\mathbf{k}'} W_{\mathbf{k}\mathbf{k}'} \kla\left( \bra\langle \hat{b}_{\mathbf{k}}^\dagger \rangle\hat{c}_{-\mathbf{k}'\downarrow} \hat{c}_{\mathbf{k}'\uparrow} + \bra\langle \hat{b}_{\mathbf{k}'} \rangle \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \right) - \sum_{\textbf{k}\textbf{k}'} W_{\textbf{k}\textbf{k}'} \bra\langle \hat{b}_{\mathbf{k}'} \rangle\bra\langle \hat{b}_{\mathbf{k}}^\dagger \rangle
\end{align}
where we introduced
\begin{align}
W_{\textbf{k}\textbf{k}'} &:= J_\textbf{k} + J_{\textbf{k}'} = \frac{J_0}{2} \kla\left( \cos k_x + \cos k_y + \cos k_x' + \cos k_y' \right).
\end{align}
We can bring this into a simpler form by defining 
\begin{align}
\Delta_{\mathbf{k}} &:= -\sum_{\mathbf{k}'} W_{\mathbf{k}\mathbf{k}'} \bra\langle \hat{b}_{\mathbf{k}'} \rangle.
\end{align}
Since $W_{\mathbf{k}\mathbf{k}'}^\ast = W_{\mathbf{k}'\mathbf{k}} (= W_{\mathbf{k}\mathbf{k}'})$, we may write the Hamiltonian as
\begin{align}
\hat{H} &= \hat{H}_0 - \sum_{\mathbf{k}} \kla\left( \Delta_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger + \Delta_{\mathbf{k}}^\ast \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow}  \right) + \sum_{\textbf{k}} \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle.
\end{align}
Defining $\epsilon_\textbf{k} := \frac{1}{2} J_\textbf{k} + \mu $, we can write this as
\begin{align}
\hat{H} &= \sum_{\mathbf{k}} \kla\left( \epsilon_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{\mathbf{k}\uparrow} - \Delta_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger - \Delta_{\mathbf{k}}^\ast \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow} + \epsilon_{\mathbf{k}} \hat{c}_{\mathbf{k}\downarrow}^\dagger \hat{c}_{\mathbf{k}\downarrow}  \right)  + \sum_{\textbf{k}} \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle = \\
&= \sum_{\mathbf{k}} \begin{pmatrix}
\epsilon_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger - \Delta_{\mathbf{k}}^\ast \hat{c}_{-\mathbf{k}\downarrow} \\ -\Delta_{\mathbf{k}}\hat{c}_{\mathbf{k}\uparrow}^\dagger - \epsilon_{\mathbf{k}} \hat{c}_{\mathbf{k}\downarrow}
\end{pmatrix}^\top \begin{pmatrix}
\hat{c}_{\mathbf{k}\uparrow} \\ \hat{c}_{-\mathbf{k}\downarrow}^\dagger
\end{pmatrix} + \sum_{\mathbf{k}}\epsilon_{\mathbf{k}} + \sum_{\textbf{k}} \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle = \\
&= \sum_{\mathbf{k}} \begin{pmatrix}
\hat{c}_{\mathbf{k}\uparrow}^\dagger & \hat{c}_{-\mathbf{k}\downarrow}
\end{pmatrix} \underbrace{\begin{pmatrix}
\epsilon_{\mathbf{k}} & -\Delta_{\mathbf{k}} \\ -\Delta_{\mathbf{k}}^\ast & -\epsilon_{\mathbf{k}}
\end{pmatrix}}_{=:\,M_{\mathbf{k}}} \underbrace{\begin{pmatrix}
\hat{c}_{\mathbf{k}\uparrow} \\ \hat{c}_{-\mathbf{k}\downarrow}^\dagger
\end{pmatrix}}_{=:\,\vec{u}_{\mathbf{k}}}  + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right).
\end{align}
The eigenvalues of $M_{\mathbf{k}}$ can be determined by solving the quadratic equation
\begin{align}
0 &= \lambda^2 - \lambda \tecxt\text{tr\,}M_{\mathbf{k}} + \det M_{\mathbf{k}} = \lambda^2 - \epsilon_{\mathbf{k}}^2 - |\Delta_{\mathbf{k}}|^2,
\end{align}
which gives $\lambda = \pm E_{\mathbf{k}} = \pm \sqrt{\epsilon_{\mathbf{k}}^2 + |\Delta_{\mathbf{k}}|^2}$.
We can find the eigenvectors $\vec{\gamma}_{\mathbf{k}s} := (\alpha_{\mathbf{k}s}, \beta_{\mathbf{k}s})^\top$ of $M_{\mathbf{k}}$ by solving the system of equations
\begin{align}
\epsilon_{\mathbf{k}}\alpha_{\mathbf{k}s} - \Delta_{\mathbf{k}}\beta_{\mathbf{k}s} &= \pm E_\textbf{k} \alpha_{\mathbf{k}s},\\
-\Delta_{\mathbf{k}}^\ast\alpha_{\mathbf{k}s} - \epsilon_{\mathbf{k}}\beta_{\mathbf{k}s} &= \pm E_{\mathbf{k}} \beta_{\mathbf{k}s}
\end{align}
where $s\in\{\uparrow, \downarrow\}$ for $\pm E_\textbf{k}$ respectively.
This yields $\beta_{\mathbf{k}\uparrow} = \frac{\epsilon_{\mathbf{k}} - E_{\mathbf{k}}}{\Delta_{\mathbf{k}}}\alpha_{\mathbf{k}\uparrow}$ and $\alpha_{\mathbf{k}\downarrow} = \frac{E_{\mathbf{k}} - \epsilon_{\mathbf{k}}}{\Delta_{\mathbf{k}}^\ast} \beta_{\mathbf{k}\downarrow}$.
The normalized eigenvectors can then be written as
\begin{align}
\vec{\gamma}_{\mathbf{k}\uparrow} &= \frac{1}{N}\begin{pmatrix}
\Delta_{\mathbf{k}} \\ \epsilon_{\mathbf{k}} - E_{\mathbf{k}}
\end{pmatrix} = \\
&= \frac{1}{\sqrt{|\Delta_{\mathbf{k}}|^2 + \epsilon_{\mathbf{k}}^2 + E_{\mathbf{k}}^2 - 2\epsilon_{\mathbf{k}} E_{\mathbf{k}}}} \begin{pmatrix}
\Delta_{\mathbf{k}}  \\ \epsilon_{\mathbf{k}} - E_{\mathbf{k}}
\end{pmatrix} = \\
&= \frac{1}{\sqrt{2E_{\mathbf{k}} (E_{\mathbf{k}} -\epsilon_{\mathbf{k}})}} \begin{pmatrix}
\Delta_{\mathbf{k}} \\ \epsilon_{\mathbf{k}} - E_{\mathbf{k}}
\end{pmatrix} =: \begin{pmatrix}
\alpha_\textbf{k} \\ -\beta_\textbf{k}
\end{pmatrix} , \\
\vec{\gamma}_{\mathbf{k}\downarrow} &= \frac{1}{N}\begin{pmatrix}
E_{\mathbf{k}} - \epsilon_{\mathbf{k}} \\ \Delta_{\mathbf{k}}^\ast
\end{pmatrix} = \\
&= \frac{1}{\sqrt{2E_{\mathbf{k}} (E_{\mathbf{k}} -\epsilon_{\mathbf{k}})}} \begin{pmatrix}
E_{\mathbf{k}} - \epsilon_{\mathbf{k}} \\ \Delta_{\mathbf{k}}^\ast
\end{pmatrix} = \begin{pmatrix}
\beta_\textbf{k} \\ \alpha_\textbf{k}^\ast
\end{pmatrix}.
\end{align}
Here we defined the coefficients
\begin{align}
\alpha_{\mathbf{k}} &:= \frac{\Delta_{\mathbf{k}}}{\sqrt{2E_{\mathbf{k}} ( E_{\mathbf{k}} - \epsilon_{\mathbf{k}})}},\\
\beta_{\mathbf{k}} &:= \frac{E_{\mathbf{k}} - \epsilon_{\mathbf{k}}}{\sqrt{2E_{\mathbf{k}} ( E_{\mathbf{k}} - \epsilon_{\mathbf{k}})}} \ge 0
\end{align}
satisfying $|\alpha_\textbf{k}|^2 + \beta_\textbf{k}^2 = 1$.
Since $\vec{\gamma}_{\mathbf{k}\uparrow}^\dagger \vec{\gamma}_{\mathbf{k}\downarrow} = \alpha_{\mathbf{k}}^\ast \beta_\textbf{k} - \beta_\textbf{k} \alpha_\textbf{k}^\ast = 0$, the eigenvectors are orthogonal.
Thus the diagonalized Hamiltonian now takes the form
\begin{align}
\hat{H} &= \sum_{\mathbf{k}} \vec{u}_{\mathbf{k}}^\dagger  U_{\mathbf{k}}^\dagger \begin{pmatrix}
E_{\mathbf{k}} & 0 \\
0 & -E_{\mathbf{k}}
\end{pmatrix} U_{\mathbf{k}} \vec{u}_{\mathbf{k}} + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right)
\end{align}
where the transformation matrix is given by
\begin{align}
U_{\mathbf{k}} &= \begin{pmatrix}
\alpha_\textbf{k}^\ast & -\beta_\textbf{k} \\ \beta_\textbf{k} & \alpha_\textbf{k}
\end{pmatrix}.
\end{align}
We may now define
\begin{align}
\vec{\gamma}_{\mathbf{k}} &:= U_{\mathbf{k}}\vec{u}_{\mathbf{k}} = \vec{\gamma}_{\mathbf{k}\uparrow} \hat{c}_{\mathbf{k}\uparrow} + \vec{\gamma}_{\mathbf{k}\downarrow} \hat{c}_{-\mathbf{k}\downarrow}^\dagger,
\end{align}
or, written in terms of the vector components, 
\begin{align}
\hat{\gamma}_{\mathbf{k}\uparrow} &:=  \alpha^\ast_{\mathbf{k}}\hat{c}_{\mathbf{k}\uparrow} -\beta_\textbf{k} \hat{c}_{-\mathbf{k}\downarrow}^\dagger,\\
\hat{\gamma}_{-\mathbf{k}\downarrow}^\dagger &:=  \beta_\textbf{k} \hat{c}_{\mathbf{k}\uparrow} + \alpha_\textbf{k} \hat{c}_{-\mathbf{k}\downarrow}^\dagger.
\end{align}
Note that we deliberately chose the form $\vec{\gamma}_{\mathbf{k}} = (\hat{\gamma}_{\mathbf{k}\uparrow}, \hat{\gamma}_{-\mathbf{k}\downarrow}^\dagger)^\top$ for in analogy to the previous form of $\vec{u}_{\mathbf{k}}$.
These operators satisfy $\klammer\left\{ \hat{\gamma}_{\mathbf{k}s}, \hat{\gamma}_{\mathbf{k}'s'}^\dagger \right\} = \delta_{ss'} \delta_{\mathbf{k}\mathbf{k}'}$ and $\klammer\left\{ \hat{\gamma}_{\mathbf{k}s}, \hat{\gamma}_{\mathbf{k}'s'} \right\} = 0$, and may therefore be interpreted as operators creating fermionic quasi-particles.
and the explicitly diagonalized Hamiltonian becomes
\begin{align}
\hat{H} &= \sum_{\mathbf{k}} \vec{\gamma}_{\mathbf{k}}^\dagger \begin{pmatrix}
E_{\mathbf{k}} & 0 \\ 0 & -E_{\mathbf{k}}
\end{pmatrix} \vec{\gamma}_{\mathbf{k}} + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right) =  \\
&= \sum_{\mathbf{k}} \hat{\gamma}_{\mathbf{k}\uparrow}^\dagger E_{\mathbf{k}} \hat{\gamma}_{\mathbf{k}\uparrow} + \hat{\gamma}_{-\mathbf{k}\downarrow} (-E_{\mathbf{k}}) \hat{\gamma}_{-\mathbf{k}\downarrow}^\dagger + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right) =  \numberthis\\
&= \sum_{\mathbf{k}s} E_{\mathbf{k}} \hat{\gamma}_{\mathbf{k}s}^\dagger \hat{\gamma}_{\mathbf{k}s} + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} - E_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right) \numberthis \label{eqn:diagonalized hamiltonian}.
\end{align}
Since $E_{\mathbf{k}} \ge 0$, we expect there to be a ground-state with respect to the quasi-particles satisfying $\hat{\gamma}_{\mathbf{k}s} \Ket | \psi_\tecxt\text{BCS} \rangle = 0$.
Let $\Ket | \psi_\tecxt\text{BCS} \rangle = \mathcal{N}\hat{V}\Ket | 0 \rangle$, where $\mathcal{N}$ is a normalization constant, and assume that $\hat{V}$ takes the form
\begin{align}
\hat{V} &= \prod_{\mathbf{k}} \hat{V}_{\mathbf{k}} = \prod_{\mathbf{k}}  \exp\kla\left( u_{\mathbf{k}}  \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \right).
\end{align}
We want to determine the coefficients, such that 
\begin{align}
\hat{\gamma}_{\mathbf{k}\uparrow} \hat{V}\Ket | 0 \rangle = 0.
\end{align}
Note that 
\begin{align}
\klam\left[ \hat{c}_{\mathbf{k}\uparrow}, u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger \right] &= u_{\mathbf{q}} \klammer\left\{ \hat{c}_{\mathbf{k}\uparrow}, \hat{c}_{\mathbf{q}\uparrow}^\dagger \right\} \hat{c}_{-\mathbf{q}\downarrow}^\dagger = u_{\mathbf{k}} \hat{c}_{-\mathbf{k}\downarrow}^\dagger \delta_{\mathbf{k}\textbf{q}},\\
\klam\left[ \hat{c}_{\mathbf{k}\uparrow}^\dagger, u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger \right] &= 0.
\end{align}
We will make use of the identity $\klam\left[ \hat{a}, \hat{b}^n \right] = n\hat{c}\hat{b}^{n-1}$, which holds given that $\hat{c} = \klam\left[ \hat{a}, \hat{b} \right]$ commutes with $\hat{b}$.
For $\hat{a} = \hat{\gamma}_{\mathbf{k}\uparrow}$ and $\hat{b} = u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger$ this is satisfied.
We can now compute the desired commutator
\begin{align}
\klam\left[ \hat{\gamma}_{\mathbf{k}\uparrow}, \hat{V}_{\mathbf{q}} \right] &= \sum_{n\ge 0} \frac{1}{n!} \klam\left[ \hat{\gamma}_{\mathbf{k}\uparrow}, \kla\left( u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger \right)^n \right] = \\
&= \sum_{n\ge 0}\frac{1}{n!} n \klam\left[ \alpha_{\mathbf{k}}^\ast \hat{c}_{\mathbf{k}\uparrow} - \beta_{\mathbf{k}} \hat{c}_{-\mathbf{k}\downarrow}^\dagger, u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger \right] \kla\left( u_{\mathbf{q}} \hat{c}_{\mathbf{q}\uparrow}^\dagger \hat{c}_{-\mathbf{q}\downarrow}^\dagger \right)^{n-1} = \\
&= \delta_{\mathbf{k}\textbf{q}} \alpha_{\mathbf{k}}^\ast u_{\mathbf{k}} \hat{c}_{-\mathbf{k}\downarrow}^\dagger \sum_{n\ge 1} \frac{1}{(n-1)!} \kla\left( u_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \right)^{n-1} = \\
&= \delta_{\mathbf{k}\textbf{q}} \alpha_{\mathbf{k}}^\ast u_{\mathbf{k}} \hat{c}_{-\mathbf{k}\downarrow}^\dagger \hat{V}_{\mathbf{k}}.
\end{align}
Since also $\klam\left[ \hat{\gamma}_{\mathbf{k}\uparrow}, \hat{V}_{\mathbf{q}} \right] \propto \delta_{\mathbf{k}\textbf{q}}$, we find 
\begin{align}
0 &= \hat{\gamma}_{\mathbf{k}\uparrow}\hat{V} \Ket | 0 \rangle = \prod_{\mathbf{q\neq k}} \hat{V}_{\mathbf{q}} \hat{\gamma}_{\mathbf{k}\uparrow} \hat{V}_{\mathbf{k}} \Ket | 0 \rangle = \\
&= \prod_{\mathbf{q\neq k}} \hat{V}_{\mathbf{q}} \kla\left( \hat{V}_{\textbf{k}} \hat{\gamma}_{\mathbf{k}\uparrow} + \alpha_{\mathbf{k}} u_{\mathbf{k}} \hat{V}_{\mathbf{k}} \hat{c}_{-\mathbf{k}\downarrow}^\dagger  \right) \Ket | 0 \rangle = \\
&= \prod_{\mathbf{q}} \hat{V}_{\mathbf{q}} \left( - \beta_{\mathbf{k}} + \alpha_{\mathbf{k}}^\ast u_{\mathbf{k}} \right) \hat{c}_{\mathbf{k}\downarrow}^\dagger \Ket | 0 \rangle.
\end{align}
This implies $u_{\mathbf{k}} = \frac{\beta_{\mathbf{k}}}{\alpha_{\mathbf{k}}^\ast}$ and therefore
\begin{align}
\hat{V}_{\mathbf{k}} &= \sum_{n\ge 0} \frac{1}{n!} \kla\left( \frac{\beta_{\mathbf{k}}}{\alpha_{\mathbf{k}}^\ast} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \right)^n = \\
&= 1 + \frac{\beta_{\mathbf{k}}}{\alpha_{\mathbf{k}}^\ast} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger + \frac{1}{2} \frac{\beta_{\mathbf{k}}^2}{(\alpha_{\mathbf{k}}^\ast)^2} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger + \dots = \\
&= 1 + \frac{\beta_{\mathbf{k}}}{\alpha_{\mathbf{k}}^\ast} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger - \frac{1}{2} \frac{\beta_{\mathbf{k}}^2}{(\alpha_{\mathbf{k}}^\ast)^2} \hat{c}_{-\mathbf{k}\downarrow}^\dagger \kla\left( \hat{c}_{\mathbf{k}\uparrow}^\dagger \right)^2 \hat{c}_{-\mathbf{k}\downarrow}^\dagger + \dots = \\
&= 1 + \frac{\beta_{\mathbf{k}}}{\alpha_{\mathbf{k}}^\ast} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger.
\end{align}
Redefining the normalization, the BCS ground state may therefore be written as
\begin{align}
\Ket | \psi_\tecxt\text{BCS} \rangle &= \mathcal{N} \hat{V}\Ket | 0 \rangle = \mathcal{N} \prod_{\mathbf{k}} \kla\left( \alpha_{\mathbf{k}}^\ast + \beta_{\mathbf{k}} \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger  \right) \Ket | 0 \rangle.
\end{align}
%For $k \gg k_\tecxt\text{F}$ we have $\epsilon_{\mathbf{k}} \gg |\Delta_{\mathbf{k}}|$ and hence $\beta_{\mathbf{k}} \rightarrow 0$.
We can explicitly derive the normalization constant by considering 
\begin{align}
1 &= \braket\langle \psi_\tecxt\text{BCS} | \psi_\tecxt\text{BCS} \rangle = |\mathcal{N}|^2 \braket\langle 0 | \hat{V}^\dagger \hat{V} | 0 \rangle = \\
&= |\mathcal{N}|^2 \braket\langle 0 | \prod_\textbf{k} \kla\left( \alpha_\textbf{k} + \beta_\textbf{k}^\ast \hat{b}_\textbf{k} \right) \kla\left( \alpha_\textbf{k}^\ast + \beta_\textbf{k} \hat{b}_\textbf{k}^\dagger \right) | 0 \rangle = \\
&= |\mathcal{N}|^2 \braket\langle 0 | \prod_\textbf{k} \klam\left[ \kla\left( \alpha_\textbf{k}^\ast + \beta_\textbf{k} \hat{b}_\textbf{k}^\dagger \right) \kla\left( \alpha_\textbf{k} + \beta_\textbf{k}^\ast \hat{b}_\textbf{k} \right) + |\beta_\textbf{k}|^2 \klam\left[ \hat{b}_\textbf{k}, \hat{b}_\textbf{k}^\dagger \right] \right] | 0 \rangle = \\
&= |\mathcal{N}|^2 \braket\langle 0 | \prod_\textbf{k} \klam\left[ |\alpha_\textbf{k}^\ast|^2 + |\beta_\textbf{k}|^2 \kla\left( 1 - \hat{c}_{-\mathbf{k}\downarrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow} - \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{\mathbf{k}\uparrow} \right) \right] | 0 \rangle = \\
&= |\mathcal{N}|^2 \braket\langle 0 | \prod_\textbf{k} \klam\left[ |\alpha_\textbf{k}|^2 + |\beta_\textbf{k}|^2 \right] | 0 \rangle = |\mathcal{N}|^2.
\end{align}
Since $|\mathcal{N}| = 1$, the state is already normalized correctly.

\section{Energy reduction in the BCS state}
(Der Teil hier hat mit dem nicht-unitären $V$ eigentlich ganz gut funktioniert -- vorausgesetzt, das Ergebnis stimmt).\\
We still have to answer the question, whether the system prefers the state $\Ket | \psi_\tecxt\text{BCS} \rangle$ over the ``normal'' ground state
\begin{align}
\Ket | \psi_\tecxt\text{F} \rangle &:= \prod_{|\textbf{k}| < k_\tecxt\text{F},s} \hat{c}_{\textbf{k}s}^\dagger \Ket | 0 \rangle =: \hat{F} \Ket | 0 \rangle,
\end{align}
where $k_\tecxt\text{F}$ is the Fermi momentum.
Note that $\hat{c}_{\textbf{k}s}^\dagger \hat{c}_{\textbf{k}'s'}^\dagger = - \hat{c}_{\textbf{k}'s'}^\dagger \hat{c}_{\textbf{k}s}^\dagger$.
However, the product above is only well-defined up to a sign, since we have not specified an order.
We may make use of this by regrouping terms in a way that simplifies the expression to
\begin{align}
\hat{F} &= \prod_{|\textbf{k}|<k_\tecxt\text{F}} \hat{c}_{\textbf{k}\uparrow}^\dagger \hat{c}_{\textbf{k}\downarrow}^\dagger = \prod_{|\textbf{k}|<k_\tecxt\text{F}} \hat{c}_{\textbf{k}\uparrow}^\dagger \hat{c}_{-\textbf{k}\downarrow}^\dagger = \prod_{|\textbf{k}|<k_\tecxt\text{F}} \hat{b}_\textbf{k}^\dagger.
\end{align}
In the second step we made use of the fact that both $\hat{c}_{\textbf{k}\uparrow}^\dagger \hat{c}_{\textbf{k}\downarrow}^\dagger$ and $\hat{c}_{-\textbf{k}\uparrow}^\dagger \hat{c}_{-\textbf{k}\downarrow}^\dagger$ appear in the product for any $\textbf{k}$ (except $\textbf{k}=0$, but in that case $\textbf{k}=-\textbf{k}$ anyway).
Just as before, the rearranging of terms may result in a different sign.
But this should not affect any expectation values, since this ``phase'' cancels when combined with $\hat{F}^\dagger$.
\\
Note that $\hat{F}$ is not unitary, but that the Fermi-state is normalized.
To show this, consider
\begin{align}
\hat{F}^\dagger\hat{F} &= \prod_{|\textbf{k}'|<k_\tecxt\text{F}, s'} \hat{c}_{\textbf{k}'s'} \prod_{|\textbf{k}|<k_\tecxt\text{F}, s} \hat{c}_{\textbf{k}s}^\dagger = \prod_{|\textbf{k}| < k_\tecxt\text{F}, s} \hat{c}_{\textbf{k}s} \hat{c}_{\textbf{k}s}^\dagger = \prod_{|\textbf{k}| < k_\tecxt\text{F}, s}  \kla\left( 1 - \hat{c}_{\textbf{k}s}^\dagger \hat{c}_{\textbf{k}s}  \right) ,
\end{align}
where we have disregarded the global sign.
When applied to the vacuum $\Ket | 0 \rangle$ the $\hat{c}$-terms do not contribute, giving $\hat{F}^\dagger \Ket | \psi_\tecxt\text{F} \rangle = \hat{F}^\dagger\hat{F} \Ket | 0 \rangle = \Ket | 0 \rangle$.
%In this sense, we may consider $\hat{F}$ to be unitary.
We want to calculate the expected energy in this state.
We will make use of the following commutation relations
\begin{align}
\klam\left[ \hat{c}_{\textbf{k}\uparrow}, \hat{b}_{\textbf{q}}^\dagger \right] &= \klam\left[ \hat{c}_{\textbf{k}\uparrow}, \hat{c}_{\textbf{q}\uparrow}^\dagger \hat{c}_{-\textbf{q}\downarrow}^\dagger  \right] = \klammer\left\{ \hat{c}_{\textbf{k}\uparrow}, \hat{c}_{\textbf{q}\uparrow}^\dagger \right\} \hat{c}_{-\textbf{q}\downarrow}^\dagger - \hat{c}_{\textbf{q}\uparrow}^\dagger \klammer\left\{ \hat{c}_{\textbf{k}\uparrow}, \hat{c}_{-\textbf{q}\downarrow}^\dagger \right\} = \delta_{\textbf{k}\textbf{q}} \hat{c}_{-\textbf{k}\downarrow},\\
\klam\left[ \hat{c}_{\textbf{k}\downarrow}, \hat{b}_{\textbf{q}}^\dagger \right] &= \klam\left[ \hat{c}_{\textbf{k}\downarrow}, \hat{c}_{\textbf{q}\uparrow}^\dagger \hat{c}_{-\textbf{q}\downarrow}^\dagger  \right] = \klammer\left\{ \hat{c}_{\textbf{k}\downarrow}, \hat{c}_{\textbf{q}\uparrow}^\dagger \right\} \hat{c}_{-\textbf{q}\downarrow}^\dagger - \hat{c}_{\textbf{q}\uparrow}^\dagger \klammer\left\{ \hat{c}_{\textbf{k}\downarrow}, \hat{c}_{-\textbf{q}\downarrow}^\dagger \right\} = -\delta_{\textbf{k},-\textbf{q}} \hat{c}_{-\textbf{k}\uparrow},\\
\klam\left[ \hat{c}_{\textbf{k}s}^\dagger, \hat{b}_{\textbf{q}}^\dagger \right] &= 0.
\end{align}
Using these, we may calculate the commutators
\begin{align}
\klam\left[ \hat{c}_{\textbf{k}\uparrow}, \hat{F} \right] &= \bigg[ \hat{c}_{\textbf{k}\uparrow}, \prod_{|\textbf{q}| < k_\tecxt\text{F}} \hat{b}_{\textbf{q}}^\dagger \bigg] = \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}^\dagger \klam\left[ \hat{c}_{\textbf{k}\uparrow}, \hat{b}_\textbf{k}^\dagger \right] = \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}^\dagger \hat{c}_{-\textbf{k}\downarrow} ,\\
\klam\left[ \hat{c}_{\textbf{k}\downarrow}, \hat{F} \right] &= \bigg[ \hat{c}_{\textbf{k}\downarrow}, \prod_{|\textbf{q}| < k_\tecxt\text{F}} \hat{b}_{\textbf{q}}^\dagger \bigg] = \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq -\textbf{k}} \hat{b}_{\textbf{q}}^\dagger \klam\left[ \hat{c}_{\textbf{k}\downarrow}, \hat{b}_{-\textbf{k}}^\dagger \right] = - \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq -\textbf{k}} \hat{b}_{\textbf{q}}^\dagger \hat{c}_{-\textbf{k}\uparrow},\\
\klam\left[ \hat{c}_{\textbf{k}s}^\dagger, \hat{F} \right] &= 0.
\end{align}
Note that these relations only hold for $|\textbf{k}| < k_\tecxt\text{F}$, otherwise all commutators simply vanish.
\\
The diagonalized Hamiltonian from \autoref{eqn:diagonalized hamiltonian} is 
\begin{align}
\hat{H} &= \sum_{\mathbf{k}s} E_{\mathbf{k}} \hat{\gamma}_{\mathbf{k}s}^\dagger \hat{\gamma}_{\mathbf{k}s} + \sum_{\textbf{k}}\kla\left(  \epsilon_\textbf{k} - E_\textbf{k} + \Delta_\textbf{k} \langle \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \rangle \right)
\end{align}
where the new fermionic operators are given by
\begin{align}
\hat{\gamma}_{\textbf{k}\uparrow} &= \alpha_\textbf{k} \hat{c}_{\textbf{k}\uparrow} - \beta_\textbf{k} \hat{c}_{-\textbf{k}\downarrow}^\dagger,\\
\hat{\gamma}_{-\textbf{k}\downarrow}^\dagger &= \beta_\textbf{k} \hat{c}_{\textbf{k}\uparrow} - \alpha^\ast_\textbf{k} \hat{c}_{-\textbf{k}\downarrow}^\dagger.
\end{align}
We first evaluate the expectation values inside the constant term and $\Delta_\textbf{k}$.
Using $\klam\left[ {\hat{b}_{\textbf{k}}, \hat{b}_{\textbf{k}'}^\dagger } \right] \Ket | {0} \rangle = \delta_{\textbf{k}\textbf{k}'} \Ket | {0} \rangle$, we find $\klam\left[ {\hat{b}_{\textbf{k}}, \hat{F}} \right] \Ket | {0} \rangle = 0$.
Thus
\begin{align}
\bra\langle \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow} \rangle_{\psi_\tecxt\text{F}} &= \braket\langle \psi_\tecxt\text{F} | \hat{b}_{\mathbf{k}} | \psi_\tecxt\text{F} \rangle = \braket\langle 0 | \hat{F}^\dagger  \hat{b}_{\mathbf{k}} \hat{F} | 0 \rangle = \braket\langle 0 | \hat{F}^\dagger \hat{F} \hat{b}_{\textbf{k}} | 0\rangle = 0 = \bra\langle \hat{c}_{\mathbf{k}\uparrow}^\dagger \hat{c}_{-\mathbf{k}\downarrow}^\dagger \rangle_{\psi_\tecxt\text{F}}^\ast .
\end{align}
This immediately gives 
\begin{align}
\Delta_\textbf{k} &= -\sum_{\textbf{k}'} W_{\textbf{k}\textbf{k}'} \bra\langle  \hat{c}_{-\mathbf{k}\downarrow} \hat{c}_{\mathbf{k}\uparrow} \rangle = 0
\end{align}
for the Fermi state.
The more difficult task is to compute $\bra\langle \hat{\gamma}_{\mathbf{k}s}^\dagger \hat{\gamma}_{\mathbf{k}s} \rangle_{\psi_\tecxt\text{F}}$.
Consider the commutators
\begin{align}
\klam\left[ \hat{\gamma}_{\textbf{k}\uparrow}, \hat{F} \right] &= \klam\left[ \alpha_\textbf{k}^\ast \hat{c}_{\textbf{k}\uparrow}, \hat{F} \right] = \alpha_\textbf{k}^\ast \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}^\dagger \hat{c}_{-\textbf{k}\downarrow} = \klam\left[ \hat{F}^\dagger, \hat{\gamma}_{\textbf{k}\uparrow}^\dagger \right]^\dagger,\\
\klam\left[ \hat{\gamma}_{\textbf{k}\downarrow}, \hat{F} \right] &= \klam\left[ \alpha_{-\textbf{k}}^\ast \hat{c}_{\textbf{k}\downarrow}, \hat{F} \right] = -\alpha_{-\textbf{k}}^\ast \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}^\dagger \hat{c}_{-\textbf{k}\uparrow} = \klam\left[ \hat{F}^\dagger, \hat{\gamma}_{\textbf{k}\downarrow}^\dagger \right]^\dagger .
\end{align}
We may now compute the expectation values for $|\textbf{k}| < k_\tecxt\text{F}$
\begin{align}
\bra\langle \hat{\gamma}_{\mathbf{k}\uparrow}^\dagger \hat{\gamma}_{\mathbf{k}\uparrow} \rangle_{\psi_\tecxt\text{F}} &= \braket\langle 0 | \hat{F}^\dagger \hat{\gamma}_{\mathbf{k}\uparrow}^\dagger \hat{\gamma}_{\mathbf{k}\uparrow} \hat{F} | 0 \rangle = \\
&= \braket\langle 0 | \kla\Big( \hat{\gamma}_{\mathbf{k}\uparrow}^\dagger \hat{F}^\dagger + \alpha_\textbf{k} \hat{c}_{-\textbf{k}\downarrow}^\dagger \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}  \Big) \hat{F} \hat{\gamma}_{\mathbf{k}\uparrow} | 0 \rangle = \\
&= (-\beta_\textbf{k}) (-\beta_\textbf{k}^\ast) \braket\langle 0 |  \hat{c}_{-\textbf{k}\downarrow} \hat{F}^\dagger\hat{F} \hat{c}_{-\textbf{k}\downarrow}^\dagger | 0 \rangle = \\
&= |\beta_\textbf{k}|^2  \braket\langle 0 | \hat{c}_{-\textbf{k}\downarrow} \prod_{|\textbf{q}| < k_\tecxt\text{F},s} \kla\left( 1 - \hat{c}_{\textbf{q}s}^\dagger \hat{c}_{\textbf{q}s} \right) \hat{c}_{-\textbf{k}\downarrow}^\dagger | 0 \rangle = \\
&= |\beta_\textbf{k}|^2 \braket\langle 0 | \hat{c}_{-\textbf{k}\downarrow} \kla\left( 1 - \hat{c}_{-\textbf{k}\downarrow}^\dagger \hat{c}_{-\textbf{k}\downarrow}  \right) \hat{c}_{-\textbf{k}\downarrow}^\dagger  | 0 \rangle = \\
&= |\beta_\textbf{k}|^2 \braket\langle 0 | \hat{c}_{-\textbf{k}\downarrow} \hat{c}_{-\textbf{k}\downarrow} \hat{c}_{-\textbf{k}\downarrow}^\dagger \hat{c}_{-\textbf{k}\downarrow}^\dagger   | 0 \rangle = 0,\\
\bra\langle \hat{\gamma}_{\mathbf{k}\downarrow}^\dagger \hat{\gamma}_{\mathbf{k}\downarrow} \rangle_{\psi_\tecxt\text{F}} &= \braket\langle 0 | \hat{F}^\dagger \hat{\gamma}_{\mathbf{k}\downarrow}^\dagger \hat{\gamma}_{\mathbf{k}\downarrow} \hat{F} | 0 \rangle = \\
&= \braket\langle 0 | \kla\Big( \hat{\gamma}_{\mathbf{k}\downarrow}^\dagger \hat{F}^\dagger - \alpha_{-\textbf{k}} \hat{c}_{-\textbf{k}\uparrow}^\dagger \prod_{|\textbf{q}| < k_\tecxt\text{F}, \textbf{q}\neq \textbf{k}} \hat{b}_{\textbf{q}}  \Big) \hat{F} \hat{\gamma}_{\mathbf{k}\downarrow} | 0 \rangle = \\
&= (\beta_{-\textbf{k}}) (-\beta_{-\textbf{k}}^\ast) \braket\langle 0 |  \hat{c}_{-\textbf{k}\uparrow} \hat{F}^\dagger\hat{F} \hat{c}_{-\textbf{k}\uparrow}^\dagger | 0 \rangle = 0.
\end{align}
For $|\textbf{k}| \ge k_\tecxt\text{F}$ we do not have to consider potential contributions from the commutators, so an analogous calculation leads to the same result.
Thus the expectation value of the energy in the Fermi-state is simply
\begin{align}
\bra\langle \hat{H} \rangle_{\psi_\tecxt\text{F}} &= \sum_\textbf{k} \kla\left( \epsilon_\textbf{k} - E_\textbf{k} \right) = \sum_\textbf{k} \kla\left( \epsilon_\textbf{k} - \sqrt{\epsilon_\textbf{k}^2 + |\Delta_\textbf{k}|^2} \right) = 0.
\end{align}
For the BCS-state $\Ket | \psi_\tecxt\text{BCS} \rangle$ the $\hat{\gamma}^\dagger\hat{\gamma}$-terms vanish due to $\hat{\gamma}_{\textbf{k}s} \Ket | \psi_\tecxt\text{BCS} \rangle = 0$.
Thus we only need to consider the expectation value of $\hat{b}_\textbf{k}^\dagger$
\begin{align}
\bra\langle \hat{b}_\textbf{k}^\dagger \rangle_{\psi_\tecxt\text{BCS}} &= \braket\langle \psi_\tecxt\text{BCS} | \hat{b}_\textbf{k}^\dagger | \psi_\tecxt\text{BCS} \rangle = \braket\langle 0 | \hat{V}^\dagger \hat{b}_\textbf{k}^\dagger \hat{V} | 0 \rangle = \\
&= \braket\langle 0 | \prod_\textbf{q} \left( \alpha_\textbf{q} + \beta_\textbf{q}^\ast \hat{b}_\textbf{q} \right) \hat{b}_\textbf{k}^\dagger \hat{V} | 0 \rangle = \\
&= \braket\langle 0 | \left( \alpha_\textbf{k} + \beta_\textbf{k}^\ast \hat{b}_\textbf{k} \right) \hat{b}_\textbf{k}^\dagger  \prod_{\textbf{q} \neq \textbf{k}} \left( \alpha_\textbf{q} + \beta_\textbf{q}^\ast \hat{b}_\textbf{q} \right) \hat{V}  | 0 \rangle = \\
&= \beta^\ast_\textbf{k}\braket\langle 0 | \kla\left( 1 - \hat{c}_{\textbf{k}\uparrow}^\dagger \hat{c}_{\textbf{k}\uparrow} +  \hat{c}_{-\textbf{k}\downarrow}^\dagger \hat{c}_{-\textbf{k}\downarrow} \right)  \prod_{\textbf{q} \neq \textbf{k}} \left( \alpha_\textbf{q} + \beta_\textbf{q}^\ast \hat{b}_\textbf{q} \right) \hat{V} | 0 \rangle  = \\
&= \beta_\textbf{k}^\ast \braket\langle 0 | \left( \alpha_\textbf{k}^\ast + \beta_\textbf{k} \hat{b}_\textbf{k}^\dagger \right) \prod_{\textbf{q} \neq \textbf{k}} \left( \alpha_\textbf{q} + \beta_\textbf{q}^\ast \hat{b}_\textbf{q} \right) \left( \alpha_\textbf{q}^\ast + \beta_\textbf{q} \hat{b}_\textbf{q}^\dagger \right) | 0 \rangle = \\
&= \alpha_\textbf{k}^\ast \beta_\textbf{k}^\ast \braket\langle 0 | \prod_{\textbf{q} \neq \textbf{k}} \klam\left[ |\alpha_\textbf{q}^\ast|^2 + |\beta_\textbf{q}|^2 \kla\left( 1 - \hat{c}_{\textbf{q}\uparrow}^\dagger \hat{c}_{\textbf{q}\uparrow} +  \hat{c}_{-\textbf{q}\downarrow}^\dagger \hat{c}_{-\textbf{q}\downarrow} \right) \right] | 0 \rangle = \\
&= \alpha_\textbf{k}^\ast \beta_\textbf{k}^\ast \braket\langle 0 | \prod_{\textbf{q} \neq \textbf{k}} \klam\left[ |\alpha_\textbf{q}|^2 + |\beta_\textbf{q}|^2 \right] | 0 \rangle = \alpha_\textbf{k}^\ast \beta_\textbf{k}^\ast.
\end{align}
Before we combine our results in the energy expectation value, consider the gap
\begin{align}
\Delta_\textbf{k} &= -\sum_{\textbf{k}'} W_{\textbf{k}\textbf{k}'} \bra\langle \hat{b}_{\textbf{k}'} \rangle_{\psi_\tecxt\text{BCS}} = -\sum_{\textbf{k}'} W_{\textbf{k}\textbf{k}'} \kla\left( \bra\langle \hat{b}_{\textbf{k}'}^\dagger \rangle_{\psi_\tecxt\text{BCS}} \right)^\ast = \\
&= -\sum_{\textbf{k}'} W_{\textbf{k}\textbf{k}'} \alpha_{\textbf{k}'} \beta_{\textbf{k}'} = -\sum_{\mathbf{k}'} W_{\mathbf{k}\textbf{k}'} \frac{\Delta_{\mathbf{k}'} \kla\left( E_{\mathbf{k}'} - \epsilon_{\mathbf{k}'} \right)}{2E_{\mathbf{k}'} (E_{\mathbf{k}'} - \epsilon_{\mathbf{k}'})} = \\
&= -\frac{1}{2}\sum_{\mathbf{k}'} W_{\mathbf{k}\textbf{k}'} \frac{\Delta_{\mathbf{k}'}}{\sqrt{\epsilon_{\mathbf{k}'}^2 + |\Delta_{\mathbf{k}'}|^2}}.
\end{align}
This is a self-consistency equation for $\Delta_\textbf{k}$.
Before attempting a solution, note that the total energy now becomes
\begin{align}
E_\tecxt\text{BCS} &= \sum_\textbf{k} \kla\left( \epsilon_\textbf{k} - E_\textbf{k} + \Delta_\textbf{k} \langle \hat{b}_{\mathbf{k}}^\dagger \rangle \right) = \sum_\textbf{k} \kla\left( \epsilon_\textbf{k} - E_\textbf{k} + \Delta_\textbf{k} \alpha_\textbf{k}^\ast \beta_\textbf{k}^\ast \right) = \\
&= \sum_\textbf{k} \kla\left( \epsilon_\textbf{k} - E_\textbf{k} + \frac{1}{2} \Delta_\textbf{k} \frac{\Delta_\textbf{k}^\ast}{E_\textbf{k}} \right) = \sum_\textbf{k} \frac{\epsilon_\textbf{k} E_\textbf{k} - E_\textbf{k}^2 + \frac{1}{2} |\Delta_\textbf{k}|^2}{E_\textbf{k}} = \\
&= \sum_\textbf{k} \frac{\epsilon_\textbf{k}\kla\left( \sqrt{\epsilon_\textbf{k}^2 + |\Delta_\textbf{k}|^2} - \epsilon_\textbf{k} \right) - \frac{1}{2}|\Delta_\textbf{k}|^2}{\sqrt{\epsilon_\textbf{k}^2 + |\Delta_\textbf{k}|^2}}.
\end{align}
Thus the BCS state reduces the total energy regardless of $\Delta_\textbf{k}$ (except for $\Delta\equiv 0$).
\\(Das hat mich überrascht, aber vielleicht ändert sich ja etwas für $T>0$).\\
To see this, define $x := \frac{|\Delta_\textbf{k}|^2}{\epsilon_\textbf{k}^2}$.
Then the claim of $E_\tecxt\text{BCS} < 0$ is equivalent to $\sqrt{1 + x} < 1+\frac{x}{2}$ for $x>0$.
However, this is follows directly from
\begin{align}
1 + x &< 1+ x + \frac{x^2}{4} = \kla\left( 1 + \frac{x}{2} \right)^2\\
\Rightarrow\ \ \ \sqrt{1 + x} &< 1 + \frac{x}{2}.
\end{align}
We may expand the total energy in the limit of $|\Delta_\textbf{k}| \ll \epsilon_\textbf{k}$.
Then we find
\begin{align}
E_\tecxt\text{BCS} &= \sum_\textbf{k} \frac{\epsilon_\textbf{k}^2 \kla\left( \sqrt{1 + \frac{|\Delta_\textbf{k}|^2}{\epsilon_\textbf{k}^2}} - 1 \right) - \frac{|\Delta_\textbf{k}|^2}{2}}{E_\textbf{k}} \sim \\
&\sim \sum_\textbf{k}  \frac{\epsilon_\textbf{k}^2 \kla\left( \frac{|\Delta_\textbf{k}|^2}{2\epsilon_\textbf{k}^2} - \frac{1}{8} \frac{|\Delta_\textbf{k}|^4}{\epsilon_\textbf{k}^4} \right) - \frac{|\Delta_\textbf{k}|^2}{2}}{E_\textbf{k}} = -\frac{1}{8} \sum_\textbf{k} \frac{|\Delta_\textbf{k}|^4}{\epsilon_\textbf{k}^2 E_\textbf{k}}
\end{align}

To solve this equation, we can exploit the similarity between $\epsilon_{\textbf{k}}$ and $W_{\textbf{k}\textbf{k}'}$.
Without loss of generality, we may write $\Delta_{\textbf{k}} =: \epsilon_{\textbf{k}} \delta_{\textbf{k}}$.
Then we find
\begin{align}
-2\epsilon_{\textbf{k}} \delta_{\textbf{k}} &= \sum_{\textbf{k}'} \frac{J_{\textbf{k}} + J_{\textbf{k}'}}{\sqrt{1 + |\delta_{\textbf{k}'}|^2}} \delta_{\textbf{k}'} \frac{\epsilon_{\textbf{k}'}}{|\epsilon_{\textbf{k}'}|}.
\end{align}
Note that this is equivalent to
\begin{align}
-\delta_{\textbf{k}} \kla\left( J_{\textbf{k}} + 2\mu  \right) &= J_{\textbf{k}} \sum_{\textbf{k}'} \frac{\delta_{\textbf{k}'}}{\sqrt{1 + |\delta_{\textbf{k}'}|^2}} \frac{\epsilon_{\textbf{k}'}}{|\epsilon_{\textbf{k}'}|} + \sum_{\textbf{k}'} \frac{\delta_{\textbf{k}'} J_{\textbf{k}'}}{\sqrt{1 + |\delta_{\textbf{k}'}|^2}}  \frac{\epsilon_{\textbf{k}'}}{|\epsilon_{\textbf{k}'}|} \equiv J_{\textbf{k}} f + g
\end{align}
where $f$ and $g$ are both constant with respect to $\textbf{k}$.
Then our solution becomes
\begin{align}
\delta_{\textbf{k}} &= -\frac{J_{\textbf{k}}f + g}{J_{\textbf{k}} + 2\mu}.
\end{align}
This is equivalent to 
\begin{align}
\Delta_{\textbf{k}} &= \epsilon_{\textbf{k}} \delta_{\textbf{k}} = -\frac{1}{2} \left( J_{\textbf{k}} f+g\right).
\end{align}
We find a set of two coupled self-consistency equations for $f$ and $g$ from their definition
\begin{align}
f &= \sum_{\textbf{k}} \frac{\delta_{\textbf{k}}}{\sqrt{1 + |\delta_{\textbf{k}}|^2}}  \frac{\epsilon_{\textbf{k}}}{|\epsilon_{\textbf{k}}|} = - \sum_{\textbf{k}} \frac{J_{\textbf{k}}f + g}{\sqrt{\left[ J_{\textbf{k}}+2\mu\right]^2 + \left[ J_{\textbf{k}}f + g \right]^2}},\\
g &= \sum_{\textbf{k}} \frac{\delta_{\textbf{k}} J_{\textbf{k}}}{\sqrt{1 + |\delta_{\textbf{k}}|^2}} \frac{\epsilon_{\textbf{k}}}{|\epsilon_{\textbf{k}}|} = - \sum_{\textbf{k}} \frac{\left[ J_{\textbf{k}}f + g \right] J_{\textbf{k}}}{\sqrt{\left[ J_{\textbf{k}} + 2\mu \right]^2 + \left[ J_{\textbf{k}}f + g\right]^2}}.
\end{align}
We can reduce the number of independent parameters by writing
%\begin{align}
%0 &= f + \sum_{\textbf{k}} \frac{1}{\sqrt{ 1 + \left[ \frac{\cos(k_x) + \cos(k_y) + \tilde{\mu}}{[\cos(k_x) + \cos(k_y)] f + \tilde{g}} \right]^2}},\\
%0 &= \tilde{g} + \sum_{\textbf{k}} \frac{\cos(k_x) + \cos(k_y)}{\sqrt{ 1 + \left[ \frac{\cos(k_x) + \cos(k_y) + \tilde{\mu}}{[\cos(k_x) + \cos(k_y)] f + \tilde{g}} \right]^2}}
%\end{align}
\begin{align}
0 &= f + \sum_{\textbf{k}} \frac{[\cos(k_x) + \cos(k_y)] f + \tilde{g}}{\sqrt{\left[ [\cos(k_x) + \cos(k_y)] f + \tilde{g} \right]^2 + \left[ \cos(k_x) + \cos(k_y) + \tilde{\mu} \right]^2}} \equiv F(f,\tilde{g}),\\
0 &= \tilde{g} + \sum_{\textbf{k}} \frac{\left([\cos(k_x) + \cos(k_y)] f + \tilde{g} \right) [\cos(k_x) + \cos(k_y)] }{\sqrt{\left[ [\cos(k_x) + \cos(k_y)] f + \tilde{g} \right]^2 + \left[ \cos(k_x) + \cos(k_y) + \tilde{\mu} \right]^2}} \equiv G(f,\tilde{g}),
\end{align}
where $\tilde{g} = \frac{2g}{J_0}$ and $\tilde{\mu} = \frac{4\mu }{J_0}$.
Then the solution takes the form
\begin{align}
\Delta_{\textbf{k}} &= -\frac{J_0}{4} \left( [\cos(k_x) + \cos(k_y)] f + \tilde{g} \right).
\end{align}
The remaining equations would in principal be solvable using implicit root finding.
However, this requires more information about the range of the $\textbf{k}$-summation.
Finally, the expectation value of the bosonic operator becomes
\begin{align}
\langle \hat{b}_\textbf{k} \rangle &= \frac{\Delta_\textbf{k}/2}{\sqrt{\epsilon_\textbf{k}^2 + |\Delta_\textbf{k}|^2}} = \frac{1}{2} \frac{[\cos(k_x) + \cos(k_y)] f + \tilde{g}}{\sqrt{[ [\cos(k_x) + \cos(k_y)] f + \tilde{g} ]^2 + [ \cos(k_x) + \cos(k_y) + \tilde{\mu} ]^2}}.
\end{align}
(Die beiden übrigen Selbstkonsitenzgleichungen kann man zumindest numerisch lösen, bzw. es existieren Lösungen $f,g\neq 0$ aus Grenzwertbetrachtungen. Allerdings weiß ich nicht, wie die Summe aussieht. Was genau sind dort eigentlich die Grenzen?)
