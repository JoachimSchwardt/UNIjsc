\section{Hamiltonian}
Hamiltonian $\hat{H} = \hat{H}_0 + \hat{W}$ where
\begin{align}
\hat{H}_0 &= \sum_{\textbf{k}s} E_\textbf{k} \hat{a}_{\textbf{k}s}^\dagger  \hat{a}_{\textbf{k}s},\\
\hat{W} &= \sum_{\textbf{kpq}} V_\textbf{kpq}^{\uparrow\downarrow}  \hat{a}_{\textbf{k} + \textbf{q} \uparrow}^\dagger \hat{a}_{\textbf{p} - \textbf{q} \downarrow}^\dagger \hat{a}_{\textbf{p} \downarrow} \hat{a}_{\textbf{k} \uparrow}
\end{align}
and $V_\textbf{kpq}^{\uparrow\downarrow} = -E_\textbf{k} - E_\textbf{p} - E_\mathbf{k+q} - E_\mathbf{p-q}$.
Here the operators $\hat{a}$ may represent doublons $\hat{d}_{\textbf{k}s}$ or holons $\hat{h}_{\textbf{k}s}$ defined by
\begin{align}
\begin{pmatrix}
\hat{d}_{\textbf{k}s} \\ \hat{h}_{\textbf{k}s}^\dagger
\end{pmatrix} &= \begin{pmatrix}
\cos\phi_\textbf{k} & \sin\phi_\textbf{k} \\ -\sin\phi_\textbf{k} & \cos\phi_\textbf{k}
\end{pmatrix} \begin{pmatrix}
\hat{f}_{\textbf{k}s} \\ \hat{e}_{\textbf{k}s}^\dagger
\end{pmatrix}
\end{align}
where $\hat{c}_{\textbf{k}\uparrow} = \hat{c}_{\textbf{k}\uparrow} \hat{n}_{\textbf{k}\downarrow} + \hat{c}_{\textbf{k}\uparrow} (1 - \hat{n}_{\textbf{k}\downarrow}) =: \hat{f}_{\textbf{k}\uparrow} + \hat{e}_{\textbf{k}\uparrow}^\dagger$.

\begin{align}
\klammer\left\{ c_{ks}, f_{k's'} \right\} &= \klammer\left\{ c_{ks}, c_{k's'} n_{k'\bar{s}'} \right\} = c_{k's'} \klam\left[ n_{k'\bar{s}'}, c_{ks} \right] + \klammer\left\{ c_{ks}, c_{k's'} \right\} n_{k'\bar{s}'} = c_{ks}c_{k\bar{s}} \delta_{kk'} \delta_{s\bar{s}'}, \\
\klammer\left\{ c_{ks}^\dagger, f_{k's'} \right\} &= \klammer\left\{ c_{ks}^\dagger, c_{k's'} n_{k'\bar{s}'} \right\} = c_{k's'} \klam\left[ n_{k'\bar{s}'}, c_{ks}^\dagger \right] + \klammer\left\{ c_{ks}^\dagger, c_{k's'} \right\} n_{k'\bar{s}'} = \\
&= c_{k\bar{s}} c_{ks}^\dagger \delta_{kk'} \delta_{s\bar{s}'} + n_{k\bar{s}} \delta_{kk'} \delta_{ss'}, \\
\left[ n_{ks}, f_{k's'} \right] &= \left( c_{ks}^\dagger \klammer\left\{ c_{ks}, f_{k's'} \right\} - \klammer\left\{ c_{ks}^\dagger, f_{k's'} \right\} c_{ks} \right) = \\
&= \kla\left( c_{ks}^\dagger c_{ks}c_{k\bar{s}} \delta_{s\bar{s}'} - c_{k\bar{s}} c_{ks}^\dagger \delta_{s\bar{s}'} c_{ks} - n_{k\bar{s}} \delta_{ss'} c_{ks}  \right) \delta_{kk'} = \\
&= -c_{ks} n_{k\bar{s}} \delta_{ss'} \delta_{kk'} ,\\
\klammer\left\{ f_{ks}, f_{k's'} \right\} &= \klammer\left\{ c_{ks} n_{k\bar{s}}, f_{k's'} \right\} = c_{ks} \klam\left[ n_{k\bar{s}}, f_{k's'} \right] - \klammer\left\{ c_{ks}, f_{k's'} \right\} n_{k\bar{s}} = \\
&= c_{ks} \left( - c_{k\bar{s}} n_{ks} \delta_{s\bar{s}'} \right) \delta_{kk'} - c_{ks}c_{k\bar{s}} \delta_{kk'} \delta_{s\bar{s}'} = \\
&= 2c_{k\bar{s}} c_{ks} \delta_{kk'} \delta_{s\bar{s}'},\\
\klammer\left\{ f_{ks}^\dagger, f_{k's'} \right\} &= \klammer\left\{ c_{ks}^\dagger n_{k\bar{s}} , f_{k's'} \right\} = c_{ks}^\dagger \klam\left[ n_{k\bar{s}}, f_{k's'} \right] - \klammer\left\{ c_{ks}^\dagger, f_{k's'} \right\} n_{k\bar{s}} = \\
&= c_{ks}^\dagger \left( -c_{k\bar{s}} n_{ks} \delta_{s\bar{s}'} \right) \delta_{kk'} - c_{k\bar{s}} c_{ks}^\dagger \delta_{kk'} \delta_{s\bar{s}'} n_{k\bar{s}} - n_{k\bar{s}} \delta_{kk'} \delta_{ss'} n_{k\bar{s}} = \\
&= \kla\left( c_{ks}^\dagger c_{k\bar{s}} \delta_{s\bar{s}'} - n_{k\bar{s}} \delta_{ss'} \right) \delta_{kk'}, \\
\klammer\left\{ e_{ks}, f_{k's'} \right\} &= \klammer\left\{ \kla\left( c_{ks} - f_{ks} \right)^\dagger, f_{k's'} \right\} = \\
&= c_{k\bar{s}} c_{ks}^\dagger \delta_{kk'} \delta_{s\bar{s}'} + n_{k\bar{s}} \delta_{kk'} \delta_{ss'} - \left( c_{ks}^\dagger c_{k\bar{s}} \delta_{s\bar{s}'} - n_{k\bar{s}} \delta_{ss'} \right) \delta_{kk'} = \\
&= 2\kla\left( c_{k\bar{s}} c_{ks}^\dagger \delta_{s\bar{s}'} + n_{k\bar{s}} \delta_{ss'} \right) \delta_{kk'},\\
\klammer\left\{ e_{ks}^\dagger, f_{k's'} \right\} &= \klammer\left\{ c_{ks} - f_{ks}, f_{k's'} \right\} = \\
&= c_{ks}c_{k\bar{s}} \delta_{kk'} \delta_{s\bar{s}'} - 2c_{k\bar{s}} c_{ks} \delta_{kk'} \delta_{s\bar{s}'} = 3c_{ks} c_{k\bar{s}}  \delta_{kk'} \delta_{s\bar{s}'},\\
\klammer\left\{ e_{ks}^\dagger, e_{k's'}^\dagger \right\} &= \klammer\left\{ c_{ks} - f_{ks}, c_{k's'} - f_{k's'} \right\} = \\
&= \delta_{kk'} \kla\left( -c_{ks'}c_{k\bar{s}'} \delta_{s\bar{s}'} - c_{ks}c_{k\bar{s}} \delta_{s\bar{s}'} + 2c_{k\bar{s}} c_{ks} \delta_{s\bar{s}'} \right) = 2c_{k\bar{s}} c_{ks} \delta_{s\bar{s}'} \delta_{kk'} ,\\
\klammer\left\{ e_{ks}^\dagger, e_{k's'} \right\} &= \klammer\left\{ c_{ks} - f_{ks}, c_{k's'}^\dagger - f_{k's'}^\dagger \right\} = \\
&= \delta_{kk'} \kla\left( \delta_{ss'} + c_{k\bar{s}}^\dagger c_{ks} \delta_{s\bar{s}'} - n_{k\bar{s}} \delta_{ss'} - 2c_{ks} c_{k\bar{s}}^\dagger \delta_{s\bar{s}'} - 2n_{k\bar{s}} \delta_{ss'} \right) = \\
&= \kla\left( 1 + 3c_{k\bar{s}}^\dagger c_{ks} \right) \delta_{s\bar{s}'} \delta_{kk'} - 3n_{k\bar{s}} \delta_{ss'} \delta_{kk'}.
\end{align}
