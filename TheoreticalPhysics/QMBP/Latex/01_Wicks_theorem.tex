\section{Explicit calculation of correlation functions for free fermions}
Consider the states $\Ket | \psi \rangle := \hat{f}_{N_p}^\dagger \dots \hat{f}_{1}^\dagger \Ket | 0 \rangle$ and $\Ket | \phi \rangle := \hat{d}_{N_p}^\dagger \dots \hat{d}_{1}^\dagger \Ket | 0 \rangle$ where $\hat{f}_{\alpha}^\dagger = \sum_{j=1}^N \psi_\alpha(j) \hat{c}_j^\dagger$ and $\hat{d}_{\alpha}^\dagger = \sum_{j=1}^N \phi_\alpha(j) \hat{c}_j^\dagger$.
We want to show that $\brak\langle \phi | \psi \rangle = \det \mathbb{O}$ where $\mathbb{O}_{\alpha\beta} = \brak\langle \phi_\alpha | \psi_\beta \rangle$.
First, we compute the anti-commutator 
\begin{align}
\klammer\left\{ \hat{d}_\alpha, \hat{f}_\beta^\dagger \right\} &= \sum_{jk} \phi_\alpha^\ast(j)\psi_\beta(k) \klammer\left\{ \hat{c}_j, \hat{c}_k^\dagger \right\} = \sum_j \phi_\alpha^\ast(j) \psi_\beta(j) = \brak\langle \phi_\alpha | \psi_\beta \rangle = \mathbb{O}_{\alpha\beta}. 
\end{align}

Then we find
\begin{align}

&= \langle 0 | \hat{d}_{N_p}\dots \hat{d}_2 \kla\left( \klammer\left\{ \hat{d}_1, \hat{f}_1^\dagger \right\} - \hat{f}_1^\dagger \hat{d}_1 \right) \hat{f}_2^\dagger \dots \hat{f}_{N_p}^\dagger | 0 \rangle = \\

Now using induction we may assume that the remaining terms are all given by the determinant of the corresponding smaller matrices $\mathbb{O}^{(N_p-1)}_{(1,\alpha)}$ where the first row and $\alpha$-th column are removed.

\begin{align}

\end{align}

\section{Green's functions and linear response theory}
Consider a system described by an unperturbed time-independent Hamiltonian $\hat{H}_0$ and let $\hat{G}_0^\tecxt\text{ret}(t,t') := -\i \Theta (t - t') \e^{-\i \hat{H}_0 (t-t')}$.
Then we find
\begin{align}
\i\partial_t \hat{G}_0^\tecxt\text{ret} (t,t') &= \e^{-\i\hat{H}_0(t-t')} \delta(t-t')  - \i \hat{H}_0 \Theta(t-t') \e^{-\i\hat{H}_0(t-t')} = \delta(t-t') \e^{\hat{0}} + \hat{H}_0 \hat{G}_0^\tecxt\text{ret}(t,t')
\end{align}
and thus $\hat{G}_0^\tecxt\text{ret}$ satisfies the unperturbed Schrödinger equation in the sense that
\begin{align}
\kla\left( \i\partial_t - \hat{H}_0 \right) \hat{G}_0^\tecxt\text{ret}(t,t') &= \delta(t-t') \hat{I}.
\end{align}
Note that $\Ket | \psi_0(t) \rangle = \e^{-\i\hat{H}_0(t-t')} \Ket | \psi_0(t') \rangle$ and consider a perturbation of the form $\hat{H} = \hat{H}_0 + \hat{V}(t)$.
Then to first order in $\hat{V}$ the state
\begin{align}
\Ket | \psi(t) \rangle &:= \Ket | \psi_0(t) \rangle + \intr\int_{\mathbb{R}^{ }} \hat{G}_0^\tecxt\text{ret}(t,t') \hat{V}(t') \Ket | \psi_0(t') \rangle \, \mathrm{d}^{ }t'
\end{align}
satisfies the differential equation $\kla\left( \i\partial_t - \hat{H}_0 - \hat{V}(t) \right) \Ket | \psi(t) \rangle = 0$, since
\begin{align}
\kla\left( \i\partial_t - \hat{H}_0 \right) \Ket | \psi(t) \rangle &= \kla\left( \i\partial_t - \hat{H}_0 \right) \Ket | \psi_0(t) \rangle + \intr\int_{\mathbb{R}^{ }} \delta(t-t') \hat{V}(t') \Ket | \psi_0(t') \rangle \, \mathrm{d}^{ }t' = \\
&= \hat{V}(t) | \psi_0(t) \rangle = \hat{V}(t) \Ket | \psi(t) \rangle + \mathcal{O}\kla\left( \hat{V}^2 \right).
\end{align}
Furthermore, for $t\rightarrow -\infty$ we define $\Ket | \psi_0(t\rightarrow -\infty) \rangle =: \Ket | \psi_0 \rangle$ and find
\begin{align}
\Ket | \psi(t\rightarrow -\infty) \rangle &= \Ket | \psi_0 \rangle + \lim_{t\rightarrow -\infty} \int_{\mathbb{R}^{ }} (-\i)\Theta(t - t') \e^{-\i\hat{H}_0(t-t')} \hat{V}(t') \Ket | \psi_0(t') \rangle  \, \mathrm{d}^{ }t' = \\
&= \Ket | \psi_0 \rangle - \i \lim_{t\rightarrow -\infty} \int_{-\infty}^t \e^{-\i\hat{H}_0(t-t')} \hat{V}(t') \Ket | \psi_0(t') \rangle  \, \mathrm{d}^{ }t' = \Ket | \psi_0 \rangle.
\end{align}
Write $\hat{V}(t) = F(t) \hat{B}$ and define $\delta \bra\langle \hat{A} \rangle(t) := \braket\langle \psi(t) | \hat{A} | \psi(t) \rangle - \braket\langle \psi_0(t) | \hat{A} | \psi_0(t) \rangle$.
Note that
\begin{align}
\braket\langle \psi(t) | \hat{A} | \psi(t) \rangle &= \langle \psi_0(t) | \hat{A} | \psi_0(t) \rangle + \intr\int_{\mathbb{R}^{ }} \braket\langle \psi_0(t) | \hat{A} \hat{G}_0^\tecxt\text{ret}(t,t') \hat{V}(t') | \psi_0(t') \rangle  \, \mathrm{d}^{ }t' + \tecxt\text{h.c.} + \mathcal{O}\kla\left( \hat{V}^2 \right).
\end{align}
Thus, to first order, the deviation simplifies to
\begin{align}
\delta \bra\langle \hat{A} \rangle (t) &\simeq \intr\int_{\mathbb{R}^{ }} \braket\langle \psi_0(t) | \hat{A} \hat{G}_0^\tecxt\text{ret}(t,t') \hat{B} | \psi_0(t') \rangle F(t') \, \mathrm{d}^{ }t' + \tecxt\text{h.c.} =: \intr\int_{\mathbb{R}^{ }} F(t') G_{AB}^\tecxt\text{ret}(t,t') \, \mathrm{d}^{ }t'.
\end{align}
Here we introduced the two-time $AB$ retarded Green's function
\begin{align}
G_{AB}^\tecxt\text{ret}(t,t') &:= \braket\langle \psi_0(t) | \hat{A} \hat{G}_0^\tecxt\text{ret}(t,t') \hat{B} | \psi_0(t') \rangle + \tecxt\text{h.c.} = \\
&= -\i \lim_{t_0 \rightarrow -\infty} \braket\langle \psi_0 | \e^{\i\hat{H}_0 (t - t_0)} \hat{A} \Theta(t-t') \e^{-\i \hat{H}_0(t - t')} \hat{B} \e^{-\i\hat{H}_0(t' - t_0)} | \psi_0 \rangle + \tecxt\text{h.c.} = \\
&= -\i\Theta(t-t') \braket\langle \psi_0 | \hat{A}_I(t) \hat{B}_I(t') | \psi_0 \rangle + \tecxt\text{h.c.} = \\
&= -\i\Theta (t-t') \braket\langle \psi_0 | \klam\left[ \hat{A}_I(t), \hat{B}_I(t') \right] | \psi_0 \rangle
\end{align}
where we defined $\hat{A}_I(t) := \lim_{t_0\rightarrow -\infty} \e^{\i\hat{H}_0 (t-t_0)} \hat{A} \e^{-\i\hat{H}_0(t-t_0)}$.
